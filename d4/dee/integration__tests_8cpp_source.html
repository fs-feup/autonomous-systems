363void read_from_file(std::ifstream &file, double &finalxi, double &finalxf, double &finalyi,
364 double &finalyf, std::vector<Cone> &cone_array,
365 custom_interfaces::msg::Pose &vehicle_pose) {
366 std::string line;
367
368 while (std::getline(file, line)) {
369 std::istringstream iss(line);
370 std::string flag;
371 iss >> flag;
372 if (flag == "P") {
373 // Read initial vehicle state
374 iss >> vehicle_pose.x >> vehicle_pose.y >> vehicle_pose.theta;
375 } else if (flag == "C") {
376 Cone cone;
377 std::string color;
378 iss >> cone.position.x >> cone.position.y >> color;
379 cone.color = common_lib::competition_logic::get_color_enum(color);
380 cone_array.push_back(cone);
381 } else if (flag == "F") {
382 // Read final expected position range
383 iss >> finalxi >> finalxf >> finalyi >> finalyf;
384 } else {
385 // Ignore all other lines with no flags
386 }
387 }
388 file.close();
389}
