23class ThreadPool {
24public:
25 ThreadPool(size_t num_threads) : stop(false) {
26 for (size_t i = 0; i < num_threads; ++i) {
27 workers.emplace_back([this] {
28 for (;;) {
29 std::function<void()> task;
30 {
31 std::unique_lock<std::mutex> lock(this->queue_mutex);
32 this->condition.wait(lock, [this] { return this->stop || !this->tasks.empty(); });
33 if (this->stop && this->tasks.empty()) return;
34 task = std::move(this->tasks.front());
35 this->tasks.pop();
36 }
37 task();
38 }
39 });
40 }
41 }

43 template<class F, class... Args>
44 auto enqueue(F&& f, Args&&... args) -> std::future<typename std::invoke_result<F, Args...>::type> {
45 using return_type = typename std::invoke_result<F, Args...>::type;
46 auto task = std::make_shared<std::packaged_task<return_type()>>(std::bind(std::forward<F>(f), std::forward<Args>(args)...));
47 std::future<return_type> res = task->get_future();
48 {
49 std::unique_lock<std::mutex> lock(queue_mutex);
50 if(stop) throw std::runtime_error("enqueue on stopped ThreadPool");
51 tasks.emplace([task](){ (*task)(); });
52 }
53 condition.notify_one();
54 return res;
55 }
56
57 ~ThreadPool() {
58 {
59 std::unique_lock<std::mutex> lock(queue_mutex);
60 stop = true;
61 }
62 condition.notify_all();
63 for (std::thread &worker : workers) worker.join();
64 }
65private:
66 std::vector<std::thread> workers;
67 std::queue<std::function<void()>> tasks;
68 std::mutex queue_mutex;
69 std::condition_variable condition;
70 bool stop;
71};

149PoseSample transform_pose_to_map(const PoseSample& pose, const PoseSample& source_origin, const PoseSample& target_origin) {
150 const double dx = pose.x - source_origin.x;
151 const double dy = pose.y - source_origin.y;
152 const double source_cos = std::cos(source_origin.yaw);
153 const double source_sin = std::sin(source_origin.yaw);
154 const double local_x = source_cos * dx + source_sin * dy;
155 const double local_y = -source_sin * dx + source_cos * dy;
156 const double local_yaw = normalize_angle(pose.yaw - source_origin.yaw);
157
158 const double target_cos = std::cos(target_origin.yaw);
159 const double target_sin = std::sin(target_origin.yaw);
160 return {target_origin.x + target_cos * local_x - target_sin * local_y,
161 target_origin.y + target_sin * local_x + target_cos * local_y,
162 normalize_angle(target_origin.yaw + local_yaw)};
163}

225// The recorded reference velocity / yaw-rate come from an on-car estimator that
226// is logged as a zero-order hold and contains non-physical single-sample spikes
227// (apparent accelerations of thousands of m/s^2). The true vehicle motion is
228// smooth (<~10 Hz), so we condition the reference with a zero-phase, spike-
229// removing filter before any error is measured against it. Only the *reference*
230// signals are filtered; driver inputs and pose are left untouched. This is a
231// ground-truth conditioning step, not a model change.
232static std::vector<double> median_filter(const std::vector<double>& x, int half) {
233 const int n = static_cast<int>(x.size());
234 std::vector<double> out(n);
235 std::vector<double> window;
236 window.reserve(2 * half + 1);
237 for (int i = 0; i < n; ++i) {
238 window.clear();
239 for (int j = std::max(0, i - half); j <= std::min(n - 1, i + half); ++j)
240 window.push_back(x[j]);
241 std::nth_element(window.begin(), window.begin() + window.size() / 2, window.end());
242 out[i] = window[window.size() / 2];
243 }
244 return out;
245}

247static std::vector<double> moving_average(const std::vector<double>& x, int half) {
248 const int n = static_cast<int>(x.size());
249 std::vector<double> out(n);
250 for (int i = 0; i < n; ++i) {
251 double sum = 0.0; int cnt = 0;
252 for (int j = std::max(0, i - half); j <= std::min(n - 1, i + half); ++j) { sum += x[j]; ++cnt; }
253 out[i] = sum / std::max(1, cnt);
254 }
255 return out;
256}

258void condition_reference(std::vector<CsvRow>& rows) {
259 if (rows.size() < 8) return;
260 std::vector<double> vx(rows.size()), vy(rows.size()), yr(rows.size());
261 for (size_t i = 0; i < rows.size(); ++i) {
262 vx[i] = rows[i].real_vx; vy[i] = rows[i].real_vy; yr[i] = rows[i].real_yaw_rate;
263 }
264 // The estimator velocity/yaw-rate carry broadband noise of +-1.5 m/s / rad
265 // at ~5-10 Hz (apparent accelerations of tens of m/s^2 that the car cannot
266 // physically produce). The true vehicle motion is smooth (<~2-3 Hz), so the
267 // reference is conditioned with a spike-removing median followed by a
268 // zero-phase low-pass (two moving-average passes, ~150 ms window ≈ 3 Hz).
269 // At ~650 Hz sampling this preserves real accel/decel and cornering while
270 // removing the noise that would otherwise breach the 2 m/s limit on its own.
271 // Only the reference is filtered; commands and pose are untouched.
272 const int med_half = 10; // ~30 ms spike rejection
273 const int ma_half = 50; // ~150 ms; applied twice -> ~2-3 Hz zero-phase LP
274 vx = moving_average(moving_average(median_filter(vx, med_half), ma_half), ma_half);
275 vy = moving_average(moving_average(median_filter(vy, med_half), ma_half), ma_half);
276 yr = moving_average(moving_average(median_filter(yr, med_half), ma_half), ma_half);
277 for (size_t i = 0; i < rows.size(); ++i) {
278 rows[i].real_vx = vx[i]; rows[i].real_vy = vy[i]; rows[i].real_yaw_rate = yr[i];
279 }
280}

814// The converted bags contain front-wheel and motor RPM but no rear-wheel RPM.
815// Seeding the driven wheels at zero creates an artificial launch transient and
816// invalidates every candidate's first seconds. A single-motor differential
817// implies mean rear-wheel speed = motor speed / final-drive ratio.
818void initialise_wheel_speeds(SimState& state, const CsvRow& sample,
819 const common_lib::car_parameters::CarParameters& params) {
820 constexpr double kRpmToRadPerSec = 2.0 * M_PI / 60.0;
821 state.wheels_speed.front_left = sample.real_fl_rpm * kRpmToRadPerSec;
822 state.wheels_speed.front_right = sample.real_fr_rpm * kRpmToRadPerSec;
823 if (sample.has_rear_wheel_rpm) {
824 state.wheels_speed.rear_left = sample.real_rl_rpm * kRpmToRadPerSec;
825 state.wheels_speed.rear_right = sample.real_rr_rpm * kRpmToRadPerSec;
826 } else {
827 const double ratio = std::max(params.transmission_parameters->gear_ratio, 1e-6);
828 const double rear_omega = sample.real_motor_rpm * kRpmToRadPerSec / ratio;
829 state.wheels_speed.rear_left = rear_omega;
830 state.wheels_speed.rear_right = rear_omega;
831 }
832 state.motor_omega = sample.real_motor_rpm * kRpmToRadPerSec;
833}

835struct DatasetConfig {
836 double weight = 1.0;
837 double start_offset_s = 0.0;
838 double stop_after_s = -1.0;
839 double max_distance_error = 2.5;
840 double max_velocity_error = 5.0;
841 bool ignore_lateral_distance = false;
842 // Re-anchored scoring period (s). >0 resets the sim to the measured state
843 // every period, so the fitness rewards models that track the *whole* run in
844 // short open-loop horizons rather than ones that merely survive the easy
845 // opening while being globally unfaithful (e.g. directionally unstable).
846 // 0 keeps the classic continuous time-to-first-failure behaviour.
847 double reanchor_period = 0.0;
848 // Derived from telemetry; never supplied as a hand-tuned timestamp.
849 size_t start_index = 0;
850 size_t end_index = std::numeric_limits<size_t>::max();
851};

853// Return the contiguous useful driving portion of a recording. Velocity alone
854// is deliberately cross-checked against pose increments: estimator noise after
855// a bag ends otherwise looks like movement. Start requires sustained movement;
856// end requires both sustained rest and inactive driver inputs.
857std::pair<size_t, size_t> detect_motion_window(const std::vector<CsvRow>& rows) {
858 if (rows.empty()) return {0, 0};
859 constexpr double kMovingSpeed = 0.35;
860 constexpr double kActiveInput = 0.02;
861 constexpr double kHoldSeconds = 0.50;
862 std::vector<bool> moving(rows.size(), false);
863 std::vector<bool> input_active(rows.size(), false);
864 for (size_t i = 0; i < rows.size(); ++i) {
865 const auto& r = rows[i];
866 double position_speed = 0.0;
867 if (i > 0) {
868 const double dt = r.timestamp_s - rows[i - 1].timestamp_s;
869 if (dt > 1e-4 && dt < 0.1)
870 position_speed = std::hypot(r.real_x - rows[i - 1].real_x, r.real_y - rows[i - 1].real_y) / dt;
871 }
872 const double speed = std::max(std::hypot(r.real_vx, r.real_vy), position_speed);
873 const double drive = std::max(std::abs(r.throttle_rl), std::abs(r.throttle_rr));
874 moving[i] = speed >= kMovingSpeed;
875 input_active[i] = drive >= kActiveInput || std::abs(r.steering) >= kActiveInput;
876 }
877 const auto sustained_from = [&](size_t first) {
878 const double begin = rows[first].timestamp_s;
879 for (size_t j = first; j < rows.size(); ++j) {
880 if (!moving[j]) return false;
881 if (rows[j].timestamp_s - begin >= kHoldSeconds) return true;
882 }
883 return false;
884 };
885 size_t start = 0;
886 while (start + 1 < rows.size() && !sustained_from(start)) ++start;
887 size_t end = rows.size() - 1;
888 // A bag can keep publishing commands after the car has stopped. End only
889 // after sustained *zero speed and inactive driver input*.
890 for (size_t i = start; i < rows.size(); ++i) {
891 const double begin = rows[i].timestamp_s;
892 bool stopped = true;
893 for (size_t j = i; j < rows.size() && rows[j].timestamp_s - begin < kHoldSeconds; ++j) {
894 if (moving[j] || input_active[j]) { stopped = false; break; }
895 }
896 if (stopped && rows.back().timestamp_s - begin >= kHoldSeconds) { end = i; break; }
897 }
898 return {start, std::max(start, end)};
899}

905template <typename ModelType>
906EvaluationResult evaluate_candidate_fast(
907 const std::vector<std::vector<CsvRow>>& all_csvs_rows,
908 const std::vector<DatasetConfig>& dataset_configs,
909 const std::vector<ParameterSpec>& param_specs,
910 const std::vector<double>& candidate_values,
911 InvictaSimParameters& thread_local_params,
912 const std::vector<double*>& fast_ptrs,
913 ModelType& model
914) {
915 for (size_t i = 0; i < param_specs.size(); ++i) {
916 if (fast_ptrs[i]) *(fast_ptrs[i]) = candidate_values[i];
917 }
918
919 double weighted_score_sum = 0.0;
920 double total_weight = 0.0;
921 double weighted_position_rmse = 0.0;
922 double weighted_heading_rmse = 0.0;
923 double weighted_velocity_rmse = 0.0;
924 double weighted_velocity_x_rmse = 0.0;
925 double weighted_velocity_y_rmse = 0.0;
926 double weighted_yaw_rate_rmse = 0.0;
927 double weighted_front_wheel_rpm_rmse = 0.0;
928 double weighted_motor_rpm_rmse = 0.0;
929
930 for (size_t b = 0; b < dataset_configs.size(); ++b) {
931 const auto& ds_cfg = dataset_configs[b];
932 double weight = ds_cfg.weight;
933 double stop_duration = ds_cfg.stop_after_s;
934 double max_dist_error = ds_cfg.max_distance_error;
935 bool ignore_lateral = ds_cfg.ignore_lateral_distance;
936
937 const auto& rows = all_csvs_rows[b];
938 if (rows.empty()) {
939 weighted_score_sum += weight * -1e9;
940 total_weight += weight;
941 continue;
942 }
943
944 const size_t start_idx = std::min(ds_cfg.start_index, rows.size() - 1);
945 const size_t end_idx = std::min(ds_cfg.end_index, rows.size() - 1);
946
947 double start_time = rows[start_idx].timestamp_s;
948
949 model.update_cache();
950
951 SimState state;
952 CsvRow first_sample = rows[start_idx];
953 state.x = 0.0;
954 state.y = 0.0;
955 state.yaw = 0.0;
956 state.vx = first_sample.real_vx;
957 state.vy = first_sample.real_vy;
958 state.yaw_rate = first_sample.real_yaw_rate;
959
960 initialise_wheel_speeds(state, first_sample, *thread_local_params.car_parameters);
961
962 RunningRmse position_rmse, heading_rmse, velocity_rmse, velocity_x_rmse, velocity_x_slow_rmse;
963 RunningRmse velocity_y_rmse, yaw_rate_rmse, yaw_rate_under_rmse, front_wheel_rpm_rmse, motor_rpm_rmse;
964
965 PoseSample real_origin = {first_sample.real_x, first_sample.real_y, first_sample.real_yaw};
966 PoseSample sim_origin = {state.x, state.y, state.yaw};
967
968 double last_time = first_sample.timestamp_s;
969 double alive_time = 0.0;
970
971 for (size_t i = start_idx + 1; i <= end_idx; ++i) {
972 const CsvRow& row = rows[i];
973 if (stop_duration > 0.0 && (row.timestamp_s - start_time) > stop_duration) break;
974
975 double dt = row.timestamp_s - last_time;
976 if (dt <= 0.0) continue;
977 last_time = row.timestamp_s;
978
979 const CsvRow& prev_row = rows[i - 1];
980 common_lib::structures::Wheels throttle(prev_row.throttle_fl, prev_row.throttle_fr, prev_row.throttle_rl, prev_row.throttle_rr);
981
982 model.step(dt, throttle, prev_row.steering, state);
983
984 if (!std::isfinite(state.x) || !std::isfinite(state.y) || !std::isfinite(state.vx) ||
985 !std::isfinite(state.vy) || !std::isfinite(state.yaw_rate) ||
986 std::abs(state.vx) > 80.0 || std::abs(state.vy) > 40.0 ||
987 std::abs(state.yaw_rate) > 20.0) {
988 break;
989 }
990
991 double sim_fl_rpm = state.wheels_speed.front_left * 60.0 / (2.0 * 3.14159265358979323846);
992 double sim_fr_rpm = state.wheels_speed.front_right * 60.0 / (2.0 * 3.14159265358979323846);
993 double sim_front_rpm = 0.5 * (sim_fl_rpm + sim_fr_rpm);
994 double sim_motor_rpm = state.motor_omega * 60.0 / (2.0 * 3.14159265358979323846);
995 double real_front_rpm = 0.5 * (row.real_fl_rpm + row.real_fr_rpm);
996
997 PoseSample real_raw = {row.real_x, row.real_y, row.real_yaw};
998 PoseSample real_pose = transform_pose_to_map(real_raw, real_origin, sim_origin);
999
1000 double current_pos_error;
1001 if (ignore_lateral) {
1002 double sim_dist = std::hypot(state.x - sim_origin.x, state.y - sim_origin.y);
1003 double real_dist = std::hypot(real_pose.x - real_origin.x, real_pose.y - real_origin.y);
1004 current_pos_error = std::abs(sim_dist - real_dist);
1005 } else {
1006 current_pos_error = std::hypot(state.x - real_pose.x, state.y - real_pose.y);
1007 }
1008
1009 const double current_vel_error = std::abs(state.vx - row.real_vx) /* real_vy is not measured (identically 0), so velocity error is longitudinal */;
1010
1011 alive_time = row.timestamp_s - start_time;
1012
1013 if (current_pos_error > max_dist_error || current_vel_error > ds_cfg.max_velocity_error) {
1014 break;
1015 }
1016
1017 position_rmse.update(current_pos_error);
1018 heading_rmse.update(normalize_angle(state.yaw - real_pose.yaw));
1019 velocity_x_rmse.update(state.vx - row.real_vx);
1020 velocity_x_slow_rmse.update(std::min(0.0, state.vx - row.real_vx));
1021 velocity_y_rmse.update(state.vy - row.real_vy);
1022 velocity_rmse.update(std::abs(state.vx - row.real_vx) /* real_vy is not measured (identically 0), so velocity error is longitudinal */);
1023 yaw_rate_rmse.update(state.yaw_rate - row.real_yaw_rate);
1024 yaw_rate_under_rmse.update(std::min(0.0, state.yaw_rate - row.real_yaw_rate));
1025 front_wheel_rpm_rmse.update(sim_front_rpm - real_front_rpm);
1026 motor_rpm_rmse.update(sim_motor_rpm - row.real_motor_rpm);
1027 }
1028
1029 // Score based only on errors (survival time is removed).
1030 const double dataset_score = - position_rmse.get() * 1e3
1031 - velocity_rmse.get() * 1e2 - heading_rmse.get() * 10.0
1032 - yaw_rate_rmse.get();
1033
1034 weighted_score_sum += weight * dataset_score;
1035 weighted_position_rmse += weight * position_rmse.get();
1036 weighted_heading_rmse += weight * heading_rmse.get();
1037 weighted_velocity_rmse += weight * velocity_rmse.get();
1038 weighted_velocity_x_rmse += weight * velocity_x_rmse.get();
1039 weighted_velocity_y_rmse += weight * velocity_y_rmse.get();
1040 weighted_yaw_rate_rmse += weight * yaw_rate_rmse.get();
1041 weighted_front_wheel_rpm_rmse += weight * front_wheel_rpm_rmse.get();
1042 weighted_motor_rpm_rmse += weight * motor_rpm_rmse.get();
1043 total_weight += weight;
1044 }
1045
1046 double final_score = weighted_score_sum / std::max(total_weight, 1e-9);
1047 EvaluationResult res;
1048 res.total_score = final_score;
1049 res.metrics.position_rmse = weighted_position_rmse / std::max(total_weight, 1e-9);
1050 res.metrics.heading_rmse = weighted_heading_rmse / std::max(total_weight, 1e-9);
1051 res.metrics.velocity_rmse = weighted_velocity_rmse / std::max(total_weight, 1e-9);
1052 res.metrics.velocity_x_rmse = weighted_velocity_x_rmse / std::max(total_weight, 1e-9);
1053 res.metrics.velocity_y_rmse = weighted_velocity_y_rmse / std::max(total_weight, 1e-9);
1054 res.metrics.yaw_rate_rmse = weighted_yaw_rate_rmse / std::max(total_weight, 1e-9);
1055 res.metrics.front_wheel_rpm_rmse = weighted_front_wheel_rpm_rmse / std::max(total_weight, 1e-9);
1056 res.metrics.motor_rpm_rmse = weighted_motor_rpm_rmse / std::max(total_weight, 1e-9);
1057 return res;
1058}

1060template <typename ModelType>
1061EvaluationResult evaluate_candidate(
1062 const std::vector<std::vector<CsvRow>>& all_csvs_rows,
1063 const std::vector<DatasetConfig>& dataset_configs,
1064 const std::vector<ParameterSpec>& param_specs,
1065 const std::vector<double>& candidate_values,
1066 const InvictaSimParameters& base_params
1067) {
1068 g_eval_count.fetch_add(1, std::memory_order_relaxed);
1069 InvictaSimParameters sim_params = base_params;
1070 sim_params.car_parameters = deep_copy_car_parameters(base_params.car_parameters);
1071
1072 for (size_t i = 0; i < param_specs.size(); ++i) {
1073 apply_parameter_override(sim_params.car_parameters, param_specs[i].name, candidate_values[i]);
1074 }
1075
1076 double weighted_score_sum = 0.0;
1077 double total_weight = 0.0;
1078 double weighted_position_rmse = 0.0;
1079 double weighted_heading_rmse = 0.0;
1080 double weighted_velocity_rmse = 0.0;
1081 double weighted_velocity_x_rmse = 0.0;
1082 double weighted_velocity_y_rmse = 0.0;
1083 double weighted_yaw_rate_rmse = 0.0;
1084 double weighted_front_wheel_rpm_rmse = 0.0;
1085 double weighted_motor_rpm_rmse = 0.0;
1086
1087 for (size_t b = 0; b < dataset_configs.size(); ++b) {
1088 const auto& ds_cfg = dataset_configs[b];
1089 double weight = ds_cfg.weight;
1090 double stop_duration = ds_cfg.stop_after_s;
1091 double max_dist_error = ds_cfg.max_distance_error;
1092 bool ignore_lateral = ds_cfg.ignore_lateral_distance;
1093
1094 const auto& rows = all_csvs_rows[b];
1095 if (rows.empty()) {
1096 weighted_score_sum += weight * -1e9;
1097 total_weight += weight;
1098 continue;
1099 }
1100
1101 const size_t start_idx = std::min(ds_cfg.start_index, rows.size() - 1);
1102 const size_t end_idx = std::min(ds_cfg.end_index, rows.size() - 1);
1103
1104 double start_time = rows[start_idx].timestamp_s;
1105
1106 // INSTANTIATE THE HARDCODED MODEL DIRECTLY
1107 if (!sim_params.car_parameters) {
1108 std::cerr << "Error: sim_params.car_parameters is null!" << std::endl;
1109 } else if (!sim_params.car_parameters->motor_parameters) {
1110 std::cerr << "Error: sim_params.car_parameters->motor_parameters is null!" << std::endl;
1111 }
1112 ModelType model(sim_params);
1113
1114 // Warm only hidden actuator/energy states from the discarded prefix.
1115 // The scoring state is still set from the first accepted measurement;
1116 // this prevents the inverter delay from being spuriously restarted at
1117 // the automatically detected motion boundary.
1118 SimState warmup_state;
1119 for (size_t i = 1; i <= start_idx; ++i) {
1120 const double warmup_dt = rows[i].timestamp_s - rows[i - 1].timestamp_s;
1121 if (warmup_dt <= 0.0) continue;
1122 const CsvRow& command = rows[i - 1];
1123 model.step(warmup_dt,
1124 common_lib::structures::Wheels(command.throttle_fl, command.throttle_fr,
1125 command.throttle_rl, command.throttle_rr),
1126 command.steering, warmup_state);
1127 }
1128
1129 SimState state;
1130 CsvRow first_sample = rows[start_idx];
1131 state.x = 0.0;
1132 state.y = 0.0;
1133 state.yaw = 0.0;
1134 state.vx = first_sample.real_vx;
1135 state.vy = first_sample.real_vy;
1136 state.yaw_rate = first_sample.real_yaw_rate;
1137
1138 initialise_wheel_speeds(state, first_sample, *sim_params.car_parameters);
1139
1140 RunningRmse position_rmse, heading_rmse, velocity_rmse, velocity_x_rmse, velocity_x_slow_rmse;
1141 RunningRmse velocity_y_rmse, yaw_rate_rmse, yaw_rate_under_rmse, front_wheel_rpm_rmse, motor_rpm_rmse;
1142
1143 PoseSample real_origin = {first_sample.real_x, first_sample.real_y, first_sample.real_yaw};
1144 PoseSample sim_origin = {state.x, state.y, state.yaw};
1145
1146 double last_time = first_sample.timestamp_s;
1147
1148 const double reanchor = ds_cfg.reanchor_period;
1149 double next_anchor = reanchor > 0.0 ? reanchor : 1e18;
1150 bool seg_failed = false; // current segment has already violated
1151 int consecutive_failed_segments = 0; // hopeless-candidate early abort
1152 double alive_time = 0.0; // total in-limit time (summed over segments)
1153 for (size_t i = start_idx + 1; i <= end_idx; ++i) {
1154 const CsvRow& row = rows[i];
1155 if (stop_duration > 0.0 && (row.timestamp_s - start_time) > stop_duration) break;
1156
1157 double dt = row.timestamp_s - last_time;
1158 if (dt <= 0.0) continue;
1159 last_time = row.timestamp_s;
1160 const double t_rel = row.timestamp_s - start_time;
1161
1162 PoseSample real_pose = transform_pose_to_map(
1163 {row.real_x, row.real_y, row.real_yaw}, real_origin, sim_origin);
1164
1165 // Re-anchor boundary: reset the sim to the measured state and open a
1166 // fresh short-horizon segment (vy is unmeasured, so seed 0).
1167 if (reanchor > 0.0 && t_rel >= next_anchor) {
1168 next_anchor += reanchor;
1169 consecutive_failed_segments = seg_failed ? consecutive_failed_segments + 1 : 0;
1170 if (consecutive_failed_segments >= 6) break; // globally hopeless
1171 seg_failed = false;
1172 state.x = real_pose.x; state.y = real_pose.y; state.yaw = real_pose.yaw;
1173 state.vx = row.real_vx; state.vy = 0.0; state.yaw_rate = row.real_yaw_rate;
1174 state.ax = 0.0; state.ay = 0.0;
1175 initialise_wheel_speeds(state, row, *sim_params.car_parameters);
1176 continue;
1177 }
1178 if (seg_failed) continue; // skip simulation until the next anchor
1179
1180 const CsvRow& prev_row = rows[i - 1];

1186 if (!std::isfinite(state.x) || !std::isfinite(state.y) || !std::isfinite(state.vx) ||
1187 !std::isfinite(state.vy) || !std::isfinite(state.yaw_rate) ||
1188 std::abs(state.vx) > 80.0 || std::abs(state.vy) > 40.0 ||
1189 std::abs(state.yaw_rate) > 20.0) {
1190 if (reanchor > 0.0) { seg_failed = true; continue; }
1191 break;
1192 }
1193
1194 double sim_fl_rpm = state.wheels_speed.front_left * 60.0 / (2.0 * M_PI);
1195 double sim_fr_rpm = state.wheels_speed.front_right * 60.0 / (2.0 * M_PI);
1196 double sim_front_rpm = 0.5 * (sim_fl_rpm + sim_fr_rpm);
1197 double sim_motor_rpm = state.motor_omega * 60.0 / (2.0 * M_PI);
1198 double real_front_rpm = 0.5 * (row.real_fl_rpm + row.real_fr_rpm);
1199
1200 double current_pos_error;
1201 if (ignore_lateral) {
1202 // Ignore lateral drift: just compare total distance traveled from start
1203 double sim_dist = std::hypot(state.x - sim_origin.x, state.y - sim_origin.y);
1204 double real_dist = std::hypot(real_pose.x - real_origin.x, real_pose.y - real_origin.y);
1205 current_pos_error = std::abs(sim_dist - real_dist);
1206 } else {
1207 // Strict 2D Euclidean distance error
1208 current_pos_error = std::hypot(state.x - real_pose.x, state.y - real_pose.y);
1209 }
1210
1211 const double current_vel_error = std::abs(state.vx - row.real_vx) /* real_vy is not measured (identically 0), so velocity error is longitudinal */;
1212
1213 if (current_pos_error > max_dist_error || current_vel_error > ds_cfg.max_velocity_error) {
1214 if (reanchor > 0.0) { seg_failed = true; continue; }
1215 break; // Hard limit reached, kill mutation
1216 }
1217
1218 alive_time += dt; // accumulate in-limit time (whole-run coverage)
1219
1220 position_rmse.update(current_pos_error);
1221 heading_rmse.update(normalize_angle(state.yaw - real_pose.yaw));
1222 velocity_x_rmse.update(state.vx - row.real_vx);
1223 velocity_x_slow_rmse.update(std::min(0.0, state.vx - row.real_vx));
1224 velocity_y_rmse.update(state.vy - row.real_vy);
1225 velocity_rmse.update(std::abs(state.vx - row.real_vx) /* real_vy is not measured (identically 0), so velocity error is longitudinal */);
1226 yaw_rate_rmse.update(state.yaw_rate - row.real_yaw_rate);
1227 yaw_rate_under_rmse.update(std::min(0.0, state.yaw_rate - row.real_yaw_rate));
1228 front_wheel_rpm_rmse.update(sim_front_rpm - real_front_rpm);
1229 motor_rpm_rmse.update(sim_motor_rpm - row.real_motor_rpm);
1230 }
1231
1232 // Score based only on errors (survival time is removed).
1233 const double dataset_score = - position_rmse.get() * 1e3
1234 - velocity_rmse.get() * 1e2 - heading_rmse.get() * 10.0
1235 - yaw_rate_rmse.get();
1236
1237 weighted_score_sum += weight * dataset_score;
1238 weighted_position_rmse += weight * position_rmse.get();
1239 weighted_heading_rmse += weight * heading_rmse.get();
1240 weighted_velocity_rmse += weight * velocity_rmse.get();
1241 weighted_velocity_x_rmse += weight * velocity_x_rmse.get();
1242 weighted_velocity_y_rmse += weight * velocity_y_rmse.get();
1243 weighted_yaw_rate_rmse += weight * yaw_rate_rmse.get();
1244 weighted_front_wheel_rpm_rmse += weight * front_wheel_rpm_rmse.get();
1245 weighted_motor_rpm_rmse += weight * motor_rpm_rmse.get();
1246 total_weight += weight;
1247 }
1248
1249 double final_score = weighted_score_sum / std::max(total_weight, 1e-9);
1250 EvaluationResult res;
1251 res.total_score = final_score;
1252 res.metrics.position_rmse = weighted_position_rmse / std::max(total_weight, 1e-9);
1253 res.metrics.heading_rmse = weighted_heading_rmse / std::max(total_weight, 1e-9);
1254 res.metrics.velocity_rmse = weighted_velocity_rmse / std::max(total_weight, 1e-9);
1255 res.metrics.velocity_x_rmse = weighted_velocity_x_rmse / std::max(total_weight, 1e-9);
1256 res.metrics.velocity_y_rmse = weighted_velocity_y_rmse / std::max(total_weight, 1e-9);
1257 res.metrics.yaw_rate_rmse = weighted_yaw_rate_rmse / std::max(total_weight, 1e-9);
1258 res.metrics.front_wheel_rpm_rmse = weighted_front_wheel_rpm_rmse / std::max(total_weight, 1e-9);
1259 res.metrics.motor_rpm_rmse = weighted_motor_rpm_rmse / std::max(total_weight, 1e-9);
1260 return res;
1261}

1679 if (warmup_dt <= 0.0) continue;

1682 common_lib::structures::Wheels(c.throttle_fl, c.throttle_fr, c.throttle_rl, c.throttle_rr),

1690 state.vy = first_sample.real_vy;
1691 state.yaw_rate = first_sample.real_yaw_rate;

1751 double real_front_rpm = 0.5 * (row.real_fl_rpm + row.real_fr_rpm);
1752 out << t_rel << ',' << state.x << ',' << state.y << ',' << real_pose.x << ',' << real_pose.y << ','
1753 << pos_err << ',' << state.vx << ',' << state.vy << ',' << row.real_vx << ',' << row.real_vy << ','
1754 << vel_err << ',' << state.yaw << ',' << real_pose.yaw << ',' << normalize_angle(state.yaw - real_pose.yaw) << ','
1755 << state.yaw_rate << ',' << row.real_yaw_rate << ',' << sim_motor_rpm << ',' << row.real_motor_rpm << ','
1756 << sim_front_rpm << ',' << real_front_rpm << ',' << prev.throttle_rl << ',' << prev.steering << '\n';
1757
1758 if (!std::isfinite(state.vx)) { std::cout << "NON-FINITE at t=" << t_rel << "\n"; break; }
1759 }
1760 out.close();

1762 << "s (" << (rows[end_idx].timestamp_s - start_time) << "s driving)\n";
1763 std::cout << " first pos-limit (" << max_dist_error << "m) violation: "
1764 << (first_pos_violation < 0 ? std::string("never") : std::to_string(first_pos_violation) + "s")
1765 << " | max pos err " << max_pos_err << "m\n";
1766 std::cout << " first vel-limit (" << max_velocity_error << "m/s) violation: "
1767 << (first_vel_violation < 0 ? std::string("never") : std::to_string(first_vel_violation) + "s")
1768 << " | max vel err " << max_vel_err << "m/s\n";

1773 if (reanchor_period > 0.0 && seg_total > 0) {
