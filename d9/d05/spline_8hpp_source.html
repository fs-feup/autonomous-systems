55// spline interpolation
56class spline
57{
58public:
59 // spline types
60 enum spline_type {
61 linear = 10, // linear interpolation
62 cspline = 30, // cubic splines (classical C^2)
63 cspline_hermite = 31 // cubic hermite splines (local, only C^1)
64 };
65
66 // boundary condition type for the spline end-points
67 enum bd_type {
68 first_deriv = 1,
69 second_deriv = 2,
70 not_a_knot = 3
71 };
72
73protected:
74 std::vector<double> m_x,m_y; // x,y coordinates of points
75 // interpolation parameters
76 // f(x) = a_i + b_i*(x-x_i) + c_i*(x-x_i)^2 + d_i*(x-x_i)^3
77 // where a_i = y_i, or else it won't go through grid points
78 std::vector<double> m_b,m_c,m_d; // spline coefficients
79 double m_c0; // for left extrapolation
80 spline_type m_type;
81 bd_type m_left, m_right;
82 double m_left_value, m_right_value;
83 bool m_made_monotonic;
84 void set_coeffs_from_b(); // calculate c_i, d_i from b_i
85 size_t find_closest(double x) const; // closest idx so that m_x[idx]<=x
86
87public:
88 // default constructor: set boundary condition to be zero curvature
89 // at both ends, i.e. natural splines
90 spline(): m_type(cspline),
91 m_left(second_deriv), m_right(second_deriv),
92 m_left_value(0.0), m_right_value(0.0), m_made_monotonic(false)
93 {
94 ;
95 }
96 spline(const std::vector<double>& X, const std::vector<double>& Y,
97 spline_type type = cspline,
98 bool make_monotonic = false,
99 bd_type left = second_deriv, double left_value = 0.0,
100 bd_type right = second_deriv, double right_value = 0.0
101 ):
102 m_type(type),
103 m_left(left), m_right(right),
104 m_left_value(left_value), m_right_value(right_value),
105 m_made_monotonic(false) // false correct here: make_monotonic() sets it
106 {
107 this->set_points(X,Y,m_type);
108 if(make_monotonic) {
109 this->make_monotonic();
110 }
111 }
112
113
114 // modify boundary conditions: if called it must be before set_points()
115 void set_boundary(bd_type left, double left_value,
116 bd_type right, double right_value);
117
118 // set all data points (cubic_spline=false means linear interpolation)
119 void set_points(const std::vector<double>& x,
120 const std::vector<double>& y,
121 spline_type type=cspline);
122
123 // adjust coefficients so that the spline becomes piecewise monotonic
124 // where possible
125 // this is done by adjusting slopes at grid points by a non-negative
126 // factor and this will break C^2
127 // this can also break boundary conditions if adjustments need to
128 // be made at the boundary points
129 // returns false if no adjustments have been made, true otherwise
130 bool make_monotonic();
131
132 // evaluates the spline at point x
133 double operator() (double x) const;
134 double deriv(int order, double x) const;
135
136 // solves for all x so that: spline(x) = y
137 std::vector<double> solve(double y, bool ignore_extrapolation=true) const;
138
139 // returns the input data points
140 std::vector<double> get_x() const { return m_x; }
141 std::vector<double> get_y() const { return m_y; }
142 double get_x_min() const { assert(!m_x.empty()); return m_x.front(); }
143 double get_x_max() const { assert(!m_x.empty()); return m_x.back(); }
144
145#ifdef HAVE_SSTREAM
146 // spline info string, i.e. spline type, boundary conditions etc.
147 std::string info() const;
148#endif // HAVE_SSTREAM
149
150};

219void spline::set_coeffs_from_b()
220{
221 assert(m_x.size()==m_y.size());
222 assert(m_x.size()==m_b.size());
223 assert(m_x.size()>2);
224 size_t n=m_b.size();
225 if(m_c.size()!=n)
226 m_c.resize(n);
227 if(m_d.size()!=n)
228 m_d.resize(n);
229
230 for(size_t i=0; i<n-1; i++) {
231 const double h = m_x[i+1]-m_x[i];
232 // from continuity and differentiability condition
233 m_c[i] = ( 3.0*(m_y[i+1]-m_y[i])/h - (2.0*m_b[i]+m_b[i+1]) ) / h;
234 // from differentiability condition
235 m_d[i] = ( (m_b[i+1]-m_b[i])/(3.0*h) - 2.0/3.0*m_c[i] ) / h;
236 }
237
238 // for left extrapolation coefficients
239 m_c0 = (m_left==first_deriv) ? 0.0 : m_c[0];
240}

242void spline::set_points(const std::vector<double>& x,
243 const std::vector<double>& y,
244 spline_type type)
245{
246 assert(x.size()==y.size());
247 assert(x.size()>=3);
248 // not-a-knot with 3 points has many solutions
249 if(m_left==not_a_knot || m_right==not_a_knot)
250 assert(x.size()>=4);
251 m_type=type;
252 m_made_monotonic=false;
253 m_x=x;
254 m_y=y;
255 int n = (int) x.size();
256 // check strict monotonicity of input vector x
257 for(int i=0; i<n-1; i++) {
258 assert(m_x[i]<m_x[i+1]);
259 }
260
261
262 if(type==linear) {
263 // linear interpolation
264 m_d.resize(n);
265 m_c.resize(n);
266 m_b.resize(n);
267 for(int i=0; i<n-1; i++) {
268 m_d[i]=0.0;
269 m_c[i]=0.0;
270 m_b[i]=(m_y[i+1]-m_y[i])/(m_x[i+1]-m_x[i]);
271 }
272 // ignore boundary conditions, set slope equal to the last segment
273 m_b[n-1]=m_b[n-2];
274 m_c[n-1]=0.0;
275 m_d[n-1]=0.0;
276 } else if(type==cspline) {
277 // classical cubic splines which are C^2 (twice cont differentiable)
278 // this requires solving an equation system
279
280 // setting up the matrix and right hand side of the equation system
281 // for the parameters b[]
282 int n_upper = (m_left == spline::not_a_knot) ? 2 : 1;
283 int n_lower = (m_right == spline::not_a_knot) ? 2 : 1;
284 internal::band_matrix A(n,n_upper,n_lower);
285 std::vector<double> rhs(n);
286 for(int i=1; i<n-1; i++) {
287 A(i,i-1)=1.0/3.0*(x[i]-x[i-1]);
288 A(i,i)=2.0/3.0*(x[i+1]-x[i-1]);
289 A(i,i+1)=1.0/3.0*(x[i+1]-x[i]);
290 rhs[i]=(y[i+1]-y[i])/(x[i+1]-x[i]) - (y[i]-y[i-1])/(x[i]-x[i-1]);
291 }
292 // boundary conditions
293 if(m_left == spline::second_deriv) {
294 // 2*c[0] = f''
295 A(0,0)=2.0;
296 A(0,1)=0.0;
297 rhs[0]=m_left_value;
298 } else if(m_left == spline::first_deriv) {
299 // b[0] = f', needs to be re-expressed in terms of c:
300 // (2c[0]+c[1])(x[1]-x[0]) = 3 ((y[1]-y[0])/(x[1]-x[0]) - f')
301 A(0,0)=2.0*(x[1]-x[0]);
302 A(0,1)=1.0*(x[1]-x[0]);
303 rhs[0]=3.0*((y[1]-y[0])/(x[1]-x[0])-m_left_value);
304 } else if(m_left == spline::not_a_knot) {
305 // f'''(x[1]) exists, i.e. d[0]=d[1], or re-expressed in c:
306 // -h1*c[0] + (h0+h1)*c[1] - h0*c[2] = 0
307 A(0,0) = -(x[2]-x[1]);
308 A(0,1) = x[2]-x[0];
309 A(0,2) = -(x[1]-x[0]);
310 rhs[0] = 0.0;
311 } else {
312 assert(false);
313 }
314 if(m_right == spline::second_deriv) {
315 // 2*c[n-1] = f''
316 A(n-1,n-1)=2.0;
317 A(n-1,n-2)=0.0;
318 rhs[n-1]=m_right_value;
319 } else if(m_right == spline::first_deriv) {
320 // b[n-1] = f', needs to be re-expressed in terms of c:
321 // (c[n-2]+2c[n-1])(x[n-1]-x[n-2])
322 // = 3 (f' - (y[n-1]-y[n-2])/(x[n-1]-x[n-2]))
323 A(n-1,n-1)=2.0*(x[n-1]-x[n-2]);
324 A(n-1,n-2)=1.0*(x[n-1]-x[n-2]);
325 rhs[n-1]=3.0*(m_right_value-(y[n-1]-y[n-2])/(x[n-1]-x[n-2]));
326 } else if(m_right == spline::not_a_knot) {
327 // f'''(x[n-2]) exists, i.e. d[n-3]=d[n-2], or re-expressed in c:
328 // -h_{n-2}*c[n-3] + (h_{n-3}+h_{n-2})*c[n-2] - h_{n-3}*c[n-1] = 0
329 A(n-1,n-3) = -(x[n-1]-x[n-2]);
330 A(n-1,n-2) = x[n-1]-x[n-3];
331 A(n-1,n-1) = -(x[n-2]-x[n-3]);
332 rhs[0] = 0.0;
333 } else {
334 assert(false);
335 }
336
337 // solve the equation system to obtain the parameters c[]
338 m_c=A.lu_solve(rhs);
339
340 // calculate parameters b[] and d[] based on c[]
341 m_d.resize(n);
342 m_b.resize(n);
343 for(int i=0; i<n-1; i++) {
344 m_d[i]=1.0/3.0*(m_c[i+1]-m_c[i])/(x[i+1]-x[i]);
345 m_b[i]=(y[i+1]-y[i])/(x[i+1]-x[i])
346 - 1.0/3.0*(2.0*m_c[i]+m_c[i+1])*(x[i+1]-x[i]);
347 }
348 // for the right extrapolation coefficients (zero cubic term)
349 // f_{n-1}(x) = y_{n-1} + b*(x-x_{n-1}) + c*(x-x_{n-1})^2
350 double h=x[n-1]-x[n-2];
351 // m_c[n-1] is determined by the boundary condition
352 m_d[n-1]=0.0;
353 m_b[n-1]=3.0*m_d[n-2]*h*h+2.0*m_c[n-2]*h+m_b[n-2]; // = f'_{n-2}(x_{n-1})
354 if(m_right==first_deriv)
355 m_c[n-1]=0.0; // force linear extrapolation
356
357 } else if(type==cspline_hermite) {
358 // hermite cubic splines which are C^1 (cont. differentiable)
359 // and derivatives are specified on each grid point
360 // (here we use 3-point finite differences)
361 m_b.resize(n);
362 m_c.resize(n);
363 m_d.resize(n);
364 // set b to match 1st order derivative finite difference
365 for(int i=1; i<n-1; i++) {
366 const double h = m_x[i+1]-m_x[i];
367 const double hl = m_x[i]-m_x[i-1];
368 m_b[i] = -h/(hl*(hl+h))*m_y[i-1] + (h-hl)/(hl*h)*m_y[i]
369 + hl/(h*(hl+h))*m_y[i+1];
370 }
371 // boundary conditions determine b[0] and b[n-1]
372 if(m_left==first_deriv) {
373 m_b[0]=m_left_value;
374 } else if(m_left==second_deriv) {
375 const double h = m_x[1]-m_x[0];
376 m_b[0]=0.5*(-m_b[1]-0.5*m_left_value*h+3.0*(m_y[1]-m_y[0])/h);
377 } else if(m_left == not_a_knot) {
378 // f''' continuous at x[1]
379 const double h0 = m_x[1]-m_x[0];
380 const double h1 = m_x[2]-m_x[1];
381 m_b[0]= -m_b[1] + 2.0*(m_y[1]-m_y[0])/h0
382 + h0*h0/(h1*h1)*(m_b[1]+m_b[2]-2.0*(m_y[2]-m_y[1])/h1);
383 } else {
384 assert(false);
385 }
386 if(m_right==first_deriv) {
387 m_b[n-1]=m_right_value;
388 m_c[n-1]=0.0;
389 } else if(m_right==second_deriv) {
390 const double h = m_x[n-1]-m_x[n-2];
391 m_b[n-1]=0.5*(-m_b[n-2]+0.5*m_right_value*h+3.0*(m_y[n-1]-m_y[n-2])/h);
392 m_c[n-1]=0.5*m_right_value;
393 } else if(m_right == not_a_knot) {
394 // f''' continuous at x[n-2]
395 const double h0 = m_x[n-2]-m_x[n-3];
396 const double h1 = m_x[n-1]-m_x[n-2];
397 m_b[n-1]= -m_b[n-2] + 2.0*(m_y[n-1]-m_y[n-2])/h1 + h1*h1/(h0*h0)
398 *(m_b[n-3]+m_b[n-2]-2.0*(m_y[n-2]-m_y[n-3])/h0);
399 // f'' continuous at x[n-1]: c[n-1] = 3*d[n-2]*h[n-2] + c[n-1]
400 m_c[n-1]=(m_b[n-2]+2.0*m_b[n-1])/h1-3.0*(m_y[n-1]-m_y[n-2])/(h1*h1);
401 } else {
402 assert(false);
403 }
404 m_d[n-1]=0.0;
405
406 // parameters c and d are determined by continuity and differentiability
407 set_coeffs_from_b();
408
409 } else {
410 assert(false);
411 }
412
413 // for left extrapolation coefficients
414 m_c0 = (m_left==first_deriv) ? 0.0 : m_c[0];
415}

417bool spline::make_monotonic()
418{
419 assert(m_x.size()==m_y.size());
420 assert(m_x.size()==m_b.size());
421 assert(m_x.size()>2);
422 bool modified = false;
423 const int n=(int)m_x.size();
424 // make sure: input data monotonic increasing --> b_i>=0
425 // input data monotonic decreasing --> b_i<=0
426 for(int i=0; i<n; i++) {
427 int im1 = std::max(i-1, 0);
428 int ip1 = std::min(i+1, n-1);
429 if( ((m_y[im1]<=m_y[i]) && (m_y[i]<=m_y[ip1]) && m_b[i]<0.0) ||
430 ((m_y[im1]>=m_y[i]) && (m_y[i]>=m_y[ip1]) && m_b[i]>0.0) ) {
431 modified=true;
432 m_b[i]=0.0;
433 }
434 }
435 // if input data is monotonic (b[i], b[i+1], avg have all the same sign)
436 // ensure a sufficient criteria for monotonicity is satisfied:
437 // sqrt(b[i]^2+b[i+1]^2) <= 3 |avg|, with avg=(y[i+1]-y[i])/h,
438 for(int i=0; i<n-1; i++) {
439 double h = m_x[i+1]-m_x[i];
440 double avg = (m_y[i+1]-m_y[i])/h;
441 if( avg==0.0 && (m_b[i]!=0.0 || m_b[i+1]!=0.0) ) {
442 modified=true;
443 m_b[i]=0.0;
444 m_b[i+1]=0.0;
445 } else if( (m_b[i]>=0.0 && m_b[i+1]>=0.0 && avg>0.0) ||
446 (m_b[i]<=0.0 && m_b[i+1]<=0.0 && avg<0.0) ) {
447 // input data is monotonic
448 double r = sqrt(m_b[i]*m_b[i]+m_b[i+1]*m_b[i+1])/std::fabs(avg);
449 if(r>3.0) {
450 // sufficient criteria for monotonicity: r<=3
451 // adjust b[i] and b[i+1]
452 modified=true;
453 m_b[i] *= (3.0/r);
454 m_b[i+1] *= (3.0/r);
455 }
456 }
457 }
458
459 if(modified==true) {
460 set_coeffs_from_b();
461 m_made_monotonic=true;
462 }
463
464 return modified;
465}

467// return the closest idx so that m_x[idx] <= x (return 0 if x<m_x[0])
468size_t spline::find_closest(double x) const
469{
470 std::vector<double>::const_iterator it;
471 it=std::upper_bound(m_x.begin(),m_x.end(),x); // *it > x
472 size_t idx = std::max( int(it-m_x.begin())-1, 0); // m_x[idx] <= x
473 return idx;
474}

476double spline::operator() (double x) const
477{
478 // polynomial evaluation using Horner's scheme
479 // TODO: consider more numerically accurate algorithms, e.g.:
480 // - Clenshaw
481 // - Even-Odd method by A.C.R. Newbery
482 // - Compensated Horner Scheme
483 size_t n=m_x.size();
484 size_t idx=find_closest(x);
485
486 double h=x-m_x[idx];
487 double interpol;
488 if(x<m_x[0]) {
489 // extrapolation to the left
490 interpol=(m_c0*h + m_b[0])*h + m_y[0];
491 } else if(x>m_x[n-1]) {
492 // extrapolation to the right
493 interpol=(m_c[n-1]*h + m_b[n-1])*h + m_y[n-1];
494 } else {
495 // interpolation
496 interpol=((m_d[idx]*h + m_c[idx])*h + m_b[idx])*h + m_y[idx];
497 }
498 return interpol;
499}
500
501double spline::deriv(int order, double x) const
502{
503 assert(order>0);
504 size_t n=m_x.size();
505 size_t idx = find_closest(x);
506
507 double h=x-m_x[idx];
508 double interpol;
509 if(x<m_x[0]) {
510 // extrapolation to the left
511 switch(order) {
512 case 1:
513 interpol=2.0*m_c0*h + m_b[0];
514 break;
515 case 2:
516 interpol=2.0*m_c0;
517 break;
518 default:
519 interpol=0.0;
520 break;
521 }
522 } else if(x>m_x[n-1]) {
523 // extrapolation to the right
524 switch(order) {
525 case 1:
526 interpol=2.0*m_c[n-1]*h + m_b[n-1];
527 break;
528 case 2:
529 interpol=2.0*m_c[n-1];
530 break;
531 default:
532 interpol=0.0;
533 break;
534 }
535 } else {
536 // interpolation
537 switch(order) {
538 case 1:
539 interpol=(3.0*m_d[idx]*h + 2.0*m_c[idx])*h + m_b[idx];
540 break;
541 case 2:
542 interpol=6.0*m_d[idx]*h + 2.0*m_c[idx];
543 break;
544 case 3:
545 interpol=6.0*m_d[idx];
546 break;
547 default:
548 interpol=0.0;
549 break;
550 }
551 }
552 return interpol;
553}

555std::vector<double> spline::solve(double y, bool ignore_extrapolation) const
556{
557 std::vector<double> x; // roots for the entire spline
558 std::vector<double> root; // roots for each piecewise cubic
559 const size_t n=m_x.size();
560
561 // left extrapolation
562 if(ignore_extrapolation==false) {
563 root = internal::solve_cubic(m_y[0]-y,m_b[0],m_c0,0.0,1);
564 for(size_t j=0; j<root.size(); j++) {
565 if(root[j]<0.0) {
566 x.push_back(m_x[0]+root[j]);
567 }
568 }
569 }
570
571 // brute force check if piecewise cubic has roots in their resp. segment
572 // TODO: make more efficient
573 for(size_t i=0; i<n-1; i++) {
574 root = internal::solve_cubic(m_y[i]-y,m_b[i],m_c[i],m_d[i],1);
575 for(size_t j=0; j<root.size(); j++) {
576 double h = (i>0) ? (m_x[i]-m_x[i-1]) : 0.0;
577 double eps = internal::get_eps()*512.0*std::min(h,1.0);
578 if( (-eps<=root[j]) && (root[j]<m_x[i+1]-m_x[i]) ) {
579 double new_root = m_x[i]+root[j];
580 if(x.size()>0 && x.back()+eps > new_root) {
581 x.back()=new_root; // avoid spurious duplicate roots
582 } else {
583 x.push_back(new_root);
584 }
585 }
586 }
587 }
588
589 // right extrapolation
590 if(ignore_extrapolation==false) {
591 root = internal::solve_cubic(m_y[n-1]-y,m_b[n-1],m_c[n-1],0.0,1);
592 for(size_t j=0; j<root.size(); j++) {
593 if(0.0<=root[j]) {
594 x.push_back(m_x[n-1]+root[j]);
595 }
596 }
597 }
598
599 return x;
600};

684// LR-Decomposition of a band matrix
685void band_matrix::lu_decompose()
686{
687 int i_max,j_max;
688 int j_min;
689 double x;
690
691 // preconditioning
692 // normalize column i so that a_ii=1
693 for(int i=0; i<this->dim(); i++) {
694 assert(this->operator()(i,i)!=0.0);
695 this->saved_diag(i)=1.0/this->operator()(i,i);
696 j_min=std::max(0,i-this->num_lower());
697 j_max=std::min(this->dim()-1,i+this->num_upper());
698 for(int j=j_min; j<=j_max; j++) {
699 this->operator()(i,j) *= this->saved_diag(i);
700 }
701 this->operator()(i,i)=1.0; // prevents rounding errors
702 }
703
704 // Gauss LR-Decomposition
705 for(int k=0; k<this->dim(); k++) {
706 i_max=std::min(this->dim()-1,k+this->num_lower()); // num_lower not a mistake!
707 for(int i=k+1; i<=i_max; i++) {
708 assert(this->operator()(k,k)!=0.0);
709 x=-this->operator()(i,k)/this->operator()(k,k);
710 this->operator()(i,k)=-x; // assembly part of L
711 j_max=std::min(this->dim()-1,k+this->num_upper());
712 for(int j=k+1; j<=j_max; j++) {
713 // assembly part of R
714 this->operator()(i,j)=this->operator()(i,j)+x*this->operator()(k,j);
715 }
716 }
717 }
718}
719// solves Ly=b
720std::vector<double> band_matrix::l_solve(const std::vector<double>& b) const
721{
722 assert( this->dim()==(int)b.size() );
723 std::vector<double> x(this->dim());
724 int j_start;
725 double sum;
726 for(int i=0; i<this->dim(); i++) {
727 sum=0;
728 j_start=std::max(0,i-this->num_lower());
729 for(int j=j_start; j<i; j++) sum += this->operator()(i,j)*x[j];
730 x[i]=(b[i]*this->saved_diag(i)) - sum;
731 }
732 return x;
733}
734// solves Rx=y
735std::vector<double> band_matrix::r_solve(const std::vector<double>& b) const
736{
737 assert( this->dim()==(int)b.size() );
738 std::vector<double> x(this->dim());
739 int j_stop;
740 double sum;
741 for(int i=this->dim()-1; i>=0; i--) {
742 sum=0;
743 j_stop=std::min(this->dim()-1,i+this->num_upper());
744 for(int j=i+1; j<=j_stop; j++) sum += this->operator()(i,j)*x[j];
745 x[i]=( b[i] - sum ) / this->operator()(i,i);
746 }
747 return x;
748}

750std::vector<double> band_matrix::lu_solve(const std::vector<double>& b,
751 bool is_lu_decomposed)
752{
753 assert( this->dim()==(int)b.size() );
754 std::vector<double> x,y;
755 if(is_lu_decomposed==false) {
756 this->lu_decompose();
757 }
758 y=this->l_solve(b);
759 x=this->r_solve(y);
760 return x;
761}

770// solutions for a + b*x = 0
771std::vector<double> solve_linear(double a, double b)
772{
773 std::vector<double> x; // roots
774 if(b==0.0) {
775 if(a==0.0) {
776 // 0*x = 0
777 x.resize(1);
778 x[0] = 0.0; // any x solves it but we need to pick one
779 return x;
780 } else {
781 // 0*x + ... = 0, no solution
782 return x;
783 }
784 } else {
785 x.resize(1);
786 x[0] = -a/b;
787 return x;
788 }
789}

791// solutions for a + b*x + c*x^2 = 0
792std::vector<double> solve_quadratic(double a, double b, double c,
793 int newton_iter=0)
794{
795 if(c==0.0) {
796 return solve_linear(a,b);
797 }
798 // rescale so that we solve x^2 + 2p x + q = (x+p)^2 + q - p^2 = 0
799 double p=0.5*b/c;
800 double q=a/c;
801 double discr = p*p-q;
802 const double eps=0.5*internal::get_eps();
803 double discr_err = (6.0*(p*p)+3.0*fabs(q)+fabs(discr))*eps;
804
805 std::vector<double> x; // roots
806 if(fabs(discr)<=discr_err) {
807 // discriminant is zero --> one root
808 x.resize(1);
809 x[0] = -p;
810 } else if(discr<0) {
811 // no root
812 } else {
813 // two roots
814 x.resize(2);
815 x[0] = -p - sqrt(discr);
816 x[1] = -p + sqrt(discr);
817 }
818
819 // improve solution via newton steps
820 for(size_t i=0; i<x.size(); i++) {
821 for(int k=0; k<newton_iter; k++) {
822 double f = (c*x[i] + b)*x[i] + a;
823 double f1 = 2.0*c*x[i] + b;
824 // only adjust if slope is large enough
825 if(fabs(f1)>1e-8) {
826 x[i] -= f/f1;
827 }
828 }
829 }
830
831 return x;
832}
833
834// solutions for the cubic equation: a + b*x +c*x^2 + d*x^3 = 0
835// this is a naive implementation of the analytic solution without
836// optimisation for speed or numerical accuracy
837// newton_iter: number of newton iterations to improve analytical solution
838// see also
839// gsl: gsl_poly_solve_cubic() in solve_cubic.c
840// octave: roots.m - via eigenvalues of the Frobenius companion matrix
841std::vector<double> solve_cubic(double a, double b, double c, double d,
842 int newton_iter)
843{
844 if(d==0.0) {
845 return solve_quadratic(a,b,c,newton_iter);
846 }
847
848 // convert to normalised form: a + bx + cx^2 + x^3 = 0
849 if(d!=1.0) {
850 a/=d;
851 b/=d;
852 c/=d;
853 }
854
855 // convert to depressed cubic: z^3 - 3pz - 2q = 0
856 // via substitution: z = x + c/3
857 std::vector<double> z; // roots of the depressed cubic
858 double p = -(1.0/3.0)*b + (1.0/9.0)*(c*c);
859 double r = 2.0*(c*c)-9.0*b;
860 double q = -0.5*a - (1.0/54.0)*(c*r);
861 double discr=p*p*p-q*q; // discriminant
862 // calculating numerical round-off errors with assumptions:
863 // - each operation is precise but each intermediate result x
864 // when stored has max error of x*eps
865 // - only multiplication with a power of 2 introduces no new error
866 // - a,b,c,d and some fractions (e.g. 1/3) have rounding errors eps
867 // - p_err << |p|, q_err << |q|, ... (this is violated in rare cases)
868 // would be more elegant to use boost::numeric::interval<double>
869 const double eps = internal::get_eps();
870 double p_err = eps*((3.0/3.0)*fabs(b)+(4.0/9.0)*(c*c)+fabs(p));
871 double r_err = eps*(6.0*(c*c)+18.0*fabs(b)+fabs(r));
872 double q_err = 0.5*fabs(a)*eps + (1.0/54.0)*fabs(c)*(r_err+fabs(r)*3.0*eps)
873 + fabs(q)*eps;
874 double discr_err = (p*p) * (3.0*p_err + fabs(p)*2.0*eps)
875 + fabs(q) * (2.0*q_err + fabs(q)*eps) + fabs(discr)*eps;
876
877 // depending on the discriminant we get different solutions
878 if(fabs(discr)<=discr_err) {
879 // discriminant zero: one or two real roots
880 if(fabs(p)<=p_err) {
881 // p and q are zero: single root
882 z.resize(1);
883 z[0] = 0.0; // triple root
884 } else {
885 z.resize(2);
886 z[0] = 2.0*q/p; // single root
887 z[1] = -0.5*z[0]; // double root
888 }
889 } else if(discr>0) {
890 // three real roots: via trigonometric solution
891 z.resize(3);
892 double ac = (1.0/3.0) * acos( q/(p*sqrt(p)) );
893 double sq = 2.0*sqrt(p);
894 z[0] = sq * cos(ac);
895 z[1] = sq * cos(ac-2.0*M_PI/3.0);
896 z[2] = sq * cos(ac-4.0*M_PI/3.0);
897 } else if (discr<0.0) {
898 // single real root: via Cardano's fromula
899 z.resize(1);
900 double sgnq = (q >= 0 ? 1 : -1);
901 double basis = fabs(q) + sqrt(-discr);
902 double C = sgnq * pow(basis, 1.0/3.0); // c++11 has std::cbrt()
903 z[0] = C + p/C;
904 }
905 for(size_t i=0; i<z.size(); i++) {
906 // convert depressed cubic roots to original cubic: x = z - c/3
907 z[i] -= (1.0/3.0)*c;
908 // improve solution via newton steps
909 for(int k=0; k<newton_iter; k++) {
910 double f = ((z[i] + c)*z[i] + b)*z[i] + a;
911 double f1 = (3.0*z[i] + 2.0*c)*z[i] + b;
912 // only adjust if slope is large enough
913 if(fabs(f1)>1e-8) {
914 z[i] -= f/f1;
915 }
916 }
917 }
918 // ensure if a=0 we get exactly x=0 as root
919 // TODO: remove this fudge
920 if(a==0.0) {
921 assert(z.size()>0); // cubic should always have at least one root
922 double xmin=fabs(z[0]);
923 size_t imin=0;
924 for(size_t i=1; i<z.size(); i++) {
925 if(xmin>fabs(z[i])) {
926 xmin=fabs(z[i]);
927 imin=i;
928 }
929 }
930 z[imin]=0.0; // replace the smallest absolute value with 0
931 }
932 std::sort(z.begin(), z.end());
933 return z;
934}
