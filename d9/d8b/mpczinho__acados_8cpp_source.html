32void MPCzinhoAcadosSolver::apply_cost_weights() {
33 // Set from YAML so weights are tunable without regenerating; generated weights are the fallback.
34 const int N = nlp_dims_->N;
35 auto set_stage_weights = [this](const std::vector<double>& weights, int stage,
36 const char* label) {
37 if (weights.empty()) return false;
38 int dims_out[2] = {0, 0};
39 ocp_nlp_cost_dims_get_from_attr(nlp_config_, nlp_dims_, nlp_out_, stage, "W", dims_out);
40 const int ny = dims_out[0];
41 if (static_cast<int>(weights.size()) != ny) {
42 RCLCPP_ERROR(rclcpp::get_logger("MPCzinhoAcadosSolver"),
43 "%s has %zu entries but stage %d expects %d; keeping generated weights",
44 label, weights.size(), stage, ny);
45 return false;
46 }
47 std::vector<double> W(static_cast<size_t>(ny) * ny, 0.0);
48 for (int i = 0; i < ny; ++i) W[i * ny + i] = weights[i];
49 ocp_nlp_cost_model_set(nlp_config_, nlp_dims_, nlp_in_, stage, "W", W.data());
50 return true;
51 };
52
53 bool ok = true;
54 for (int stage = 0; stage < N; ++stage) {
55 ok &= set_stage_weights(this->control_params_->lateral_mpc_cost_weights_, stage, "lateral_mpc_cost_weights");
56 }
57 ok &= set_stage_weights(this->control_params_->lateral_mpc_terminal_cost_weights_, N,
58 "lateral_mpc_terminal_cost_weights");
59 if (ok && !this->control_params_->lateral_mpc_cost_weights_.empty()) {
60 RCLCPP_INFO(rclcpp::get_logger("MPCzinhoAcadosSolver"), "Applied cost weights from config");
61 }
62}

152 // Update averages

158 this->regularization_count_++;

181common_lib::structures::ControlCommand MPCzinhoAcadosSolver::solve(int* solver_status) {
182 common_lib::structures::ControlCommand command;
183 if (!(this->has_state_ && this->has_path_)) {
184 return command;
185 }
186
187 this->set_path_point_per_stage();
188
189 double first_x = this->parameters_per_stage[0];
190 double first_y = this->parameters_per_stage[1];
191 double first_v = this->parameters_per_stage[2];
192 double first_orientation = this->parameters_per_stage[3];
193 first_x -= this->latest_state_.x;
194 first_y -= this->latest_state_.y;
195 first_orientation -= this->latest_state_.orientation;
196 first_v -= this->latest_state_.velocity_x;
197 //DEBUG PRINT
198 if (std::fabs(first_x) > 0.01 || std::fabs(first_y) > 0.01 || std::fabs(first_v) > 0.01 || std::fabs(first_orientation) > 0.01) {
199 RCLCPP_ERROR(rclcpp::get_logger("AcadosSolver"), "ERROR: first point doens't match state x:%.2f, y:%.2f, v:%.2f, orientation:%.2f", this->latest_state_.x, this->latest_state_.y, this->latest_state_.velocity_x, this->latest_state_.orientation);
200 }
201
202
203 if (!this->is_initialized_) {
204 this->initialize_solver_memory();
205 }
206
207 int status = mpczinho_acados_solve(this->capsule_);
208 if (status != ACADOS_SUCCESS) {
209 RCLCPP_ERROR(rclcpp::get_logger("MPCzinhoAcadosSolver"), "Acados solver failed with status %d", status);
210 if (this->sanity_check_output()) {
211 *solver_status = 1; // Positive to indicate benign failure (e.g. infeasibility)
212 } else {
213 print_debug_info(); // If solver failed, print debug info
214 *solver_status = -1; // Negative to indicate malign failure
215 }
216 }
217
218 this->update_mpc_stats();
219 // Extracting 3 controls
220 std::vector<common_lib::structures::ControlCommand> full_solution = this->get_full_solution();
221
222 // Compensate for solver delay
223 double total_solver_time_ms = this->_execution_times_->at(0);
224 // Add an extra 5 ms as estimate for state estimation delay
225 double total_delay_ms = total_solver_time_ms + 5.0;
226 double total_delay_s = total_delay_ms / 1000.0;
227
228 double time_step = this->control_params_->lateral_mpc_prediction_horizon_seconds_ / static_cast<double>(this->control_params_->lateral_mpc_prediction_horizon_steps_);
229 unsigned int steps_ahead = static_cast<unsigned int>(std::floor(total_delay_s / time_step));
230 if (steps_ahead >= full_solution.size() - 1) {
231 RCLCPP_WARN(rclcpp::get_logger("AcadosSolver"), "Total delay of %.2f ms exceeds prediction horizon, using last available control", total_delay_ms);
232 return full_solution.back();
233 }
234
235 common_lib::structures::ControlCommand command_zero = full_solution[steps_ahead];
236 common_lib::structures::ControlCommand command_one = full_solution[steps_ahead + 1];
237
238 // Linear interpolation of control commands
239 double alpha = (total_delay_s - steps_ahead * time_step) / time_step;
240
241 // DEBUG STRING
242 this->total_delay_debug = "Total delay: " + std::to_string(total_delay_ms) + " ms, steps ahead: " + std::to_string(steps_ahead) + ", interpolation alpha: " + std::to_string(alpha);
243
244 command.throttle_rl = (1 - alpha) * command_zero.throttle_rl + alpha * command_one.throttle_rl;
245 command.throttle_rr = (1 - alpha) * command_zero.throttle_rr + alpha * command_one.throttle_rr;
246 command.steering_angle = (1 - alpha) * command_zero.steering_angle + alpha * command_one.steering_angle;
247 return command;
248}
