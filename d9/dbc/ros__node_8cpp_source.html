43void ControlNode::control_timer_callback() {
44 if (!go_signal_) return;
45
46 rclcpp::Time start = this->now();
47 common_lib::structures::ControlCommand command = this->controller_->get_control_command();
48 double execution_time = (this->now() - start).seconds() * 1000;
49
50 this->publish_command(command);
51
52 this->_execution_times_->at(0) = execution_time;
53 this->_execution_times_->at(1) =
54 (this->_execution_times_->at(1) * static_cast<double>(this->number_of_loops_executed_) +
55 execution_time) /
56 (static_cast<double>(this->number_of_loops_executed_ + 1));
57 this->number_of_loops_executed_++;
58 std_msgs::msg::Float64MultiArray execution_time_msg;
59 execution_time_msg.data = *this->_execution_times_;
60 this->execution_time_pub_->publish(execution_time_msg);
61 this->controller_->publish_solver_data(shared_from_this(), publisher_map_);
62}
