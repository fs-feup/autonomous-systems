108TEST(VelocityPlanning, AccelerationBrakesAllTheWayToZero) {
109 VelocityPlanning velocity_planning(acceleration_config());
110 // Stopping from sqrt(759) m/s at 5 m/s^2 needs 75.9 m, so the path has to be long enough to
111 // contain it: 180 points is 179 m, leaving 104 m of braking room after the 75 m run.
112 auto path = straight_run(180);
113 velocity_planning.acceleration_velocity(path, 75.0);
114
115 EXPECT_DOUBLE_EQ(path.back().ideal_velocity, 0.0);
116
117 // Stays stopped: no point after the first zero recovers to the minimum velocity.
118 bool stopped = false;
119 for (const auto &point : path) {
120 if (point.ideal_velocity == 0.0) {
121 stopped = true;
122 } else if (stopped) {
123 ADD_FAILURE() << "velocity recovered to " << point.ideal_velocity << " after stopping";
124 break;
125 }
126 }
127 EXPECT_TRUE(stopped);
128}
