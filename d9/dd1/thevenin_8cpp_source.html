9double Thevenin::calculate_allowed_current(double requested_current) const {
10 if (requested_current <= 0.0) {
11 return std::max(requested_current,
12 -this->car_parameters_->battery_parameters->max_charge_current);
13 }
14
15 double ocv_cell = get_cell_ocv(soc_);
16 double r0_cell = get_cell_r0(soc_);
17
18 // Calculate pack-level parameters
19 double ocv_eff_pack =
20 (ocv_cell - v_rc_) * this->car_parameters_->battery_parameters->cells_series;
21 double r0_pack =
22 r0_cell * (static_cast<double>(this->car_parameters_->battery_parameters->cells_series) /
23 this->car_parameters_->battery_parameters->cells_parallel);
24
25 // FSG rules 80kW limit
26 double disc_p = std::pow(ocv_eff_pack, 2) - 4.0f * r0_pack * 80000.0f;
27 double i_limit_80kw = (disc_p < 0.0f) ? (ocv_eff_pack / (2.0f * r0_pack))
28 : ((ocv_eff_pack - std::sqrt(disc_p)) / (2.0f * r0_pack));
29
30 // Minimum voltage limit
31 double i_limit_v_min =
32 (ocv_eff_pack - this->car_parameters_->battery_parameters->min_voltage) / r0_pack;
33
34 // Maximum discharge current
35 double i_limit_hardware = this->car_parameters_->battery_parameters->max_discharge_current;
36
37 // Calculate the maximum allowed current based on all limits
38 double max_allowed = std::min({i_limit_80kw, i_limit_v_min, i_limit_hardware});
39
40 // Return the minimum of the requested current and the maximum allowed current
41 return std::min(requested_current, max_allowed);
42}

80void Thevenin::update_state(double current_draw, double dt) {
81 // Update SoC (Coulomb Counting)
82 soc_ -= (current_draw * dt) / (3600.0f * this->car_parameters_->battery_parameters->capacity_ah);
83 soc_ = std::clamp(soc_, 0.0, 1.0);
84
85 // Update Polarization Voltage (V_rc)
86 double r1 = get_cell_r1(soc_);
87 double c1 = get_cell_c1(soc_);
88
89 double tau = std::max(r1 * c1, 1e-6);
90 double i_cell = current_draw / this->car_parameters_->battery_parameters->cells_parallel;
91
92 double ak = std::exp(-dt / tau);
93 double bk = r1 * (1.0f - ak);
94
95 this->v_rc_ = ak * this->v_rc_ + bk * i_cell;
96 this->current_ = current_draw;
97}
