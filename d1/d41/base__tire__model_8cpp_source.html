7Eigen::Vector2d calculate_contact_patch_velocity(const TireInput& tire_input, double track_width) {
8 const bool is_front = tire_input.tire == FL || tire_input.tire == FR;
9 const bool is_left = tire_input.tire == FL || tire_input.tire == RL;
10
11 const double wheel_position_x = is_front ? tire_input.distance_to_CG : -tire_input.distance_to_CG;
12 const double wheel_position_y = is_left ? track_width / 2.0 : -track_width / 2.0;
13
14 const double vehicle_x = tire_input.vx - tire_input.yaw_rate * wheel_position_y;
15 const double vehicle_y = tire_input.vy + tire_input.yaw_rate * wheel_position_x;
16
17 const double cos_delta = std::cos(tire_input.steering_angle);
18 const double sin_delta = std::sin(tire_input.steering_angle);
19
20 return Eigen::Vector2d(vehicle_x * cos_delta + vehicle_y * sin_delta,
21 -vehicle_x * sin_delta + vehicle_y * cos_delta);
22}

26void TireModel::calculate_slip_angle_front_not_transient(TireInput& tire_input) {
27 constexpr double longitudinal_epsilon = 0.5;
28
29 const Eigen::Vector2d contact_patch =
30 calculate_contact_patch_velocity(tire_input, car_parameters_->track_width);
31 const double Vcx = contact_patch.x();
32 const double Vcy = contact_patch.y();
33 tire_input.vcx = Vcx;
34 tire_input.vcy = Vcy;
35
36 const double Vlong_reg = std::hypot(Vcx, longitudinal_epsilon);
37 const double direction = Vcx / Vlong_reg;
38 tire_input.slip_angle = std::atan2(Vcy, Vlong_reg) * direction;
39 tire_input.last_slip_angle[tire_input.tire] = tire_input.slip_angle;

58void TireModel::calculate_slip_ratio_not_transient(TireInput& tire_input) {
59 constexpr double V_floor = 1.0;
60
61 const Eigen::Vector2d contact_patch =
62 calculate_contact_patch_velocity(tire_input, car_parameters_->track_width);
63 const double Vcx = contact_patch.x();
64 const double Vw =
65 tire_input.wheel_angular_speed * car_parameters_->tire_parameters->effective_tire_r;
66 tire_input.vcx = Vcx;
67 tire_input.vcy = contact_patch.y();
68
69 if (std::abs(Vcx) < 0.01 && std::abs(Vw) < 0.01) {
70 tire_input.slip_ratio = 0.0;
71 tire_input.last_slip_ratio[tire_input.tire] = 0.0;
72 return;
73 }
74
75 const double denominator = std::max({std::abs(Vw), std::abs(Vcx), V_floor});
76 tire_input.slip_ratio = std::clamp((Vw - Vcx) / denominator, -1.0, 1.0);
77 tire_input.last_slip_ratio[tire_input.tire] = tire_input.slip_ratio;
78}

109void TireModel::calculate_slip_angle_front(TireInput& tire_input) {
110 double L_alpha = car_parameters_->tire_parameters->slip_angle_relaxation_length;
111 if (L_alpha <= 0.001) {
112 calculate_slip_angle_front_not_transient(tire_input);
113 return;
114 }
115
116 constexpr double longitudinal_epsilon = 0.5;
117
118 const Eigen::Vector2d contact_patch =
119 calculate_contact_patch_velocity(tire_input, car_parameters_->track_width);
120 const double Vcx = contact_patch.x();
121 const double Vcy = contact_patch.y();
122 tire_input.vcx = Vcx;
123 tire_input.vcy = Vcy;
124
125 const double Vlong_reg = std::hypot(Vcx, longitudinal_epsilon);
126 const double direction = Vcx / Vlong_reg;
127 const double alpha_target = std::atan2(Vcy, Vlong_reg) * direction;
128 const double V_mag = std::max(std::abs(Vcx), 0.5); // Avoid division by zero
129
130 const double relaxation = std::exp(-(V_mag / L_alpha) * tire_input.dt);
131 tire_input.slip_angle =
132 alpha_target + (tire_input.last_slip_angle[tire_input.tire] - alpha_target) * relaxation;
133 tire_input.last_slip_angle[tire_input.tire] = tire_input.slip_angle;
134}

166 calculate_slip_ratio_not_transient(tire_input);
167 return;
168 }
169

171
172 const Eigen::Vector2d contact_patch =
173 calculate_contact_patch_velocity(tire_input, car_parameters_->track_width);
174 const double Vcx = contact_patch.x();
175 tire_input.vcx = Vcx;

178

180 tire_input.slip_ratio = 0.0;
181 tire_input.last_slip_ratio[tire_input.tire] = 0.0;
182 return;
183 }
184
185 const double denominator = std::max({std::abs(Vw), std::abs(Vcx), V_floor});
186 const double slip_target = std::clamp((Vw - Vcx) / denominator, -1.0, 1.0);
187 double V_mag = std::max(std::abs(Vcx), 0.5);
188
189 double relaxation = std::exp(-(V_mag / L) * tire_input.dt);
190 tire_input.slip_ratio =
191 slip_target + (tire_input.last_slip_ratio[tire_input.tire] - slip_target) * relaxation;
192 tire_input.slip_ratio = std::clamp(tire_input.slip_ratio, -1.0, 1.0);
