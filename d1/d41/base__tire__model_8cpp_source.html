13

16 v_wheel_x * cos(tire_input.steering_angle) + v_wheel_y * sin(tire_input.steering_angle);

18 -v_wheel_x * sin(tire_input.steering_angle) + v_wheel_y * cos(tire_input.steering_angle);
19

22 if (std::sqrt(Vcx * Vcx + Vcy * Vcy) >= 0.05) {

24 double direction = (Vcx >= 0.0) ? 1.0 : -1.0;
25 alpha_target = atan2(Vcy, Vlong_reg) * direction;
26 }
27

29 double L_alpha = car_parameters_->tire_parameters->relaxation_length;

31

33 double alpha_derivative =

84void TireModel::calculate_slip_ratio(TireInput& tire_input) {

88 double v_wheel_x =

92 v_wheel_y += (tire_input.yaw_rate * tire_input.distance_to_CG);
93 } else {
94 v_wheel_y -= (tire_input.yaw_rate * tire_input.distance_to_CG);
95 }
96
97 // Project to the wheel frame
98 double Vcx =
99 v_wheel_x * cos(tire_input.steering_angle) + v_wheel_y * sin(tire_input.steering_angle);
100 double Vw = tire_input.wheel_angular_speed * car_parameters_->tire_parameters->effective_tire_r;
101
102 // Low speed stabilization
103 if (std::abs(Vcx) < 0.01 && std::abs(Vw) < 0.01) {
104 tire_input.slip_ratio = 0.0;
105 tire_input.last_slip_ratio[tire_input.tire] = 0.0;
106 return;
107 }
108
109 // Target slip ratio
110 double stabilizer_epsilon = 0.1;
111 double denominator = std::sqrt(Vcx * Vcx + stabilizer_epsilon * stabilizer_epsilon);
112 double slip_target = (Vw - Vcx) / denominator;
113 slip_target = std::clamp(slip_target, -1.0, 1.0);
114
115 // Transient slip ratio
116 double L = car_parameters_->tire_parameters->relaxation_length;
117 double V_mag = std::max(std::abs(Vcx), 0.5);
118
119 // First-order lag system: d(slip_ratio)/dt = (V / L) * (slip_target - slip_current)
120 double slip_derivative =
121 (V_mag / L) * (slip_target - tire_input.last_slip_ratio[tire_input.tire]);
122
123 // Integrate
