181double LapTimerStatistics::signed_line_distance(
182 const TimingLine& line, const common_lib::structures::Position& position) const {
183 const double line_x = std::get<1>(line).x - std::get<0>(line).x;
184 const double line_y = std::get<1>(line).y - std::get<0>(line).y;
185 const double length = std::hypot(line_x, line_y);
186 if (length <= std::numeric_limits<double>::epsilon()) {
187 return 0.0;
188 }
189 return (line_x * (position.y - std::get<0>(line).y) -
190 line_y * (position.x - std::get<0>(line).x)) /
191 length;
192}

194bool LapTimerStatistics::is_inside_line_gate(
195 const TimingLine& line, const common_lib::structures::Position& position) const {
196 const double line_x = std::get<1>(line).x - std::get<0>(line).x;
197 const double line_y = std::get<1>(line).y - std::get<0>(line).y;
198 const double length_sq = line_x * line_x + line_y * line_y;
199 if (length_sq <= std::numeric_limits<double>::epsilon()) {
200 return false;
201 }
202
203 const double t =
204 ((position.x - std::get<0>(line).x) * line_x + (position.y - std::get<0>(line).y) * line_y) /
205 length_sq;
206 const double margin_ratio = timing_line_gate_margin_m_ / std::sqrt(length_sq);
207 return t >= -margin_ratio && t <= 1.0 + margin_ratio;
208}

210bool LapTimerStatistics::crossed_line(const TimingLine& line, std::optional<double> previous_side,
211 const common_lib::structures::Position& position,
212 double current_side) const {
213 if (!previous_side || !is_inside_line_gate(line, position)) {
214 return false;
215 }
216
217 return (*previous_side <= 0.0 && current_side > 0.0) ||
218 (*previous_side >= 0.0 && current_side < 0.0);
219}

221bool LapTimerStatistics::crossed_line_from_approach_side(
222 const TimingLine& line, std::optional<double> previous_side,
223 const common_lib::structures::Position& position, double current_side,
224 double approach_side) const {
225 if (!previous_side || !is_inside_line_gate(line, position)) {
226 return false;
227 }
228
229 if (std::abs(approach_side) <= std::numeric_limits<double>::epsilon()) {
230 return crossed_line(line, previous_side, position, current_side);
231 }
232
233 return approach_side > 0.0 ? *previous_side >= 0.0 && current_side < 0.0
234 : *previous_side <= 0.0 && current_side > 0.0;
235}
