14double MapBasedMotor::get_efficiency(double torque, double rpm) const {
15 // Handle efficiency map
16 if (car_parameters_->motor_parameters->efficiency_map.empty()) {
17 return 0.95f; // Default efficiency if no map provided
18 }
19
20 // Find closest RPM entry
21 auto rpm_upper = car_parameters_->motor_parameters->efficiency_map.lower_bound(rpm);
22
23 if (rpm_upper == car_parameters_->motor_parameters->efficiency_map.end()) {
24 // RPM beyond map, use last entry
25 rpm_upper = std::prev(car_parameters_->motor_parameters->efficiency_map.end());
26 }
27
28 if (rpm_upper == car_parameters_->motor_parameters->efficiency_map.begin()) {
29 // RPM before map, use first entry
30 // Just use this efficiency map
31 return interpolate_from_map(rpm_upper->second, torque);
32 }
33
34 // Interpolate between two RPM values
35 auto rpm_lower = std::prev(rpm_upper);
36
37 double eff_lower = interpolate_from_map(rpm_lower->second, torque);
38 double eff_upper = interpolate_from_map(rpm_upper->second, torque);
39
40 double rpm1 = rpm_lower->first;
41 double rpm2 = rpm_upper->first;
42
43 // Linear interpolation
44 double t = (rpm - rpm1) / (rpm2 - rpm1);
45 return eff_lower + t * (eff_upper - eff_lower);
46}

48double MapBasedMotor::get_max_torque_at_rpm(double rpm) const {
49 double thermal_ratio = thermal_state_ / thermal_capacity_;
50
51 // Calculate limits based on current state
52 double current_max_t =
53 car_parameters_->motor_parameters->max_peak_torque -
54 (thermal_ratio * (car_parameters_->motor_parameters->max_peak_torque -
55 car_parameters_->motor_parameters->max_continuous_torque));
56
57 double current_max_p =
58 car_parameters_->motor_parameters->max_peak_power -
59 (thermal_ratio * (car_parameters_->motor_parameters->max_peak_power -
60 car_parameters_->motor_parameters->max_continuous_power));
61
62 double current_max_rpm =
63 car_parameters_->motor_parameters->max_peak_rpm -
64 (thermal_ratio * (car_parameters_->motor_parameters->max_peak_rpm -
65 car_parameters_->motor_parameters->max_continuous_rpm));
66
67 if (rpm >= current_max_rpm) {
68 return 0.0f; // Should not happen due to decay, but just in case
69 }
70
71 if (rpm < 0.1) {
72 return current_max_t;
73 }
74
75 double omega = (rpm * 2.0 * M_PI) / 60.0;
76
77 // Normal constant torque / constant power behavior
78 double torque_limit = std::min(current_max_t, current_max_p / omega);
79
80 // Fade torque to zero as we approach max RPM
81 double fade_start_rpm = car_parameters_->motor_parameters->fade_start * current_max_rpm;
82 if (rpm > fade_start_rpm) {
83 double fade =
84 (current_max_rpm - rpm) / (current_max_rpm - fade_start_rpm);
85 torque_limit = std::max(0.0, fade * torque_limit);
86 }
87
88 return torque_limit;
89}

91void MapBasedMotor::update_state(double current_draw, double torque, double dt) {
92 // I²t thermal model
93 // Heat generation: H = (I² - I_cont²) * dt
94 double i_cont_sq = car_parameters_->motor_parameters->max_continuous_current *
95 car_parameters_->motor_parameters->max_continuous_current;
96 double i_current_sq = current_draw * current_draw;
97
98 double heat_generation = std::max(0.0, i_current_sq - i_cont_sq) * dt;
99
100 // Heat dissipation: exponential cooling towards zero
101 double cooling_rate = this->thermal_state_ / car_parameters_->motor_parameters->peak_duration;
102 double heat_dissipation = cooling_rate * dt;
103
104 // Update thermal state
105 this->thermal_state_ += heat_generation - heat_dissipation;
106 this->thermal_state_ = std::max(0.0, std::min(this->thermal_capacity_, this->thermal_state_));
107 this->current_ = current_draw;
108 this->torque_ = torque;
109}

125double MapBasedMotor::interpolate_from_map(const std::map<double, double>& map, double key) const {
126 if (map.empty()) {
127 return 0.0;
128 }
129
130 // Find bounds
131 auto upper = map.lower_bound(key);
132
133 // Key is beyond the map range
134 if (upper == map.end()) {
135 return map.rbegin()->second; // Return last value
136 }
137
138 // Key is before the map range
139 if (upper == map.begin()) {
140 return upper->second; // Return first value
141 }

145

147 double y1 = lower->second;
148 double x2 = upper->first;
149 double y2 = upper->second;
150
151 // Linear interpolation
152 double t = (key - x1) / (x2 - x1);
153 return y1 + t * (y2 - y1);
