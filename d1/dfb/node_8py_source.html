14class RunningStats:
15 def __init__(self) -> None:
16 self.count = 0
17 self.sum = 0.0
18 self.abs_sum = 0.0
19 self.sum_sq = 0.0
20
21 def add(self, value: float) -> None:
22 self.count += 1
23 self.sum += value
24 self.abs_sum += abs(value)
25 self.sum_sq += value * value
26
27 def average(self) -> float:
28 if self.count == 0:
29 return 0.0
30 return self.sum / self.count
31
32 def average_abs(self) -> float:
33 if self.count == 0:
34 return 0.0
35 return self.abs_sum / self.count
36
37 def mse(self) -> float:
38 if self.count == 0:
39 return 0.0
40 return self.sum_sq / self.count
41
42 def rmse(self) -> float:
43 return math.sqrt(self.mse())

211 def _load_adapter_from_global_config(self) -> str:
212 config_path = str(self.get_parameter("global_config_path").value).strip()
213 resolved_path = self._resolve_global_config_path(config_path)
214
215 if resolved_path is None:
216 self.get_logger().warn(
217 "Could not locate config/global/global_config.yaml; using adapter='vehicle'"
218 )
219 return "vehicle"
220
221 try:
222 with resolved_path.open("r", encoding="utf-8") as handle:
223 cfg = yaml.safe_load(handle) or {}
224 adapter = str(cfg.get("global", {}).get("adapter", "vehicle")).strip()
225 return adapter if adapter else "vehicle"
226 except Exception as exc:
227 self.get_logger().warn(
228 f"Failed to read global config at {resolved_path}: {exc}; using adapter='vehicle'"
229 )
230 return "vehicle"

383 def _find_closest_path_point(self):
384 best_index = 0
385 best_dist_sq = float("inf")
386
387 for index, (path_x, path_y, path_v) in enumerate(self._path_points):
388 dx = self._current_x - path_x
389 dy = self._current_y - path_y
390 dist_sq = dx * dx + dy * dy
391 if dist_sq < best_dist_sq:
392 best_dist_sq = dist_sq
393 best_index = index

395 return best_index

397 def _interpolated_path_reference(self):
398 if len(self._path_points) == 1:
399 point_x, point_y, point_v = self._path_points[0]
400 distance = math.sqrt(
401 (self._current_x - point_x) * (self._current_x - point_x)
402 + (self._current_y - point_y) * (self._current_y - point_y)
403 )
404 return distance, 0.0, point_v
405
406 closest_index = self._find_closest_path_point()
407 candidates = []
408
409 if closest_index > 0:
410 start = self._path_points[closest_index - 1]
411 end = self._path_points[closest_index]
412 candidates.append((start, end))
413
414 if closest_index < len(self._path_points) - 1:
415 start = self._path_points[closest_index]
416 end = self._path_points[closest_index + 1]
417 candidates.append((start, end))
418
419 best = None
420 for start, end in candidates:
421 distance, interpolation = self._distance_and_interpolation_to_segment(start, end)
422 if best is None or distance < best[0]:
423 best = (distance, interpolation, start, end)
424
425 if best is None:
426 point_x, point_y, point_v = self._path_points[closest_index]
427 distance = math.sqrt(
428 (self._current_x - point_x) * (self._current_x - point_x)
429 + (self._current_y - point_y) * (self._current_y - point_y)
430 )
431 return distance, 0.0, point_v
432
433 distance, interpolation, start, end = best
434 start_x, start_y, start_v = start
435 end_x, end_y, end_v = end
436
437 ref_heading = math.atan2(end_y - start_y, end_x - start_x)
438 ref_velocity = start_v + interpolation * (end_v - start_v)
439 return distance, ref_heading, ref_velocity

441 def _distance_and_interpolation_to_segment(self, start, end):
442 start_x, start_y, _ = start
443 end_x, end_y, _ = end
444
445 segment_x = end_x - start_x
446 segment_y = end_y - start_y
447 segment_len_sq = segment_x * segment_x + segment_y * segment_y
448
449 if segment_len_sq <= 1e-12:
450 distance = math.sqrt(
451 (self._current_x - start_x) * (self._current_x - start_x)
452 + (self._current_y - start_y) * (self._current_y - start_y)
453 )
454 return distance, 0.0
455
456 interpolation = (
457 (self._current_x - start_x) * segment_x
458 + (self._current_y - start_y) * segment_y
459 ) / segment_len_sq
460 interpolation = max(0.0, min(1.0, interpolation))
461
462 proj_x = start_x + interpolation * segment_x
463 proj_y = start_y + interpolation * segment_y
464 distance = math.sqrt(
465 (self._current_x - proj_x) * (self._current_x - proj_x)
466 + (self._current_y - proj_y) * (self._current_y - proj_y)
467 )
468
469 return distance, interpolation

471 def _reference_heading_at(self, index: int) -> float:
472 if len(self._path_points) < 2:
473 return 0.0
474
475 if index <= 0:
476 x1, y1, _ = self._path_points[0]
477 x2, y2, _ = self._path_points[1]
478 elif index >= len(self._path_points) - 1:
479 x1, y1, _ = self._path_points[-2]
480 x2, y2, _ = self._path_points[-1]
481 else:
482 x1, y1, _ = self._path_points[index - 1]
483 x2, y2, _ = self._path_points[index + 1]
484
485 return math.atan2(y2 - y1, x2 - x1)
