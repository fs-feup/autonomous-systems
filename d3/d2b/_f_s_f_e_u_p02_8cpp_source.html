129 // Update wheel speeds
130 if (state_->ebs_active) {
131 state_->wheels_speed = {0.0, 0.0, 0.0, 0.0};
132 } else {
133 double R = car_parameters_->tire_parameters->effective_tire_r;
134 double I = car_parameters_->tire_parameters->wheel_inertia;
135
136 // Activation function for smooth rolling resistance at low speeds
137 double sign_rl = 2.0 / M_PI * std::atan(10.0 * state_->wheels_speed.rear_left);
138 double sign_rr = 2.0 / M_PI * std::atan(10.0 * state_->wheels_speed.rear_right);
139 double sign_fl = 2.0 / M_PI * std::atan(10.0 * state_->wheels_speed.front_left);
140 double sign_fr = 2.0 / M_PI * std::atan(10.0 * state_->wheels_speed.front_right);
141
142 // Rear Wheels: Input Torque (Contains transmission losses) - Tire Reaction - Rolling Resistance
143 state_->wheels_speed.rear_left +=
144 ((state_->wheels_torque.rear_left - (state_->rear_left_forces[0] * R) -
145 (std::abs(state_->rear_left_forces[2]) * sign_rl)) /
146 I) *
147 dt;
148
149 state_->wheels_speed.rear_right +=
150 ((state_->wheels_torque.rear_right - (state_->rear_right_forces[0] * R) -
151 (std::abs(state_->rear_right_forces[2]) * sign_rr)) /
152 I) *
153 dt;
154
155 // Front Wheels: Tire Reaction - Bearing Drag - Rolling Resistance
156 state_->wheels_speed.front_left +=
157 ((-(state_->front_left_forces[0] * R) -
158 (car_parameters_->front_bearing_drag * state_->wheels_speed.front_left) -
159 (std::abs(state_->front_left_forces[2]) * sign_fl)) /
160 I) *
161 dt;
162
163 state_->wheels_speed.front_right +=
164 ((-(state_->front_right_forces[0] * R) -
165 (car_parameters_->front_bearing_drag * state_->wheels_speed.front_right) -
166 (std::abs(state_->front_right_forces[2]) * sign_fr)) /
167 I) *
168 dt;
169 }
170
171 // Vehicle State Update
172 // Sum of all forces normalized to the vehicle coordinate system
173 double Fx_fl = state_->front_left_forces[0] * cos(actual_steering_fl) -
174 state_->front_left_forces[1] * sin(actual_steering_fl);
175 double Fy_fl = state_->front_left_forces[0] * sin(actual_steering_fl) +
176 state_->front_left_forces[1] * cos(actual_steering_fl);
177 double Fx_fr = state_->front_right_forces[0] * cos(actual_steering_fr) -
178 state_->front_right_forces[1] * sin(actual_steering_fr);
179 double Fy_fr = state_->front_right_forces[0] * sin(actual_steering_fr) +
180 state_->front_right_forces[1] * cos(actual_steering_fr);
181 double total_fx =
182 Fx_fl + Fx_fr + state_->rear_left_forces[0] + state_->rear_right_forces[0] + aero_forces[0];
183 double total_fy =
184 Fy_fl + Fy_fr + state_->rear_left_forces[1] + state_->rear_right_forces[1] + aero_forces[1];
185 state_->total_force_x = total_fx;
186 state_->total_force_y = total_fy;
187
188 // Update accelerations
189 state_->ax = total_fx / car_parameters_->total_mass + state_->vy * state_->yaw_rate;
190 state_->ay = total_fy / car_parameters_->total_mass - state_->vx * state_->yaw_rate;
191
192 // Update velocities
193 state_->vx += state_->ax * dt;
194 state_->vy += state_->ay * dt;
195
196 // Prevent oscillations at very low speeds by forcing a dead stop
197 double speed = std::sqrt(state_->vx * state_->vx + state_->vy * state_->vy);
198 if (speed < 0.05 && std::abs(throttle_input) < 0.01) {
199 state_->vx = 0.0;

201 state_->ax = 0.0;
202 state_->ay = 0.0;
203 state_->yaw_rate = 0.0;
204 state_->wheels_speed.front_left = 0.0;
205 state_->wheels_speed.front_right = 0.0;
206 state_->wheels_speed.rear_left = 0.0;
207 state_->wheels_speed.rear_right = 0.0;
208 state_->wheels_slip_ratio = {0.0, 0.0, 0.0, 0.0};
209 state_->wheels_slip_angle = {0.0, 0.0, 0.0, 0.0};
210 }
211
212 double lr = car_parameters_->cg_2_rear_axis; // Distance from CG to rear axle
213 double lf = car_parameters_->wheelbase - lr; // Distance from CG to front axle
214 double half_width = car_parameters_->track_width / 2.0;
215
216 // 1. Moment from Lateral Forces (Fy)
217 double moment_fy =
218 (Fy_fl + Fy_fr) * lf - (state_->rear_left_forces[1] + state_->rear_right_forces[1]) * lr;
219 state_->moment_fy = moment_fy;
220
221 // 2. Moment from Longitudinal Forces (Fx)
222 double moment_fx =
223 (Fx_fr - Fx_fl) * half_width // front axle
224 + (state_->rear_right_forces[0] - state_->rear_left_forces[0]) * half_width; // rear axle
225 state_->moment_fx = moment_fx;
226
227 // 3. Self-Aligning Moments (Mz) from the tires themselves
228 double total_mz = state_->front_left_forces[3] + state_->front_right_forces[3] +
229 state_->rear_left_forces[3] + state_->rear_right_forces[3];
230 state_->self_aligning_moment = total_mz;
231
232 double total_torque = moment_fy + moment_fx + total_mz;
233 state_->total_torque_z = total_torque;
234
235 // Update yaw
236 double yaw_a = total_torque / car_parameters_->Izz;
237 state_->yaw_rate += yaw_a * dt;
238 state_->yaw += state_->yaw_rate * dt;
239
240 // Keep yaw within [-pi, pi]
241 if (state_->yaw > M_PI) {
242 state_->yaw -= 2.0 * M_PI;
243 }
244 if (state_->yaw < -M_PI) {
245 state_->yaw += 2.0 * M_PI;
246 }
247
248 // Update X and Y positions
249 // 1. Calculate Global Velocities
250 double cos_yaw = cos(state_->yaw);
251 double sin_yaw = sin(state_->yaw);
252
253 double v_global_x = state_->vx * cos_yaw - state_->vy * sin_yaw;
254 double v_global_y = state_->vx * sin_yaw + state_->vy * cos_yaw;
255
256 // 2. Update Global Positions (Integration)
257 state_->x += v_global_x * dt;
258 state_->y += v_global_y * dt;
259

261 execution_times_->powertrain_ms =
262 std::chrono::duration<double, std::milli>(powertrain_end - powertrain_start).count();
263 execution_times_->transmission_ms =

267 execution_times_->steering_ms =

269 execution_times_->load_transfer_ms =

320double FSFEUP02Model::calculate_powertrain_torque(double throttle_input, double dt) {
321 double motor_omega = transmission_->calculate_motor_omega(state_->wheels_speed);
322 double motor_rpm = std::abs(motor_omega * 60.0f / (2.0f * M_PI));
323
324 // Calculate Max Torque at current RPM
325 double max_motor_torque = motor_->get_max_torque_at_rpm(motor_rpm);
326 double reference_motor_torque = throttle_input * max_motor_torque;
327
328 // Motor Efficiency at this state
329 double motor_efficiency = motor_->get_efficiency(std::abs(reference_motor_torque), motor_rpm);
330
331 // Preserve current sign so regen charges the battery.
332 double requested_motor_current =
333 reference_motor_torque /
334 (car_parameters_->motor_parameters->kt_constant * std::max(motor_efficiency, 0.05));
335
336 // Calculate the allowed current from the battery
337 double allowed_motor_current = battery_->calculate_allowed_current(requested_motor_current);
338
339 // Actual motor torque limited by the battery
340 double actual_motor_torque =
341 allowed_motor_current * car_parameters_->motor_parameters->kt_constant * motor_efficiency;
342
343 battery_->update_state(allowed_motor_current, dt);
344 motor_->update_state(allowed_motor_current, actual_motor_torque, dt);
345
346 state_->motor_torque = actual_motor_torque;
347 state_->motor_omega = motor_omega;
348 state_->motor_current = motor_->get_current();
349 state_->motor_thermal_state = motor_->get_thermal_state();
350 state_->motor_thermal_capacity = motor_->get_thermal_capacity();
351 state_->battery_current = battery_->get_current();
352 state_->battery_voltage = battery_->get_voltage();
353 state_->battery_soc = battery_->get_soc();
354 state_->battery_open_circuit_voltage = battery_->get_open_circuit_voltage();
355
356 return static_cast<double>(actual_motor_torque);
357}
