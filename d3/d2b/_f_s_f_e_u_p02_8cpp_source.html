28void FSFEUP02Model::step(double dt, common_lib::structures::Wheels throttle, double angle) {
29 // The driven-wheel / tyre-slip mode is stiff (time constant ~ I*denom/(kxk*r^2),
30 // sub-millisecond at low speed), so a single explicit RK4 step at the incoming
31 // dt is numerically unstable near launch and causes runaway wheel spin. The
32 // stable step scales with speed, so we substep adaptively: one step at cruise,
33 // finer steps only in the first low-speed instants. This mirrors the offline
34 // tuning model (inline_02.hpp) so tuned parameters transfer to the live sim.
35 if (!(dt > 0.0)) return;
36 constexpr double kBaseSubDt = 2.0e-4; // stable step at ~1 m/s
37 constexpr int kMaxSub = 64;
38 const double speed = std::sqrt(state_->vx * state_->vx + state_->vy * state_->vy);
39 const double stable_dt = kBaseSubDt * std::max(1.0, speed);
40 const int n_sub = std::min(kMaxSub, std::max(1, static_cast<int>(std::ceil(dt / stable_dt))));
41 const double sub_dt = dt / static_cast<double>(n_sub);
42 for (int k = 0; k < n_sub; ++k) integrate_substep(sub_dt, throttle, angle);
43}

45void FSFEUP02Model::integrate_substep(double dt, common_lib::structures::Wheels throttle, double angle) {
46 using Clock = std::chrono::steady_clock;
47
48 const auto powertrain_start = Clock::now();
49 const double throttle_input =
50 (throttle.rear_left + throttle.rear_right) / 2.0; // Average throttle for rear-wheel drive
51
52 common_lib::structures::Wheels brake_torques;
53 double inverter_command = throttle_input;
54 if (control_mode_ == "manual" && throttle_input < 0.0) {
55 brake_torques = brake_->calculate_brake_torques(-throttle_input);
56 inverter_command = 0.0;
57 }
58

60 brake_torques.rear_left > 0.0 || brake_torques.rear_right > 0.0;
61 const double motor_input = inverter_->calculate_inverter_throttle(inverter_command, dt);
62 const double motor_torque =
63 calculate_powertrain_torque(motor_input, dt);
64 const auto powertrain_end = Clock::now();
65
66 VehicleModelExecutionTimes current_times;
67 current_times.powertrain_ms =
68 std::chrono::duration<double, std::milli>(powertrain_end - powertrain_start).count();
69
70 const auto integration_start = Clock::now();
71 StateVec s = StateVec::Zero();
72 s(VX) = state_->vx;
73 s(VY) = state_->vy;
74 s(YAW_RATE) = state_->yaw_rate;
75 s(YAW) = state_->yaw;
76 s(PX) = state_->x;
77 s(PY) = state_->y;
78 s(ST_ANGLE) = state_->steering_angle;
79 s(FL_W) = state_->wheels_speed.front_left;
80 s(FR_W) = state_->wheels_speed.front_right;
81 s(RL_W) = state_->wheels_speed.rear_left;
82 s(RR_W) = state_->wheels_speed.rear_right;
83 s(AX) = state_->ax;
84 s(AY) = state_->ay;
85
86 const StateVec k1 =
87 get_state_derivative(s, motor_torque, brake_torques, angle, dt, false, &current_times);
88 const StateVec k2 = get_state_derivative(s + 0.5 * dt * k1, motor_torque, brake_torques, angle,
89 dt, false, &current_times);
90 const StateVec k3 = get_state_derivative(s + 0.5 * dt * k2, motor_torque, brake_torques, angle,
91 dt, false, &current_times);
92 const StateVec k4 = get_state_derivative(s + dt * k3, motor_torque, brake_torques, angle, dt,
93 false, &current_times);
94 StateVec s_next = s + (dt / 6.0) * (k1 + 2.0 * k2 + 2.0 * k3 + k4);
95
96 state_->vx = s_next(VX);
97 state_->vy = s_next(VY);
98 state_->yaw_rate = s_next(YAW_RATE);
99 state_->yaw = s_next(YAW);
100 state_->x = s_next(PX);
101 state_->y = s_next(PY);
102 state_->steering_angle = s_next(ST_ANGLE);
103 state_->wheels_speed =
104 common_lib::structures::Wheels(s_next(FL_W), s_next(FR_W), s_next(RL_W), s_next(RR_W));
105 state_->ax = s_next(AX);
106 state_->ay = s_next(AY);
107
108 if (state_->ebs_active) {
109 state_->wheels_speed = {0.0, 0.0, 0.0, 0.0};
110 }
111
112 // Prevent oscillations at very low speeds by forcing a dead stop
113 double speed = std::sqrt(state_->vx * state_->vx + state_->vy * state_->vy);
114 if (speed < 0.05 && (std::abs(throttle_input) < 0.01 || braking)) {
115 state_->vx = 0.0;
116 state_->vy = 0.0;
117 state_->ax = 0.0;
118 state_->ay = 0.0;
119 state_->yaw_rate = 0.0;
120 state_->wheels_speed.front_left = 0.0;
121 state_->wheels_speed.front_right = 0.0;
122 state_->wheels_speed.rear_left = 0.0;
123 state_->wheels_speed.rear_right = 0.0;
124 state_->wheels_slip_ratio = {0.0, 0.0, 0.0, 0.0};
125 state_->wheels_slip_angle = {0.0, 0.0, 0.0, 0.0};
126 }
127
128 // Keep yaw within [-pi, pi]
129 if (state_->yaw > M_PI) {
130 state_->yaw -= 2.0 * M_PI;
131 }
132 if (state_->yaw < -M_PI) {
133 state_->yaw += 2.0 * M_PI;
134 }
135
136 s_next(VX) = state_->vx;
137 s_next(VY) = state_->vy;
138 s_next(YAW_RATE) = state_->yaw_rate;
139 s_next(YAW) = state_->yaw;
140 s_next(PX) = state_->x;
141 s_next(PY) = state_->y;
142 s_next(ST_ANGLE) = state_->steering_angle;
143 s_next(FL_W) = state_->wheels_speed.front_left;
144 s_next(FR_W) = state_->wheels_speed.front_right;
145 s_next(RL_W) = state_->wheels_speed.rear_left;
146 s_next(RR_W) = state_->wheels_speed.rear_right;
147 s_next(AX) = state_->ax;
148 s_next(AY) = state_->ay;
149 get_state_derivative(s_next, motor_torque, brake_torques, angle, dt, true, &current_times);
150
151 const auto integration_end = Clock::now();
152
153 current_times.integration_ms =
154 std::chrono::duration<double, std::milli>(integration_end - integration_start).count();
155 *execution_times_ = current_times;
156}

158double FSFEUP02Model::calculate_powertrain_torque(double throttle_input, double dt) {
159 double motor_omega = transmission_->calculate_motor_omega(state_->wheels_speed);
160 double motor_rpm = std::abs(motor_omega * 60.0f / (2.0f * M_PI));
161
162 // Calculate max motor torque at current RPM
163 double max_motor_torque = motor_->get_max_torque_at_rpm(motor_rpm);
164 double reference_motor_torque = throttle_input * max_motor_torque;
165
166 // Avoid zero power request at launch
167 const double min_omega_for_power = 10.0; // rad/s,
168
169 double omega_sign_source =
170 std::abs(motor_omega) > 1e-3 ? motor_omega : reference_motor_torque;
171
172 double omega_for_power =
173 std::copysign(std::max(std::abs(motor_omega), min_omega_for_power),
174 omega_sign_source);
175
176 // Mechanical power request
177 double mechanical_power_request =
178 reference_motor_torque * omega_for_power;
179
180 // Mechanical-Eletrical efficiency
181 double motor_efficiency = motor_->get_efficiency(std::abs(reference_motor_torque), motor_rpm);
182 double inverter_efficiency = inverter_->get_efficiency();
183 double total_efficiency = motor_efficiency * inverter_efficiency;
184
185 // Requested battery power
186 double battery_power_request;
187 if (mechanical_power_request >= 0.0) {
188 // Normal
189 battery_power_request = mechanical_power_request / total_efficiency;
190 } else {
191 // Regen
192 battery_power_request = mechanical_power_request * total_efficiency;
193 }
194
195 // Battery current limit
196 double battery_voltage = battery_->get_voltage();
197 double requested_battery_current =
198 battery_power_request / battery_voltage;
199 double allowed_battery_current =
200 battery_->calculate_allowed_current(requested_battery_current);
201
202 // Provided battery power
203 double allowed_battery_voltage =
204 battery_->get_voltage(allowed_battery_current);
205 double allowed_battery_power =
206 allowed_battery_current * allowed_battery_voltage;
207
208 // Convert back to available mechanical power
209 double allowed_mechanical_power;
210 if (allowed_battery_power >= 0.0) {
211 allowed_mechanical_power = allowed_battery_power * total_efficiency;
212 } else {
213 allowed_mechanical_power = allowed_battery_power / total_efficiency;
214 }
215
216 double actual_motor_torque = std::copysign(
217 std::min(std::abs(allowed_mechanical_power / omega_for_power),
218 std::abs(reference_motor_torque)),
219 reference_motor_torque);
220
221 // Motor phase current magnitude. Sign follows the battery current convention:
222 // positive while discharging, negative while charging/regenerating.
223 double motor_phase_current =
224 std::abs(actual_motor_torque) / car_parameters_->motor_parameters->kt_constant;
225
226 // Inverter phase current limit
227 double max_inverter_phase_current =
228 car_parameters_->inverter_parameters->max_phase_current;
229 if (motor_phase_current > max_inverter_phase_current) {
230 double limited_torque =
231 max_inverter_phase_current *
232 car_parameters_->motor_parameters->kt_constant;
233
234 actual_motor_torque =
235 std::copysign(limited_torque, actual_motor_torque);
236
237 motor_phase_current = max_inverter_phase_current;
238 }
239
240 const double signed_motor_phase_current =
241 std::abs(allowed_battery_current) > 1e-9
242 ? std::copysign(motor_phase_current, allowed_battery_current)
243 : 0.0;
244
245 // Update models
246 battery_->update_state(allowed_battery_current, dt);
247 motor_->update_state(signed_motor_phase_current,
248 actual_motor_torque,
249 dt);
250
251 state_->motor_torque = actual_motor_torque;
252 state_->motor_omega = motor_omega;
253 state_->motor_current = motor_->get_current();
254 state_->motor_thermal_state = motor_->get_thermal_state();
255 state_->motor_thermal_capacity = motor_->get_thermal_capacity();
256 state_->battery_current = battery_->get_current();
257 state_->battery_voltage = battery_->get_voltage();
258 state_->battery_soc = battery_->get_soc();
259 state_->battery_open_circuit_voltage = battery_->get_open_circuit_voltage();
260
261 return static_cast<double>(actual_motor_torque);
262}

264FSFEUP02Model::StateVec FSFEUP02Model::get_state_derivative(
265 const StateVec& s, double motor_torque, const common_lib::structures::Wheels& brake_torques,
266 double steering_target, double dt, bool write_telemetry,
267 VehicleModelExecutionTimes* execution_times) {
268 using Clock = std::chrono::steady_clock;
269 StateVec ds = StateVec::Zero();
270
271 const common_lib::structures::Wheels wheel_speeds(s(FL_W), s(FR_W), s(RL_W), s(RR_W));
272
273 const auto transmission_start = Clock::now();
274 const common_lib::structures::Wheels wheel_torques =
275 transmission_->calculate_wheel_torques(motor_torque, wheel_speeds);
276 const auto transmission_end = Clock::now();
277 execution_times->transmission_ms +=
278 std::chrono::duration<double, std::milli>(transmission_end - transmission_start).count();
279 const Eigen::Vector4d torques(wheel_torques.front_left, wheel_torques.front_right,
280 wheel_torques.rear_left, wheel_torques.rear_right);
281
282 const auto steering_start = Clock::now();
283 Eigen::Vector4d wheel_angles = steering_->calculate_steering_angles(s(ST_ANGLE));
284 const auto steering_angles_end = Clock::now();
285 execution_times->steering_ms +=
286 std::chrono::duration<double, std::milli>(steering_angles_end - steering_start).count();
287
288 const auto aero_start = Clock::now();
289 const Eigen::Vector3d aero_forces =
290 aero_->aero_forces(Eigen::Vector3d(s(VX), s(VY), s(YAW_RATE)));
291 const auto aero_end = Clock::now();
292 execution_times->aero_ms +=
293 std::chrono::duration<double, std::milli>(aero_end - aero_start).count();
294
295 const auto load_transfer_start = Clock::now();
296 const common_lib::structures::Wheels load_distribution = load_transfer_->compute_loads(
297 LoadTransferInput{s(AX), s(AY), aero_forces[2]}); // aero_forces{Fx,Fy,Fz}
298 const auto load_transfer_end = Clock::now();
299 execution_times->load_transfer_ms +=
300 std::chrono::duration<double, std::milli>(load_transfer_end - load_transfer_start).count();
301 const Eigen::Vector4d vertical_loads(load_distribution.front_left, load_distribution.front_right,
302 load_distribution.rear_left, load_distribution.rear_right);
303
304 const auto tire_start = Clock::now();
305 Eigen::VectorXd tire_forces(16);
306 Eigen::Vector4d slip_ratio = Eigen::Vector4d::Zero();
307 Eigen::Vector4d slip_angle = Eigen::Vector4d::Zero();
308 Eigen::Vector4d contact_patch_longitudinal_velocity = Eigen::Vector4d::Zero();
309 TireInput tire_input;
310 tire_input.dt = dt;
311 tire_input.vx = s(VX);
312 tire_input.vy = s(VY);
313 tire_input.yaw_rate = s(YAW_RATE);
314 tire_input.last_slip_ratio =
315 Eigen::Vector4d(state_->wheels_slip_ratio.front_left, state_->wheels_slip_ratio.front_right,
316 state_->wheels_slip_ratio.rear_left, state_->wheels_slip_ratio.rear_right);
317 tire_input.last_slip_angle =
318 Eigen::Vector4d(state_->wheels_slip_angle.front_left, state_->wheels_slip_angle.front_right,
319 state_->wheels_slip_angle.rear_left, state_->wheels_slip_angle.rear_right);
320
321 for (Tire tire : {FL, FR, RL, RR}) {
322 tire_input.tire = tire;
323 tire_input.steering_angle = wheel_angles(tire);
324 tire_input.wheel_angular_speed = s(FL_W + tire);
325 tire_input.vertical_load = vertical_loads(tire);
326 tire_forces.segment<4>(tire * 4) = tire_model_->calculate_tire_forces(tire_input);
327 slip_ratio(tire) = tire_input.slip_ratio;
328 slip_angle(tire) = tire_input.slip_angle;
329 contact_patch_longitudinal_velocity(tire) = tire_input.vcx;
330 }
331 const auto tire_end = Clock::now();
332 execution_times->tire_ms +=
333 std::chrono::duration<double, std::milli>(tire_end - tire_start).count();
334
335 double total_fx = aero_forces[0];
336 double total_fy = aero_forces[1];
337 double moment_fy = 0.0;
338 double moment_fx = 0.0;
339 double self_aligning_moment = 0.0;
340 double total_torque = 0.0;
341
342 const double lr = car_parameters_->cg_2_rear_axis;
343 const double lf = car_parameters_->wheelbase - lr;
344 const double half_width = car_parameters_->track_width / 2.0;
345 const double wheel_radius = car_parameters_->tire_parameters->effective_tire_r;
346 const Eigen::Vector4d brake_torques_by_tire(brake_torques.front_left,
347 brake_torques.front_right,
348 brake_torques.rear_left,
349 brake_torques.rear_right);
350
351 for (Tire tire : {FL, FR, RL, RR}) {
352 const double fx_tire = tire_forces(tire * 4);
353 const double fy_tire = tire_forces(tire * 4 + 1);
354 const double mz_tire = tire_forces(tire * 4 + 3);
355 const double cos_delta = std::cos(wheel_angles(tire));
356 const double sin_delta = std::sin(wheel_angles(tire));
357
358 const double fx_vehicle = fx_tire * cos_delta - fy_tire * sin_delta;
359 const double fy_vehicle = fx_tire * sin_delta + fy_tire * cos_delta;
360 const double arm_x = (tire == FL || tire == FR) ? lf : -lr;
361 const double arm_y = (tire == FL || tire == RL) ? half_width : -half_width;
362
363 total_fx += fx_vehicle;
364 total_fy += fy_vehicle;
365 moment_fy += arm_x * fy_vehicle;
366 moment_fx += -arm_y * fx_vehicle;
367 self_aligning_moment += mz_tire;
368 total_torque += arm_x * fy_vehicle - arm_y * fx_vehicle + mz_tire;
369 }
370
371 const double ax = total_fx / car_parameters_->total_mass + s(VY) * s(YAW_RATE);
372 const double ay = total_fy / car_parameters_->total_mass - s(VX) * s(YAW_RATE);
373 ds(VX) = ax;
374 ds(VY) = ay;
375 ds(AX) = ax - s(AX);
376 ds(AY) = ay - s(AY);
377 ds(YAW_RATE) = total_torque / car_parameters_->Izz;
378 ds(YAW) = s(YAW_RATE);
379 ds(PX) = s(VX) * std::cos(s(YAW)) - s(VY) * std::sin(s(YAW));
380 ds(PY) = s(VX) * std::sin(s(YAW)) + s(VY) * std::cos(s(YAW));
381 const auto steering_motor_start = Clock::now();
382 ds(ST_ANGLE) = steering_motor_->compute_steering_rate(s(ST_ANGLE), steering_target);
383 const auto steering_motor_end = Clock::now();
384 execution_times->steering_ms +=
385 std::chrono::duration<double, std::milli>(steering_motor_end - steering_motor_start).count();
386
387 if (state_->ebs_active) {
388 ds(FL_W) = -s(FL_W) / std::max(dt, 1e-6);
389 ds(FR_W) = -s(FR_W) / std::max(dt, 1e-6);
390 ds(RL_W) = -s(RL_W) / std::max(dt, 1e-6);
391 ds(RR_W) = -s(RR_W) / std::max(dt, 1e-6);
392 } else {
393 // Front: tire + rim inertia only; Rear: tire + rim + motor + transmission (reflected)
394 const double front_inertia = car_parameters_->front_wheel_inertia;
395 const double rear_inertia = car_parameters_->rear_wheel_inertia;
396 for (Tire tire : {FL, FR, RL, RR}) {
397 const double wheel_omega = s(FL_W + tire);
398 double net_torque =
399 torques(tire) - tire_forces(tire * 4) * wheel_radius - tire_forces(tire * 4 + 2);
400 const double brake_sign = 2.0 / M_PI * std::atan(10.0 * wheel_omega);
401 const double brake_torque = brake_torques_by_tire(tire);
402 net_torque -= brake_torque * brake_sign;
403
404 if (tire == FL || tire == FR) {
405 net_torque -= car_parameters_->front_bearing_drag * wheel_omega;
406 }
407
408 const double inertia = (tire == FL || tire == FR) ? front_inertia : rear_inertia;
409 const double wheel_acceleration = net_torque / inertia;
410 if (brake_torque > 0.0 && wheel_omega * (wheel_omega + wheel_acceleration * dt) <= 0.0) {
411 ds(FL_W + tire) = -wheel_omega / std::max(dt, 1e-6);
412 } else {
413 ds(FL_W + tire) = wheel_acceleration;
414 }
415 }
416 }
417
418 if (write_telemetry) {
419 state_->front_left_forces = tire_forces.segment<4>(FL * 4);
420 state_->front_right_forces = tire_forces.segment<4>(FR * 4);
421 state_->rear_left_forces = tire_forces.segment<4>(RL * 4);
422 state_->rear_right_forces = tire_forces.segment<4>(RR * 4);
423 state_->wheels_torque = wheel_torques;
424 state_->wheels_vertical_load = load_distribution;
425 state_->wheels_slip_ratio = common_lib::structures::Wheels(slip_ratio(FL), slip_ratio(FR),
426 slip_ratio(RL), slip_ratio(RR));
427 state_->wheels_slip_angle = common_lib::structures::Wheels(slip_angle(FL), slip_angle(FR),
428 slip_angle(RL), slip_angle(RR));
429 state_->aero_drag = aero_forces[0];
430 state_->aero_downforce = aero_forces[2];
431 state_->total_force_x = total_fx;
432 state_->total_force_y = total_fy;
433 state_->moment_fy = moment_fy;
434 state_->moment_fx = moment_fx;
435 state_->self_aligning_moment = self_aligning_moment;
436 state_->total_torque_z = total_torque;
437 }
438
439 return ds;
440}
