23TuningEvaluator::PoseSample TuningEvaluator::transform_pose_to_map(
24 const PoseSample& pose, const PoseSample& source_origin, const PoseSample& target_origin) {
25 const double dx = pose.x - source_origin.x;
26 const double dy = pose.y - source_origin.y;
27 const double source_cos = std::cos(source_origin.yaw);
28 const double source_sin = std::sin(source_origin.yaw);
29 const double local_x = source_cos * dx + source_sin * dy;
30 const double local_y = -source_sin * dx + source_cos * dy;
31 const double local_yaw = normalize_angle(pose.yaw - source_origin.yaw);
32
33 const double target_cos = std::cos(target_origin.yaw);
34 const double target_sin = std::sin(target_origin.yaw);
35 return {target_origin.x + target_cos * local_x - target_sin * local_y,
36 target_origin.y + target_sin * local_x + target_cos * local_y,
37 normalize_angle(target_origin.yaw + local_yaw)};
38}

179void TuningEvaluator::on_timer() {
180 if (!started_ || !has_real_pose_) {
181 return;
182 }
183
184 const VehicleStateSnapshot sim_state = simulator_->get_vehicle_state_snapshot();
185 const SensorsSnapshot sim_sensors = simulator_->get_sensors_snapshot();
186 const PoseSample real_raw = {latest_real_pose_.x, latest_real_pose_.y, latest_real_pose_.theta};
187 const PoseSample real_pose = transform_pose_to_map(real_raw, real_origin_, sim_origin_);
188 const PoseSample sim_pose = {sim_state.position.x, sim_state.position.y, sim_state.yaw};
189
190 const double position_error = std::hypot(sim_pose.x - real_pose.x, sim_pose.y - real_pose.y);
191 const double heading_error = normalize_angle(sim_pose.yaw - real_pose.yaw);
192 publish_float(position_error_pub_, position_error);
193 publish_float(position_rmse_pub_, position_rmse_.update(position_error));
194 publish_float(heading_error_pub_, heading_error);

196
197 if (has_real_velocity_) {
198 const VelocitySample real_velocity = {latest_real_velocity_.velocity_x,
199 latest_real_velocity_.velocity_y,
200 latest_real_velocity_.angular_velocity};
201 const VelocitySample sim_velocity = {sim_state.velocity_x, sim_state.velocity_y,
202 sim_state.yaw_rate};
203 const double velocity_error =
204 std::hypot(sim_velocity.velocity_x - real_velocity.velocity_x,
205 sim_velocity.velocity_y - real_velocity.velocity_y);
206 const double yaw_rate_error = sim_velocity.yaw_rate - real_velocity.yaw_rate;
207 publish_float(velocity_error_pub_, velocity_error);
208 publish_float(velocity_rmse_pub_, velocity_rmse_.update(velocity_error));
209 publish_float(yaw_rate_error_pub_, yaw_rate_error);
210 publish_float(yaw_rate_rmse_pub_, yaw_rate_rmse_.update(yaw_rate_error));
211 }
212
213 if (has_real_fl_rpm_ && has_real_fr_rpm_) {
214 const double real_front_rpm = 0.5 * (latest_real_fl_rpm_ + latest_real_fr_rpm_);
215 const double sim_front_rpm =
216 0.5 * (sim_sensors.wheel_rpm.front_left + sim_sensors.wheel_rpm.front_right);
217 const double front_wheel_rpm_error = sim_front_rpm - real_front_rpm;
218 publish_float(front_wheel_rpm_error_pub_, front_wheel_rpm_error);
219 publish_float(front_wheel_rpm_rmse_pub_, front_wheel_rpm_rmse_.update(front_wheel_rpm_error));
220 }
221
222 if (has_real_motor_rpm_) {
223 const double motor_rpm_error = sim_sensors.motor_rpm - latest_real_motor_rpm_;
224 publish_float(motor_rpm_error_pub_, motor_rpm_error);
225 publish_float(motor_rpm_rmse_pub_, motor_rpm_rmse_.update(motor_rpm_error));
226 }
227
228 publish_pose(real_pose, true);
229 publish_pose(sim_pose, false);
230 publish_comparison_visualization(real_pose, sim_pose, position_error);
231 write_csv_row(real_pose, sim_pose, position_error, heading_error);
232}
