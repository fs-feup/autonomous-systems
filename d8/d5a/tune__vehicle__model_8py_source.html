190 def calculate_metrics(metric_rows, min_samples, min_usable_samples=0):
191 if len(metric_rows) < int(min_samples):
192 if len(metric_rows) < int(min_usable_samples):
193 return {
194 "score": float("inf"),
195 "samples": len(metric_rows),
196 "error": f"not enough samples in {csv_path}",
197 }
198
199 position_rmse = rmse([as_float(row, "position_error") for row in metric_rows])
200 heading_rmse = rmse([as_float(row, "heading_error") for row in metric_rows])
201
202 velocity_errors = []
203 velocity_x_errors = []
204 velocity_x_slow_errors = []
205 velocity_y_errors = []
206 yaw_rate_errors = []
207 yaw_rate_under_errors = []
208 for row in metric_rows:
209 rvx = as_float(row, "real_velocity_x")
210 rvy = as_float(row, "real_velocity_y")
211 svx = as_float(row, "sim_velocity_x")
212 svy = as_float(row, "sim_velocity_y")
213 ryr = as_float(row, "real_yaw_rate")
214 syr = as_float(row, "sim_yaw_rate")
215 if None not in (rvx, rvy, svx, svy):
216 velocity_x_error = svx - rvx
217 velocity_errors.append(math.hypot(velocity_x_error, svy - rvy))
218 velocity_x_errors.append(velocity_x_error)
219 velocity_x_slow_errors.append(min(0.0, velocity_x_error))
220 velocity_y_errors.append(svy - rvy)
221 if None not in (ryr, syr):
222 yaw_rate_error = syr - ryr
223 yaw_rate_errors.append(yaw_rate_error)
224 yaw_rate_under_errors.append(min(0.0, yaw_rate_error))
225
226 metrics = {
227 "position_rmse": position_rmse,
228 "heading_rmse": heading_rmse,
229 "velocity_rmse": rmse(velocity_errors),
230 "velocity_x_rmse": rmse(velocity_x_errors),
231 "velocity_x_mean_error": mean(velocity_x_errors),
232 "velocity_x_slow_rmse": rmse(velocity_x_slow_errors),
233 "velocity_x_slow_mean": -mean(velocity_x_slow_errors)
234 if mean(velocity_x_slow_errors) is not None
235 else None,
236 "velocity_y_rmse": rmse(velocity_y_errors),
237 "yaw_rate_rmse": rmse(yaw_rate_errors),
238 "yaw_rate_under_rmse": rmse(yaw_rate_under_errors),
239 "front_wheel_rpm_rmse": rmse(
240 [as_float(row, "front_wheel_rpm_error") for row in metric_rows]
241 ),
242 "motor_rpm_rmse": rmse([as_float(row, "motor_rpm_error") for row in metric_rows]),
243 "samples": len(metric_rows),
244 }
245 return metrics
246
247 metrics = calculate_metrics(
248 rows,
249 score_config.get("min_samples", 0),
250 score_config.get("min_usable_samples", score_config.get("min_samples", 0)),
251 )
252 if metrics.get("score") == float("inf"):
253 return metrics
254
255 weights = score_config.get("weights", {})
256 score = 0.0
257 for key, weight in weights.items():
258 value = metrics.get(key)
259 if value is None:
260 return {"score": float("inf"), **metrics, "error": f"missing {key}"}
261 score += float(weight) * value
262
263 min_samples = int(score_config.get("min_samples", 0))
264 if min_samples > 0 and metrics["samples"] < min_samples:
265 shortfall = (min_samples - metrics["samples"]) / min_samples
266 score += float(score_config.get("short_sample_penalty", 0.0)) * shortfall
267 metrics["warning"] = f"used {metrics['samples']} samples below target {min_samples}"
268
269 limit_warnings = []
270 for key, limit in score_config.get("limits", {}).items():
271 value = metrics.get(key)
272 if value is None:
273 return {"score": float("inf"), **metrics, "error": f"missing limit metric {key}"}
274 maximum = float(limit.get("max", float("inf")))
275 if value > maximum:
276 excess = value - maximum
277 score += float(limit.get("penalty", 0.0)) * excess
278 limit_warnings.append(f"{key} {value:.4g} > {maximum:.4g}")
279 if limit_warnings:
280 previous_warning = metrics.get("warning")
281 metrics["warning"] = (
282 f"{previous_warning}; " if previous_warning else ""
283 ) + "; ".join(limit_warnings)
284
285 metrics["score"] = score
286 return metrics

301def run_bag_trial(config, candidate_dir, bag, trial_name, score_config=None):
302 playback = config["tuning"]["playback"]
303 candidate_dir.mkdir(parents=True, exist_ok=True)
304
305 csv_dir = candidate_dir / "evaluator_csv" / trial_name
306 csv_dir.mkdir(parents=True, exist_ok=True)
307
308 sim_process = None
309 bag_process = None
310 original_global_config = None
311 try:
312 original_global_config = apply_invictasim_runtime_config(bag, playback)
313 if bag.get("track_name"):
314 append_status(candidate_dir, f"{trial_name}: using track {bag['track_name']}")
315 if playback.get("sim_speed") is not None:
316 append_status(candidate_dir, f"{trial_name}: using sim_speed {playback['sim_speed']}")
317 if float(playback.get("rate", playback["sim_speed"])) != float(playback["sim_speed"]):
318 append_status(
319 candidate_dir,
320 f"{trial_name}: WARNING bag rate {playback.get('rate')} differs from sim_speed "
321 f"{playback['sim_speed']}",
322 )
323
324 track_label = f" with track {bag['track_name']}" if bag.get("track_name") else ""
325 print(f"[{trial_name}] starting InvictaSim{track_label}", flush=True)
326 append_status(candidate_dir, f"{trial_name}: starting InvictaSim{track_label}")
327 sim_cmd = (
328 "ros2 run invictasim invictasim --ros-args "
329 f"-p tuning_output_directory:={shlex.quote(str(csv_dir))} "
330 "-p tuning_output_samples:=true"
331 )
332 sim_process = start_process(sim_cmd)
333 append_status(candidate_dir, f"{trial_name}: InvictaSim pid {sim_process.pid}")
334 time.sleep(float(playback.get("startup_sleep_s", 3.0)))
335 if sim_process.poll() is not None:
336 append_status(candidate_dir, f"{trial_name}: InvictaSim exited early")
337 return {
338 "score": float("inf"),
339 "error": f"InvictaSim exited early with code {sim_process.returncode}",
340 }
341
342 topics = " ".join(shlex.quote(topic) for topic in playback.get("topics", []))
343 playback_rate = float(playback.get("rate", playback.get("sim_speed", 1.0)))
344 bag_cmd = (
345 f"ros2 bag play {shlex.quote(str(bag['path']))} "
346 f"--rate {playback_rate} "
347 "--disable-keyboard-controls"
348 )
349 if topics:
350 bag_cmd += f" --topics {topics}"
351 if float(bag.get("start_offset_s", 0.0)) > 0.0:
352 bag_cmd += f" --start-offset {float(bag['start_offset_s'])}"
353
354 print(f"[{trial_name}] playing bag: {bag['path']}", flush=True)
355 append_status(candidate_dir, f"{trial_name}: playing bag {bag['path']}")
356 bag_process = subprocess.Popen(
357 ["bash", "-lc", ros_command(bag_cmd)],
358 cwd=WORKSPACE,
359 stdout=None,
360 stderr=None,
361 preexec_fn=os.setsid,
362 )
363 append_status(candidate_dir, f"{trial_name}: bag pid {bag_process.pid}")
364 bag_timeout = bag.get("max_wall_time_s")
365 if bag.get("stop_after_bag_s") is not None:
366 bag_timeout = float(bag["stop_after_bag_s"]) / playback_rate
367 try:
368 bag_returncode = bag_process.wait(
369 timeout=float(bag_timeout) if bag_timeout else None
370 )
371 except subprocess.TimeoutExpired:
372 append_status(candidate_dir, f"{trial_name}: bag timeout after {bag_timeout}s")
373 stop_process(bag_process, float(playback.get("shutdown_grace_s", 3.0)))
374 bag_returncode = 124
375
376 append_status(candidate_dir, f"{trial_name}: bag exit code {bag_returncode}")
377 if bag_returncode not in (0, 124):
378 return {
379 "score": float("inf"),
380 "error": f"bag playback failed with exit code {bag_returncode}",
381 }
382 time.sleep(0.5)
383 finally:
384 grace = float(playback.get("shutdown_grace_s", 3.0))
385 if bag_process is not None:
386 stop_process(bag_process, grace)
387 if sim_process is not None:
388 stop_process(sim_process, grace)
389 if original_global_config is not None:
390 restore_files(original_global_config)
391
392 csv_path = newest_csv(csv_dir)
393 if csv_path is None:
394 append_status(candidate_dir, f"{trial_name}: evaluator did not produce csv")
395 remove_path_if_empty(csv_dir)
396 return {"score": float("inf"), "error": "evaluator did not produce csv"}
397 result = score_csv(csv_path, score_config or bag.get("score", config["tuning"]["score"]))
398 if bag.get("track_name"):
399 result["track_name"] = str(bag["track_name"])
400 try:
401 csv_path.unlink()
402 remove_path_if_empty(csv_path.parent)
403 except OSError as exc:
404 append_status(candidate_dir, f"{trial_name}: could not remove evaluator csv: {exc}")
405 append_status(candidate_dir, f"{trial_name}: score {result['score']}")
406 return result

409def run_candidate(
410 config, parameters, values, run_dir, candidate_index, original_text, bags=None,
411 score_config=None
412):
413 apply_values(parameters, values)
414 candidate_dir = run_dir / "tries" / f"try_{candidate_index:04d}"
415 append_status(candidate_dir, "starting try")
416 append_status(
417 candidate_dir,
418 "parameters " + json.dumps(
419 {spec["name"]: value for spec, value in zip(parameters, values)},
420 sort_keys=True,
421 ),
422 )
423 bag_results = []
424 weighted_score = 0.0
425 total_weight = 0.0
426 try:
427 selected_bags = bags if bags is not None else config["tuning"]["bags"]
428 for bag_index, bag in enumerate(selected_bags):
429 bag_score_config = resolve_score_config(
430 score_config, bag, config["tuning"]["score"]
431 )
432 result = run_bag_trial(
433 config, candidate_dir, bag, f"bag_{bag_index:02d}", bag_score_config
434 )
435 bag_results.append(result)
436 weight = float(bag.get("weight", 1.0))
437 weighted_score += weight * result["score"]
438 total_weight += weight
439 score = weighted_score / max(total_weight, 1e-9)
440 status = "ok" if math.isfinite(score) else "failed"
441 result = {
442 "try": candidate_index,
443 "status": status,
444 "score": score,
445 "values": {spec["name"]: value for spec, value in zip(parameters, values)},
446 "bags": bag_results,
447 }
448 append_status(candidate_dir, f"finished try with status {status} and score {score}")
449 return result
450 finally:
451 restore_files(original_text)

554def mutate_individual(parameters, values, active_indices, rng, mutation_rate, mutation_scale):
555 mutated = list(values)
556 for index in active_indices:
557 if rng.random() > mutation_rate:
558 continue
559 spec = parameters[index]
560 low = float(spec["min"])
561 high = float(spec["max"])
562 span = high - low
563 mutated[index] = clamp(mutated[index] + rng.gauss(0.0, mutation_scale * span), low, high)
564 return mutated

567def crossover_individuals(parameters, parent_a, parent_b, active_indices, rng, blend_alpha):
568 child = list(parent_a)
569 for index in active_indices:
570 spec = parameters[index]
571 low = float(spec["min"])
572 high = float(spec["max"])
573 alpha = rng.uniform(-blend_alpha, 1.0 + blend_alpha)
574 child[index] = clamp(alpha * parent_a[index] + (1.0 - alpha) * parent_b[index], low, high)
575 return child
