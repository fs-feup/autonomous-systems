67 inline double calculate_powertrain_torque(double throttle_input, double dt, SimState& state) {
68 double motor_omega = transmission_->calculate_motor_omega(state.wheels_speed);
69 double motor_rpm = std::abs(motor_omega * 60.0 / (2.0 * M_PI));
70
71 double max_motor_torque = motor_->get_max_torque_at_rpm(motor_rpm);
72 double reference_motor_torque = throttle_input * max_motor_torque;
73
74 const double min_omega_for_power = 10.0;
75 double omega_sign_source = std::abs(motor_omega) > 1e-3 ? motor_omega : reference_motor_torque;
76 double omega_for_power = std::copysign(std::max(std::abs(motor_omega), min_omega_for_power), omega_sign_source);
77
78 double mechanical_power_request = reference_motor_torque * omega_for_power;
79 double motor_efficiency = motor_->get_efficiency(std::abs(reference_motor_torque), motor_rpm);
80 double inverter_efficiency = inverter_->get_efficiency();
81 double total_efficiency = motor_efficiency * inverter_efficiency;
82
83 double battery_power_request = (mechanical_power_request >= 0.0)
84 ? (mechanical_power_request / total_efficiency)
85 : (mechanical_power_request * total_efficiency);
86
87 double battery_voltage = battery_->get_voltage();
88 double requested_battery_current = battery_power_request / battery_voltage;
89 double allowed_battery_current = battery_->calculate_allowed_current(requested_battery_current);
90
91 double allowed_battery_voltage = battery_->get_voltage(allowed_battery_current);
92 double allowed_battery_power = allowed_battery_current * allowed_battery_voltage;
93
94 double allowed_mechanical_power = (allowed_battery_power >= 0.0)
95 ? (allowed_battery_power * total_efficiency)
96 : (allowed_battery_power / total_efficiency);
97
98 double actual_motor_torque = std::copysign(
99 std::min(std::abs(allowed_mechanical_power / omega_for_power), std::abs(reference_motor_torque)),
100 reference_motor_torque);
101
102 double motor_phase_current = std::abs(actual_motor_torque) / car_parameters_->motor_parameters->kt_constant;
103 double max_inverter_phase_current = car_parameters_->inverter_parameters->max_phase_current;
104
105 if (motor_phase_current > max_inverter_phase_current) {
106 actual_motor_torque = std::copysign(max_inverter_phase_current * car_parameters_->motor_parameters->kt_constant, actual_motor_torque);
107 motor_phase_current = max_inverter_phase_current;
108 }
109
110 const double signed_motor_phase_current = std::abs(allowed_battery_current) > 1e-9
111 ? std::copysign(motor_phase_current, allowed_battery_current) : 0.0;
112
113 battery_->update_state(allowed_battery_current, dt);
114 motor_->update_state(signed_motor_phase_current, actual_motor_torque, dt);
115
116 state.motor_torque = actual_motor_torque;
117 state.motor_omega = motor_omega;
118 state.motor_current = motor_->get_current();
119 state.motor_thermal_state = motor_->get_thermal_state();
120 state.motor_thermal_capacity = motor_->get_thermal_capacity();
121 state.battery_current = battery_->get_current();
122 state.battery_voltage = battery_->get_voltage();
123 state.battery_soc = battery_->get_soc();
124 state.battery_open_circuit_voltage = battery_->get_open_circuit_voltage();
125
126 return actual_motor_torque;
127 }

132 double steering_target, double dt, SimState& state,
133 Eigen::Vector4d& out_slip_ratio, Eigen::Vector4d& out_slip_angle, StateVec& ds) {
134
135 ds.setZero();
136 const common_lib::structures::Wheels wheel_speeds(s(FL_W), s(FR_W), s(RL_W), s(RR_W));
137 const common_lib::structures::Wheels wheel_torques = transmission_->calculate_wheel_torques(motor_torque, wheel_speeds);
138 const Eigen::Vector4d torques(wheel_torques.front_left, wheel_torques.front_right, wheel_torques.rear_left, wheel_torques.rear_right);
139
140 Eigen::Vector4d wheel_angles = steering_->calculate_steering_angles(s(ST_ANGLE));
141 const Eigen::Vector3d aero_forces = aero_->aero_forces(Eigen::Vector3d(s(VX), s(VY), s(YAW_RATE)));
142 const common_lib::structures::Wheels load_distribution = load_transfer_->compute_loads(LoadTransferInput{s(AX), s(AY), aero_forces[2]});
143 const Eigen::Vector4d vertical_loads(load_distribution.front_left, load_distribution.front_right, load_distribution.rear_left, load_distribution.rear_right);
144
145 // STACK ALLOCATION (Fast): Replaced dynamic VectorXd(16) with fixed-size array
146 Eigen::Matrix<double, 16, 1> tire_forces;
147 Eigen::Vector4d contact_patch_longitudinal_velocity = Eigen::Vector4d::Zero();
148
149 TireInput tire_input;
150 tire_input.dt = dt;
151 tire_input.vx = s(VX);
152 tire_input.vy = s(VY);
153 tire_input.yaw_rate = s(YAW_RATE);
154 tire_input.last_slip_ratio << state.wheels_slip_ratio.front_left, state.wheels_slip_ratio.front_right, state.wheels_slip_ratio.rear_left, state.wheels_slip_ratio.rear_right;

156
157 for (Tire tire : {FL, FR, RL, RR}) {
158 tire_input.tire = tire;
159 tire_input.steering_angle = wheel_angles(tire);
160 tire_input.wheel_angular_speed = s(FL_W + tire);
161 tire_input.vertical_load = vertical_loads(tire);
162 tire_forces.segment<4>(tire * 4) = tire_model_->calculate_tire_forces(tire_input);
163 out_slip_ratio(tire) = tire_input.slip_ratio;
164 out_slip_angle(tire) = tire_input.slip_angle;
165 contact_patch_longitudinal_velocity(tire) = tire_input.vcx;
166 }
167
168 double total_fx = aero_forces[0], total_fy = aero_forces[1], total_torque = 0.0;
169 const double lr = car_parameters_->cg_2_rear_axis;
170 const double lf = car_parameters_->wheelbase - lr;
171 const double wheel_radius = car_parameters_->tire_parameters->effective_tire_r;
172 const Eigen::Vector4d brake_torques_by_tire(brake_torques.front_left, brake_torques.front_right, brake_torques.rear_left, brake_torques.rear_right);
173
174 for (Tire tire : {FL, FR, RL, RR}) {
175 double fx_tire = tire_forces(tire * 4);
176 const double fy_tire = tire_forces(tire * 4 + 1);
177 const double mz_tire = tire_forces(tire * 4 + 3);
178 const double cos_delta = std::cos(wheel_angles(tire));
179 const double sin_delta = std::sin(wheel_angles(tire));
180 const double brake_torque = brake_torques_by_tire(tire);
181
182 if (brake_torque > 0.0) {
183 const double brake_sign = 2.0 / M_PI * std::atan(10.0 * contact_patch_longitudinal_velocity(tire));
184 fx_tire -= brake_torque * brake_sign / std::max(wheel_radius, 1e-6);
185 }
186
187 const double fx_vehicle = fx_tire * cos_delta - fy_tire * sin_delta;
188 const double fy_vehicle = fx_tire * sin_delta + fy_tire * cos_delta;
189 const double arm_x = (tire == FL || tire == FR) ? lf : -lr;
190 const double arm_y = (tire == FL || tire == RL) ? half_track_width_ : -half_track_width_;
191
192 total_fx += fx_vehicle;
193 total_fy += fy_vehicle;
194 total_torque += arm_x * fy_vehicle - arm_y * fx_vehicle + mz_tire;
195 }
196
197 // Fast multiplication instead of division
198 const double ax = total_fx * inv_mass_ + s(VY) * s(YAW_RATE);
199 const double ay = total_fy * inv_mass_ - s(VX) * s(YAW_RATE);
200
201 ds(VX) = ax; ds(VY) = ay; ds(AX) = ax - s(AX); ds(AY) = ay - s(AY);
202 ds(YAW_RATE) = total_torque * inv_izz_; ds(YAW) = s(YAW_RATE);
203 ds(PX) = s(VX) * std::cos(s(YAW)) - s(VY) * std::sin(s(YAW));
204 ds(PY) = s(VX) * std::sin(s(YAW)) + s(VY) * std::cos(s(YAW));
205 ds(ST_ANGLE) = steering_motor_->compute_steering_rate(s(ST_ANGLE), steering_target);
206
207 if (state.ebs_active) {
208 double inv_dt_clamped = 1.0 / std::max(dt, 1e-6);
209 ds(FL_W) = -s(FL_W) * inv_dt_clamped;
210 ds(FR_W) = -s(FR_W) * inv_dt_clamped;
211 ds(RL_W) = -s(RL_W) * inv_dt_clamped;
212 ds(RR_W) = -s(RR_W) * inv_dt_clamped;
213 } else {
214 for (Tire tire : {FL, FR, RL, RR}) {
215 const double wheel_omega = s(FL_W + tire);
216 double net_torque = torques(tire) - tire_forces(tire * 4) * wheel_radius - tire_forces(tire * 4 + 2);
217
218 const double brake_sign = 2.0 / M_PI * std::atan(10.0 * wheel_omega);
219 const double brake_torque = brake_torques_by_tire(tire);
220 net_torque -= brake_torque * brake_sign;
221
222 if (tire == FL || tire == FR) {
223 net_torque -= car_parameters_->front_bearing_drag * wheel_omega;
224 }
225
226 // Fast multiplication instead of division
227 const double inv_inertia = (tire == FL || tire == FR) ? inv_front_wheel_inertia_ : inv_rear_wheel_inertia_;
228 const double wheel_acceleration = net_torque * inv_inertia;
229 if (brake_torque > 0.0 && std::abs(wheel_omega) < 0.5 && wheel_acceleration * wheel_omega <= 0.0) {
230 ds(FL_W + tire) = -wheel_omega / std::max(dt, 1e-6);
231 } else {
232 ds(FL_W + tire) = wheel_acceleration;
233 }
234 }
235 }

269 // The driven-wheel / tyre-slip mode is stiff: its time constant is
270 // tau = I_wheel * denom / (kxk * r^2), where denom = max(|Vw|,|Vcx|,1) grows
271 // with speed. So an explicit RK4 step is only unstable at low speed (near
272 // launch); at cruise the mode is well damped and a single step at the raw
273 // telemetry spacing (~1.5-2.8 ms) is fine. We therefore substep adaptively:
274 // the stable step scales with vehicle speed, so most of the run integrates in
275 // one step and only the first low-speed seconds pay for fine substeps.
276 static constexpr double kBaseSubDt = 2.0e-4; // stable step at ~1 m/s (dt_stable ~ 0.2ms * speed)
277 static constexpr int kMaxSub = 64;
278
279 void step(double dt, const common_lib::structures::Wheels& throttle, double angle, SimState& state) {
280 if (!(dt > 0.0)) return;
281 const double speed = std::sqrt(state.vx * state.vx + state.vy * state.vy);
282 const double stable_dt = kBaseSubDt * std::max(1.0, speed);
283 const int n_sub = std::min(kMaxSub, std::max(1, static_cast<int>(std::ceil(dt / stable_dt))));
284 const double sub_dt = dt / static_cast<double>(n_sub);
285 for (int k = 0; k < n_sub; ++k) {
286 integrate_substep(sub_dt, throttle, angle, state);
287 }
288 }

290 void integrate_substep(double dt, const common_lib::structures::Wheels& throttle, double angle, SimState& state) {
291 const double throttle_input = (throttle.rear_left + throttle.rear_right) / 2.0;
292 common_lib::structures::Wheels brake_torques;
293 double inverter_command = throttle_input;
294
295 if (control_mode_ == "manual" && throttle_input < 0.0) {

297 inverter_command = 0.0;
298 }

328 }
329

331 if (speed < 0.05 && (std::abs(throttle_input) < 0.01 || braking)) {
332 state.vx = 0.0; state.vy = 0.0; state.ax = 0.0; state.ay = 0.0; state.yaw_rate = 0.0;

344 state.ax, state.ay;
