3common_lib::structures::Wheels VDLoadTransferModel::compute_loads(
4 const LoadTransferInput& input) const {
5 double front_mass_distribution =
6 car_parameters_->cg_2_rear_axis / car_parameters_->wheelbase; // Value between 0 and 1
7 double downforce = input.downforce;
8 double front_static_load =
9 (car_parameters_->total_mass * car_parameters_->physical_constants->gravity) *
10 front_mass_distribution;
11 double rear_static_load =
12 (car_parameters_->total_mass * car_parameters_->physical_constants->gravity) *
13 (1 - front_mass_distribution);
14 double aero_load_front = abs(downforce) * car_parameters_->aero_parameters->aero_balance_front;
15 double aero_load_rear =
16 abs(downforce) * (1 - car_parameters_->aero_parameters->aero_balance_front);
17 double longitudinal_transfer = calculate_longitudinal_transfer(input.longitudinal_acceleration);
18 double lateral_transfer_front =
19 calculate_front_lateral_transfer(front_mass_distribution, input.lateral_acceleration);
20 double lateral_transfer_rear =
21 calculate_rear_lateral_transfer(front_mass_distribution, input.lateral_acceleration);
22 /*
23 Important notes: This calculation assumes that static load is distributed evenly between left and
24 right tires same for aero. Front static load returns the value of both tires Aero load front
25 returns the value of both tires Longitudinal transfer returns the total value of load transfer
26 that is then distributed between front and rear Lateral transfer returns the total value of load
27 transfer that is then distributed between each axis
28 */
29 double front_left_load = (front_static_load / 2) + (aero_load_front / 2) -
30 (longitudinal_transfer / 2) - lateral_transfer_front; // Front Left
31 double front_right_load = (front_static_load / 2) + (aero_load_front / 2) -
32 (longitudinal_transfer / 2) + lateral_transfer_front; // Front Right
33 double rear_left_load = (rear_static_load / 2) + (aero_load_rear / 2) +
34 (longitudinal_transfer / 2) - lateral_transfer_rear; // Rear Left
35 double rear_right_load = (rear_static_load / 2) + (aero_load_rear / 2) +
36 (longitudinal_transfer / 2) + lateral_transfer_rear; // Rear Right
37 return common_lib::structures::Wheels(front_left_load, front_right_load, rear_left_load,
38 rear_right_load);
39}

43double VDLoadTransferModel::calculate_front_lateral_transfer(double massDistribution,
44 double lateral_acceleration) const {
45 double unsprung_load_transfer = ((car_parameters_->unsprung_mass * massDistribution) *
46 lateral_acceleration * car_parameters_->unsprung_cg_z) /
47 car_parameters_->track_width;
48 double geometric_load_transfer =
49 ((car_parameters_->sprung_mass * massDistribution) * lateral_acceleration *
50 car_parameters_->load_transfer_parameters->front_roll_center_z) /
51 car_parameters_->track_width;
52 double elastic_load_transfer =
53 (car_parameters_->sprung_mass * lateral_acceleration *
54 car_parameters_->load_transfer_parameters->roll_axis_z *
55 car_parameters_->load_transfer_parameters->front_stiffness_distribution) /
56 car_parameters_->track_width;
57 return unsprung_load_transfer + geometric_load_transfer + elastic_load_transfer;
58}

60double VDLoadTransferModel::calculate_rear_lateral_transfer(double massDistribution,

66 ((car_parameters_->sprung_mass * (1 - massDistribution)) * lateral_acceleration *
