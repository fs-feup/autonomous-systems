24Eigen::Vector4d PacejkaMF6_2::tire_forces(const TireInput& tire_input) {
25 double Fx = 0;
26 double Fy = 0;
27 double MZ = 0;
28 double MY = 0;
29
30 // Calculate all the internal values needed for the calculations of forces and moments
31 calculate_tire_state(tire_input.slip_angle, tire_input.slip_ratio, tire_input.vertical_load,
32 tire_input.vx, tire_input.vy, tire_input.yaw_rate,
33 tire_input.wheel_angular_speed, tire_input.steering_angle,
34 tire_input.distance_to_CG, tire_input.camber_angle);
35
36 // Low speed fade for slip shifts
37 double speed = std::sqrt(tire_input.vx * tire_input.vx + tire_input.vy * tire_input.vy);
38 double shift_fade = std::clamp(speed / 0.1, 0.0, 1.0);
39
40 // Shifts for longitudinal and lateral slip
41 double SHx = calculate_SHx() * shift_fade;
42 double SHy = calculate_SHy() * shift_fade;
43
44 // (4.E20)
45 double shifted_slip_a = internal_vals.alpha_star + SHy;
46 // (4.E10)
47 double shifted_slip_r = tire_input.slip_ratio + SHx;
48
49 // Y parameter calculation
50 double Dy = calculate_Dy(tire_input.vertical_load);
51 double Cy = calculate_Cy(tire_input.vertical_load);
52 double By = calculate_By(Dy, Cy);
53 double Ey = calculate_Ey(shifted_slip_a);
54
55 // Vertical shift for lateral force
56 double SVy = calculate_SVy(tire_input.vertical_load) * shift_fade;
57
58 // X parameter calculation
59 double Dx = calculate_Dx(tire_input.vertical_load);
60 double Cx = calculate_Cx(tire_input.vertical_load);
61 double Bx = calculate_Bx(Dx, Cx);
62 double Ex = calculate_Ex(shifted_slip_r);
63
64 // Vertical shift for longitudinal force
65 double SVx = calculate_SVx(tire_input.vertical_load) * shift_fade;
66
67 // Pure slip calculations
68 double Fx0 = calculate_pure_slip(Bx, Cx, Dx, Ex, shifted_slip_r, SVx);
69 double Fy0 = calculate_pure_slip(By, Cy, Dy, Ey, shifted_slip_a, SVy);
70
71 // Combined slip calculations
72 Fx = calculate_combined_longitudinal(Fx0, tire_input.slip_ratio);
73 Fy = calculate_combined_lateral(Fy0, shifted_slip_r, tire_input.slip_ratio,
74 tire_input.vertical_load);
75
76 // Aligining moment calculation
77 MZ = calculate_combined_moment(Fx, Fy, tire_input.slip_ratio, SHy, SVy, By, Cy, internal_vals.Vcx,
78 tire_input.vertical_load);
79
80 // Rolling resistance moment calculation
81 MY = calculate_rolling_resistance_moment(tire_input.vertical_load, Fx, internal_vals.Vcx);
82
83 return Eigen::Vector4d(Fx, Fy, MY, MZ);
84}

101void PacejkaMF6_2::calculate_tire_state(double slip_angle, double slip_ratio, double vertical_load,
102 double vx, double vy, double yaw_rate,
103 double wheel_angular_speed, double steering_angle,
104 double distance_to_CG, double camber_angle) {
105 // Load related calculations
106 // (4.E1) -> Assuming we have a tire with a different nominal load we approxiamte using scaling
107 // factor LFZO The result is the adpated nominal load
108 internal_vals.Fz0_prime =
109 car_parameters_->tire_parameters->LFZO * car_parameters_->tire_parameters->FNOMIN;
110 // (4.E2) -> Since we normalized the nominal load we need to normalize the change in vertical load
111 internal_vals.dfz = (vertical_load - internal_vals.Fz0_prime) / internal_vals.Fz0_prime;
112 // Camber reduction factor
113 internal_vals.epsilong = car_parameters_->tire_parameters->PECP1 *
114 (1 + car_parameters_->tire_parameters->PECP2 * internal_vals.dfz);
115
116 // Velcoity related calculations
117 // Simple velocity of wheel contact center
118 internal_vals.Vc = sqrt(vx * vx + vy * vy);
119
120 // Velocity of wheel contact center with safety factor for zero speed calculations
121 internal_vals.Vc_prime = internal_vals.Vc + internal_vals.epsilon;
122
123 // Longitudinal velocity of the wheel center in the direction of the steering angle
124 internal_vals.Vcx = (vx * cos(steering_angle)) + (vy * sin(steering_angle)) +
125 yaw_rate * distance_to_CG * sin(steering_angle);
126 internal_vals.longitudinal_direction = std::copysign(1.0, internal_vals.Vcx);
127
128 double alpha_arg = std::clamp(std::abs(internal_vals.Vcx) / internal_vals.Vc_prime, 0.0, 1.0);
129 internal_vals.alpha_prime = acos(alpha_arg);
130
131 // Camber related calculations
132 // Total spin slip
133 internal_vals.phi =
134 (1 / internal_vals.Vc_prime) *
135 (yaw_rate - (1 - internal_vals.epsilong) * wheel_angular_speed * sin(camber_angle));
136
137 // Effective camber angle
138 internal_vals.gamma_star = camber_angle * car_parameters_->tire_parameters->camber_scaling_factor;
139
140 // Zetas
141 internal_vals.zeta1 = calculate_Zeta1(slip_ratio);
142 internal_vals.zeta2 = calculate_Zeta2(slip_angle);
143 internal_vals.zeta3 = calculate_Zeta3();
144 internal_vals.zeta7 = 1.0; // Ignore turnslip effects on self-aligning moment
145 internal_vals.zeta8 = 1.0; // same as zeta7
146
147 // Vertical load + Zetas
148 // Cornering stiffness at longitudinal slip zero
149 internal_vals.Kya =
150 car_parameters_->tire_parameters->PKY1 * internal_vals.Fz0_prime *
151 (1 + car_parameters_->tire_parameters->PPY1 * internal_vals.dpi) *
152 (1 - car_parameters_->tire_parameters->PKY3 * abs(internal_vals.gamma_star)) *
153 sin(car_parameters_->tire_parameters->PKY4 *
154 atan((vertical_load / internal_vals.Fz0_prime) /
155 ((car_parameters_->tire_parameters->PKY2 + car_parameters_->tire_parameters->PKY5 *
156 internal_vals.gamma_star *
157 internal_vals.gamma_star) *
158 (1 + car_parameters_->tire_parameters->PPY2 * internal_vals.dpi)))) *
159 internal_vals.zeta3 * car_parameters_->tire_parameters->LKY; // Ensure not dividing by zero
160
161 internal_vals.Kya_prime =
162 internal_vals.Kya + internal_vals.epsilon * std::copysign(1.0, internal_vals.Kya);
163
164 // (4.E8) Special degressive friction factor
165 internal_vals.LMUY_prime =
166 car_parameters_->tire_parameters->Amu * car_parameters_->tire_parameters->LMUY /
167 (1 + (car_parameters_->tire_parameters->Amu - 1) * car_parameters_->tire_parameters->LMUY);
168
169 internal_vals.SVyg = vertical_load *
170 (car_parameters_->tire_parameters->PVY3 +
171 car_parameters_->tire_parameters->PVY4 * internal_vals.dfz) *
172 internal_vals.gamma_star * car_parameters_->tire_parameters->LKYC *
173 internal_vals.LMUY_prime * internal_vals.zeta2;
174 // (4.E3)
175 internal_vals.alpha_star = tan(slip_angle) * internal_vals.longitudinal_direction;
176 internal_vals.muy =
177 (car_parameters_->tire_parameters->PDY1 +
178 car_parameters_->tire_parameters->PDY2 * internal_vals.dfz) *
179 (1 + car_parameters_->tire_parameters->PPY3 * internal_vals.dpi +
180 car_parameters_->tire_parameters->PPY4 * (internal_vals.dpi * internal_vals.dpi)) *
181 (1 - car_parameters_->tire_parameters->PDY3 *
182 (internal_vals.gamma_star * internal_vals.gamma_star)) *
183 car_parameters_->tire_parameters
184 ->LMUY; // we can assume dpi is zero for now and ignore pressure variations
185
186 internal_vals.Kyg0 = vertical_load *
187 (car_parameters_->tire_parameters->PKY6 +
188 car_parameters_->tire_parameters->PKY7 * internal_vals.dfz) *
189 (1 + car_parameters_->tire_parameters->PPY5 * internal_vals.dpi) *
190 car_parameters_->tire_parameters->LKYC;
191
192 internal_vals.SHyp = calculate_SHyp(camber_angle);
193
194 internal_vals.SVyk = calculate_SVyk(vertical_load, slip_ratio);
195 internal_vals.kxk =
196 vertical_load *
197 (car_parameters_->tire_parameters->PKX1 +
198 car_parameters_->tire_parameters->PKX2 * internal_vals.dfz) *
199 exp(car_parameters_->tire_parameters->PKX3 * internal_vals.dfz) *
200 (1 + car_parameters_->tire_parameters->PPX1 * internal_vals.dpi +
201 car_parameters_->tire_parameters->PPX2 * (internal_vals.dpi * internal_vals.dpi)) *
202 car_parameters_->tire_parameters->LKX;
203 return;
204}

292double PacejkaMF6_2::calculate_combined_longitudinal(double Fx0, double slip_ratio) const {
293 // (4.E53)
294 double as = internal_vals.alpha_star + car_parameters_->tire_parameters->RHX1;
295 // (4.E54)
296 double Bxa = car_parameters_->tire_parameters->RBX1 *
297 cos(atan(car_parameters_->tire_parameters->RBX2 * slip_ratio)) *
298 car_parameters_->tire_parameters->LXAL;
299 // (4.E56)
300 double Exa = (car_parameters_->tire_parameters->REX1 +
301 car_parameters_->tire_parameters->REX2 * internal_vals.dfz);
302 // (4.E52)
303 double Gxa0 = cos(car_parameters_->tire_parameters->RCX1 *
304 atan(Bxa * car_parameters_->tire_parameters->RHX1 -
305 Exa * (Bxa * car_parameters_->tire_parameters->RHX1 -
306 atan(Bxa * car_parameters_->tire_parameters->RHX1))));
307 // (4.E51)
308 double Gxa = cos(car_parameters_->tire_parameters->RCX1 *
309 atan(Bxa * as - Exa * (Bxa * as - atan(Bxa * as)))) /
310 Gxa0;
311 // (4.E50)
312 return Fx0 * Gxa;
313}

315double PacejkaMF6_2::calculate_combined_lateral(double Fy0, double shifted_slip_ratio,
316 double slip_ratio, double vertical_load) const {
317 // Calculation of SVyk
318 // (4.E67)
319 double DVyk = internal_vals.muy * vertical_load *
320 (car_parameters_->tire_parameters->RVY1 +
321 car_parameters_->tire_parameters->RVY2 * internal_vals.dfz +
322 car_parameters_->tire_parameters->RVY3 * internal_vals.gamma_star) *
323 cos(atan(car_parameters_->tire_parameters->RVY4 * internal_vals.alpha_star)) *
324 internal_vals.zeta2;
325 // (4.E66)
326 double SVyk = DVyk * sin(car_parameters_->tire_parameters->RVY5 *
327 atan(car_parameters_->tire_parameters->RVY6 * slip_ratio));
328 // Calculation of Gyk
329 // (4.E64)
330 double Eyk = car_parameters_->tire_parameters->REY1 +
331 car_parameters_->tire_parameters->REY2 * internal_vals.dfz;
332 Eyk = (Eyk < 1) ? Eyk : 1.0; // E cannot be over 1
333 // (4.E65)
334 double SHyk = car_parameters_->tire_parameters->RHY1 +
335 car_parameters_->tire_parameters->RHY2 * internal_vals.dfz;
336 // (4.E62)
337 double Byk = car_parameters_->tire_parameters->RBY1 *
338 cos(atan(car_parameters_->tire_parameters->RBY2 *
339 (internal_vals.alpha_star - car_parameters_->tire_parameters->RBY3))) *
340 car_parameters_->tire_parameters->LYKA;
341 // (4.E60)
342 double Gyk0 = cos(car_parameters_->tire_parameters->RCY1 *
343 atan(Byk * SHyk - Eyk * (Byk * SHyk - atan(Byk * SHyk))));
344 // (4.E59)
345 double Gyk = cos(car_parameters_->tire_parameters->RCY1 *
346 atan(Byk * shifted_slip_ratio -
347 Eyk * (Byk * shifted_slip_ratio - atan(Byk * shifted_slip_ratio)))) /
348 Gyk0;
349 // (4.E58)
350 return Fy0 * Gyk + SVyk;
351}

353// This function is never used (Im stupid and thought that the pure slip aligning moment was needed

355double PacejkaMF6_2::calculate_pure_moment(double Fy0, double normal_load, double SHy, double SVy,
356 double By, double Cy,
357 double longitudinal_velocity) const {
358 // MZ0_prime calculation
359 // ------------------ at calculation
360 // (4.E35)
361 double SHt = car_parameters_->tire_parameters->QHZ1 +
362 car_parameters_->tire_parameters->QHZ2 * internal_vals.dfz +
363 (car_parameters_->tire_parameters->QHZ3 +
364 car_parameters_->tire_parameters->QHZ4 * internal_vals.dfz) *
365 internal_vals.gamma_star;
366 // (4.E34)
367 double at = internal_vals.alpha_star + SHt;
368 // ------------------ Bt calculation
369 // (4.E40)
370 double Bt = (car_parameters_->tire_parameters->QBZ1 +
371 car_parameters_->tire_parameters->QBZ2 * internal_vals.dfz +
372 car_parameters_->tire_parameters->QBZ3 * internal_vals.dfz * internal_vals.dfz) *
373 (1 + car_parameters_->tire_parameters->QBZ4 * internal_vals.gamma_star +
374 car_parameters_->tire_parameters->QBZ5 * abs(internal_vals.gamma_star)) *
375 internal_vals.Kya / car_parameters_->tire_parameters->LMUY;
376 // ------------------ Dt calculation
377 // (4.91)
378 double zeta5 = cos(atan(car_parameters_->tire_parameters->QDTP1 *
379 car_parameters_->tire_parameters->effective_tire_r * internal_vals.phi));
380 // (4.E42) -> expanded to account for pressure changes (dpi = 0 to ignore)
381 double Dt0 = normal_load *
382 (car_parameters_->tire_parameters->effective_tire_r / internal_vals.Fz0_prime) *
383 (car_parameters_->tire_parameters->QDZ1 +
384 car_parameters_->tire_parameters->QDZ2 * internal_vals.dfz) *
385 (1 - car_parameters_->tire_parameters->PPZ1) *
386 car_parameters_->tire_parameters->LTR * std::copysign(1.0, longitudinal_velocity);
387 // (4.E43)
388 double Dt = Dt0 *
389 (1 + car_parameters_->tire_parameters->QDZ3 * internal_vals.gamma_star +
390 car_parameters_->tire_parameters->QDZ4 *
391 (internal_vals.gamma_star * internal_vals.gamma_star)) *
392 zeta5;
393 // ------------------ Et calculation
394 // (4.E44)
395 double Et = (car_parameters_->tire_parameters->QEZ1 +
396 car_parameters_->tire_parameters->QEZ2 * internal_vals.dfz +
397 car_parameters_->tire_parameters->QEZ3 * internal_vals.dfz * internal_vals.dfz) *
398 (1 + (car_parameters_->tire_parameters->QEZ4 * internal_vals.gamma_star +
399 car_parameters_->tire_parameters->QEZ5 * internal_vals.gamma_star *
400 internal_vals.gamma_star) *
401 (2 / M_PI) * atan(Bt * car_parameters_->tire_parameters->QCZ1 * at));
402 Et = (Et < 1) ? Et : 1.0; // Et cannot be over 1
403 // ------------------
404 // (4.E6)
405
406 // (4.E33)
407 double t0 =
408 Dt *
409 cos(car_parameters_->tire_parameters->QCZ1 * atan(Bt * at - Et * (Bt * at - atan(Bt * at)))) *
410 cos(internal_vals.alpha_prime);
411 // (4.E32)
412 double MZ0_prime = -t0 * Fy0;
413 // MZr0 calculation
414 // (4.E47)
415 double Dr = normal_load * car_parameters_->tire_parameters->effective_tire_r *
416 ((car_parameters_->tire_parameters->QDZ6 +
417 car_parameters_->tire_parameters->QDZ7 * internal_vals.dfz) *
418 car_parameters_->tire_parameters->LRES * internal_vals.zeta2 +
419 (car_parameters_->tire_parameters->QDZ8 +
420 car_parameters_->tire_parameters->QDZ9 * internal_vals.dfz) *
421 internal_vals.gamma_star * car_parameters_->tire_parameters->LKZC *
422 internal_vals.zeta0) *
423 cos(internal_vals.alpha_prime) * car_parameters_->tire_parameters->LMUY *
424 std::copysign(1.0, longitudinal_velocity) +
425 internal_vals.zeta8 - 1;
426 // (4.E45)
427 double Br = car_parameters_->tire_parameters->QBZ10 * By * Cy;
428 // (4.E38)
429 double SHf = SHy + SVy / (internal_vals.Kya +
430 internal_vals.epsilon * std::copysign(1.0, internal_vals.Kya));
431 // (4.E37)
432 double ar = internal_vals.alpha_star + SHf;
433 // (4.E36) -> currently we set zeta7 to 1 to ignore turnslip effects, but if we want to consider
434 // them we need to calculate zeta7
435 double MZr0 = Dr * cos(internal_vals.zeta7 * atan(Br * ar));
436 // (4.E31)
437 return MZ0_prime + MZr0;
438}

440double PacejkaMF6_2::calculate_combined_moment(double Fx, double Fy, double slip_ratio, double SHy,
441 double SVy, double By, double Cy,
442 double longitudinal_velocity,
443 double normal_load) const {
444 // MZ_prime calculation
445 // ------------------ at_eq calculation
446 // (4.E35)
447 double SHt = car_parameters_->tire_parameters->QHZ1 +
448 car_parameters_->tire_parameters->QHZ2 * internal_vals.dfz +
449 (car_parameters_->tire_parameters->QHZ3 +
450 car_parameters_->tire_parameters->QHZ4 * internal_vals.dfz) *
451 internal_vals.gamma_star;
452 // (4.E34)
453 double at = internal_vals.alpha_star + SHt;
454 // (4.E77)
455 double at_eq =
456 sqrt(at * at + (internal_vals.kxk / internal_vals.Kya_prime) *
457 (internal_vals.kxk / internal_vals.Kya_prime) * slip_ratio * slip_ratio) *
458 std::copysign(1.0, at);
459 // ------------------ Bt calculation
460 // (4.E40)
461 double Bt = (car_parameters_->tire_parameters->QBZ1 +
462 car_parameters_->tire_parameters->QBZ2 * internal_vals.dfz +
463 car_parameters_->tire_parameters->QBZ3 * internal_vals.dfz * internal_vals.dfz) *
464 (1 + car_parameters_->tire_parameters->QBZ4 * internal_vals.gamma_star +
465 car_parameters_->tire_parameters->QBZ5 * abs(internal_vals.gamma_star)) *
466 internal_vals.Kya / car_parameters_->tire_parameters->LMUY;
467 // ------------------ Dt calculation
468 // (4.91)
469 double zeta5 = cos(atan(car_parameters_->tire_parameters->QDTP1 *
470 car_parameters_->tire_parameters->effective_tire_r * internal_vals.phi));
471 // (4.E42) -> expanded to account for pressure changes (dpi = 0 to ignore)
472 double Dt0 = normal_load *
473 (car_parameters_->tire_parameters->effective_tire_r / internal_vals.Fz0_prime) *
474 (car_parameters_->tire_parameters->QDZ1 +
475 car_parameters_->tire_parameters->QDZ2 * internal_vals.dfz) *
476 (1 - car_parameters_->tire_parameters->PPZ1) *
477 car_parameters_->tire_parameters->LTR * internal_vals.longitudinal_direction;
478 // (4.E43)
479 double Dt = Dt0 *
480 (1 + car_parameters_->tire_parameters->QDZ3 * internal_vals.gamma_star +
481 car_parameters_->tire_parameters->QDZ4 *
482 (internal_vals.gamma_star * internal_vals.gamma_star)) *
483 zeta5;
484 // ------------------ Et calculation
485 // (4.E44)
486 double Et = (car_parameters_->tire_parameters->QEZ1 +
487 car_parameters_->tire_parameters->QEZ2 * internal_vals.dfz +
488 car_parameters_->tire_parameters->QEZ3 * internal_vals.dfz * internal_vals.dfz) *
489 (1 + (car_parameters_->tire_parameters->QEZ4 * internal_vals.gamma_star +
490 car_parameters_->tire_parameters->QEZ5 * internal_vals.gamma_star *
491 internal_vals.gamma_star) *
492 (2 / M_PI) * atan(Bt * car_parameters_->tire_parameters->QCZ1 * at));
493 Et = (Et < 1) ? Et : 1.0; // Et cannot be over 1
494 // ------------------ t calculation
495 // (4.E73)
496 double t = Dt *
497 cos(car_parameters_->tire_parameters->QCZ1 *
498 atan(Bt * at_eq - Et * (Bt * at_eq - atan(Bt * at_eq)))) *
499 cos(internal_vals.alpha_prime);
500 // (4.E74)
501 double Fy_prime = Fy - internal_vals.SVyk;
502 // (4.E72)
503 double MZ_prime = -t * Fy_prime;
504 // MZr calculation
505 // (4.E47)
506 double Dr = normal_load * car_parameters_->tire_parameters->effective_tire_r *
507 ((car_parameters_->tire_parameters->QDZ6 +
508 car_parameters_->tire_parameters->QDZ7 * internal_vals.dfz) *
509 car_parameters_->tire_parameters->LRES * internal_vals.zeta2 +
510 (car_parameters_->tire_parameters->QDZ8 +
511 car_parameters_->tire_parameters->QDZ9 * internal_vals.dfz) *
512 internal_vals.gamma_star * car_parameters_->tire_parameters->LKZC *
513 internal_vals.zeta0) *
514 cos(internal_vals.alpha_prime) * car_parameters_->tire_parameters->LMUY *
515 internal_vals.longitudinal_direction +
516 internal_vals.zeta8 - 1;
517 // (4.E45)
518 double Br = car_parameters_->tire_parameters->QBZ10 * By * Cy;
519 // (4.E38)
520 double SHf = SHy + SVy / (internal_vals.Kya +
521 internal_vals.epsilon * std::copysign(1.0, internal_vals.Kya));
522 // (4.E37)
523 double ar = internal_vals.alpha_star + SHf;
524 // (4.E78)
525 double ar_eq =
526 sqrt(ar * ar + (internal_vals.kxk / internal_vals.Kya_prime) *
527 (internal_vals.kxk / internal_vals.Kya_prime) * slip_ratio * slip_ratio) *
528 std::copysign(1.0, ar);
529 // (4.E75)
530 double MZr = Dr * cos(internal_vals.zeta7 * atan(Br * ar_eq));
531 // (4.E71)
532 // s calculation
533 // (4.E76)
534 double s = car_parameters_->tire_parameters->effective_tire_r *
535 (car_parameters_->tire_parameters->SSZ1 +
536 car_parameters_->tire_parameters->SSZ2 * (Fy / internal_vals.Fz0_prime) +
537 (car_parameters_->tire_parameters->SSZ3 +
538 car_parameters_->tire_parameters->SSZ4 * internal_vals.dfz) *
539 internal_vals.gamma_star) *
540 car_parameters_->tire_parameters->LS;
541 return MZ_prime + MZr + s * Fx;
542}

544double PacejkaMF6_2::calculate_rolling_resistance_moment(double vertical_load, double Fx,
545 double vx) const {
546 double Fz0 = internal_vals.Fz0_prime;
547 double unloaded_radius = car_parameters_->tire_parameters->UNLOADED_RADIUS;
548 double QSY1 = car_parameters_->tire_parameters->QSY1;
549 double QSY2 = car_parameters_->tire_parameters->QSY2;
550 double QSY3 = car_parameters_->tire_parameters->QSY3;
551 double QSY4 = car_parameters_->tire_parameters->QSY4;
552
553 // Standard Pacejka reference velocity is 16.666 m/s (60 km/h)
554 const double Vref = 16.666;
555
556 // (4.E85)
557 double My =
558 unloaded_radius * vertical_load *
559 (QSY1 + QSY2 * (Fx / Fz0) + QSY3 * std::abs(vx / Vref) + QSY4 * std::pow(vx / Vref, 4));
560
561 return My;
562}

569 // (4.105)

591double PacejkaMF6_2::calculate_SHyp(double camber_angle) const {
592 // MF 6.1 and 6.2
593 double KyRp0 = internal_vals.Kyg0 / (1 - internal_vals.epsilong);
594
595 double Kya0 =
596 car_parameters_->tire_parameters->PKY1 * internal_vals.Fz0_prime *
597 (1 + car_parameters_->tire_parameters->PPY1 * internal_vals.dpi) *
598 sin(car_parameters_->tire_parameters->PKY4 *
599 (1 + car_parameters_->tire_parameters->PPX3 * internal_vals.dpi +
600 car_parameters_->tire_parameters->PPX4 * (internal_vals.dpi * internal_vals.dpi)) *
601 (1 - car_parameters_->tire_parameters->PDX3 * (camber_angle * camber_angle))) *
602 car_parameters_->tire_parameters->LMUX;
603 double Kyao_prime = Kya0 + internal_vals.epsilon * std::copysign(1.0, Kya0);
604
605 double DHyp = (car_parameters_->tire_parameters->PHYP2 +
606 car_parameters_->tire_parameters->PHYP3 * internal_vals.dfz) *
607 internal_vals.longitudinal_direction;
608
609 double BHyp = KyRp0 / (car_parameters_->tire_parameters->PHYP1 * DHyp * Kyao_prime);
610
611 double EHyp =
612 (car_parameters_->tire_parameters->PHYP4 > 1) ? 1 : car_parameters_->tire_parameters->PHYP4;
613
614 return DHyp *
615 sin(car_parameters_->tire_parameters->PHYP1 *
616 atan(BHyp * car_parameters_->tire_parameters->effective_tire_r * internal_vals.phi -
617 EHyp * (BHyp * car_parameters_->tire_parameters->effective_tire_r *
618 internal_vals.phi -
619 atan(BHyp * car_parameters_->tire_parameters->effective_tire_r *
620 internal_vals.phi)))) *
621 internal_vals.longitudinal_direction;
622}
623double PacejkaMF6_2::calculate_SVyk(double vertical_load, double slip_ratio) const {
624 // (4.E67)
625 double DVyk = internal_vals.muy * vertical_load *
626 (car_parameters_->tire_parameters->RVY1 +
627 car_parameters_->tire_parameters->RVY2 * internal_vals.dfz +
628 car_parameters_->tire_parameters->RVY3 * internal_vals.gamma_star) *
629 cos(atan(car_parameters_->tire_parameters->RVY4 * internal_vals.alpha_star)) *
630 internal_vals.zeta2;
631 // (4.E66)
632 return DVyk * sin(car_parameters_->tire_parameters->RVY5 *
633 atan(car_parameters_->tire_parameters->RVY6 * slip_ratio));
