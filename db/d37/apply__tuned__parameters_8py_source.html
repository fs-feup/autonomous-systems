28def update_file_parameter(config_file: Path, target_key: str, new_value) -> bool:
29 if not config_file.exists():
30 return False
31
32 with open(config_file, 'r') as f:
33 lines = f.readlines()
34
35 # Regex to find: target_key: <number> [optional comment]
36 pattern = re.compile(r'^(\s*' + re.escape(target_key) + r'\s*:\s*)([-+]?[0-9]*\.?[0-9]+(?:[eE][-+]?[0-9]+)?)(.*)$')
37
38 replaced = False
39 for i, line in enumerate(lines):
40 match = pattern.match(line)
41 if match:
42 prefix = match.group(1)
43 suffix = match.group(3)
44
45 if isinstance(new_value, float):
46 val_str = f"{new_value:.6g}"
47 else:
48 val_str = str(new_value)
49
50 lines[i] = f"{prefix}{val_str}{suffix}\n"
51 replaced = True
52 break
53
54 if replaced:
55 with open(config_file, 'w') as f:
56 f.writelines(lines)
57
58 return replaced
