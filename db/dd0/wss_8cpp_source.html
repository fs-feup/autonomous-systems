85double WSS::apply_wss_error(double rotational_speed, WheelPosition wheel) {
86 int wheel_idx = static_cast<int>(wheel);
87 const WheelParameters& params = wheel_params_[wheel_idx];
88
89 // Apply speed cutoff
90 double measurement = rotational_speed;
91 if (std::abs(rotational_speed) < params.speed_cutoff) {
92 measurement = 0.0;
93 }
94
95 // Apply white Gaussian noise
96 measurement = measurement + gaussian_noise(params.noise_std_dev);
97
98 // Apply outlier
99 measurement = apply_outlier(measurement, wheel);
100
101 // Apply dropout
102 measurement = get_dropout_info(measurement, wheel);
103
104 return measurement;
105}

107double WSS::apply_outlier(double value, WheelPosition wheel) {
108 int wheel_idx = static_cast<int>(wheel);
109 const WheelParameters& params = wheel_params_[wheel_idx];
110
111 // Check if outlier should occur
112 if (random_uniform() < params.outlier_probability) {
113 // Generate outlier by scaling the current measurement
114 // The sign is randomly chosen
115 double sign = (random_uniform() < 0.5) ? -1.0 : 1.0;
116 return value + (sign * params.outlier_impact_factor * std::abs(value));
117 }
118 return value;
119}

137double WSS::apply_quantization(double value, WheelPosition wheel) {
138 int wheel_idx = static_cast<int>(wheel);
139 const WheelParameters& params = wheel_params_[wheel_idx];
140
141 // Quantize the value based on the configured bit depth
142 // Assumes range of [-4000, 4000] rpm
143 constexpr double MIN_RPM = -4000.0;
144 constexpr double MAX_RPM = 4000.0;
145
146 // Calculate the number of levels: 2^bits
147 long long num_levels = 1LL << params.quantization_steps; // 2^quantization_steps
148
149 // Calculate step size: (max - min) / (levels - 1)
150 double step_size = (MAX_RPM - MIN_RPM) / (num_levels - 1);
151
152 // Clamp value to range
153 double clamped = std::max(MIN_RPM, std::min(MAX_RPM, value));
154
155 // Quantize: find nearest level
156 double quantized = std::round((clamped - MIN_RPM) / step_size) * step_size + MIN_RPM;
157
158 return quantized;
159}
