85void InvictaSim::simulation_step() {
86 auto current_time = std::chrono::steady_clock::now();
87
88 // Calculate step duration
89 double step_duration = (1.0 / static_cast<double>(params_.sim_frequency)) / params_.sim_speed;
90 auto step_duration_ros = std::chrono::duration_cast<std::chrono::steady_clock::duration>(
91 std::chrono::duration<double>(step_duration));
92
93 // Sleep until the next step
94 if (current_time < next_step_time_) {
95 std::this_thread::sleep_until(next_step_time_);
96 current_time = std::chrono::steady_clock::now();
97 }
98
99 // Real time passed since last step
100 double real_dt = std::chrono::duration<double>(current_time - last_step_time_).count();
101
102 // Scale passed time by simulation speed
103 double sim_dt = real_dt * params_.sim_speed;
104
105 // Update next step
106 last_step_time_ = current_time;
107 next_step_time_ = current_time + step_duration_ros;
108
109 // Increase simulation time
110 sim_time_ += sim_dt;
111 const auto step_start = std::chrono::steady_clock::now();
112
113 // Use a copy of input so no locks are required during vehicle model step.
114 const InputSnapshot input_snapshot = get_input_snapshot();
115
116 // Use snapshot throughout step without locks
117 vehicle_model_->step(sim_dt, input_snapshot.throttle, input_snapshot.steering);
118 VehicleModelSnapshot vehicle_snapshot = build_vehicle_model_snapshot();
119
120 statistics_->update(vehicle_snapshot, sim_time_, sim_dt, input_snapshot.external_path_points);
121
122 // Compute total step execution time
123 const auto step_end = std::chrono::steady_clock::now();
124 const double total_step_ms =
125 std::chrono::duration<double, std::milli>(step_end - step_start).count();
126
127 // Update output snapshot for adapters to read (lock only to copy the data)
128 ExecutionTimesSnapshot execution_times_snapshot = build_execution_times_snapshot(total_step_ms);
129 StatisticsSnapshot statistics_snapshot = statistics_->get_snapshot();
130 MapSnapshot map_snapshot = build_map_snapshot(input_snapshot);
131 SensorsSnapshot sensors_snapshot = build_sensors_snapshot(vehicle_snapshot);
132 VehicleStateSnapshot vehicle_state_snapshot = build_vehicle_state_snapshot();
133 {
134 std::lock_guard<std::mutex> lock(output_snapshot_mutex_);
135 vehicle_model_snapshot_ = vehicle_snapshot;
136 execution_times_snapshot_ = execution_times_snapshot;
137 map_snapshot_ = map_snapshot;
138 sensors_snapshot_ = sensors_snapshot;
139 vehicle_state_snapshot_ = vehicle_state_snapshot;
140 statistics_snapshot_ = statistics_snapshot;
141 }
142}
