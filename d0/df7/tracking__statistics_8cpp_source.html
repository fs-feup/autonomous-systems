56std::pair<double, double> TrackingStatistics::tracking_error_and_objective_velocity(
57 const common_lib::structures::Position& car_position,
58 const std::vector<common_lib::structures::PathPoint>& path_points) const {
59 if (path_points.empty()) {
60 return {0.0, 0.0};
61 }
62
63 if (path_points.size() == 1) {
64 return {distance_between(car_position, path_points.front().position),
65 path_points.front().ideal_velocity};
66 }
67
68 double best_distance_sq = std::numeric_limits<double>::max();
69 double best_velocity = 0.0;
70
71 for (std::size_t i = 0; i + 1 < path_points.size(); ++i) {
72 const auto& start = path_points[i];
73 const auto& end = path_points[i + 1];
74 const double segment_x = end.position.x - start.position.x;
75 const double segment_y = end.position.y - start.position.y;
76 const double segment_length_sq = segment_x * segment_x + segment_y * segment_y;
77
78 double interpolation = 0.0;
79 if (segment_length_sq > std::numeric_limits<double>::epsilon()) {
80 const double car_x = car_position.x - start.position.x;
81 const double car_y = car_position.y - start.position.y;
82 interpolation =
83 std::clamp((car_x * segment_x + car_y * segment_y) / segment_length_sq, 0.0, 1.0);
84 }
85
86 const double projected_x = start.position.x + interpolation * segment_x;
87 const double projected_y = start.position.y + interpolation * segment_y;
88 const double error_x = car_position.x - projected_x;
89 const double error_y = car_position.y - projected_y;
90 const double distance_sq = error_x * error_x + error_y * error_y;
91
92 if (distance_sq < best_distance_sq) {
93 best_distance_sq = distance_sq;
94 best_velocity =
95 start.ideal_velocity + interpolation * (end.ideal_velocity - start.ideal_velocity);
96 }
97 }
98
99 return {std::sqrt(best_distance_sq), best_velocity};
100}
