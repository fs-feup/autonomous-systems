5unsigned int get_s_index(std::vector<double>& s, double query_s) {
6 // Simple linear search (can be optimized with binary search if needed)
7 for (size_t i = 0; i < s.size() - 1; ++i) {
8 if (s[i] > query_s) {
9 return i - 1;
10 }
11 }
12 return s.size() - 2; // Return the last valid index if out of bounds
13}

15double acceleration_for_segment(unsigned int start_idx, std::vector<double>& v, std::vector<double>& s) {
16 double distance = s[start_idx + 1] - s[start_idx];
17 double v_start = v[start_idx];
18 double v_end = v[start_idx + 1];
19
20 if (distance > 0.01) {
21 return (v_end * v_end - v_start * v_start) / (2 * distance);
22 } else {
23 return 0.0; // Avoid division by zero, treat as no acceleration
24 }
25}

27unsigned int compute_starting_index(const custom_interfaces::msg::PathPointArray& path_msg, const custom_interfaces::msg::VehicleStateVector& vehicle_state) {
28
29 size_t closest_idx = -1;
30
31 double x = vehicle_state.x;
32 double y = vehicle_state.y;
33
34 // Find the closest point to the vehicle
35 double min_distance = std::numeric_limits<double>::max();
36 for (size_t i = 0; i < path_msg.pathpoint_array.size(); ++i) {
37 double dx = path_msg.pathpoint_array[i].x - x;
38 double dy = path_msg.pathpoint_array[i].y - y;
39 double distance = std::sqrt(dx * dx + dy * dy);
40 if (distance < min_distance) {
41 min_distance = distance;
42 closest_idx = i;
43 }
44 }
45
46 // Start from the previous point for better continuity
47 if (closest_idx > 0 && closest_idx < path_msg.pathpoint_array.size() - 1) {
48 double prev_dx = path_msg.pathpoint_array[closest_idx - 1].x - x;
49 double prev_dy = path_msg.pathpoint_array[closest_idx - 1].y - y;
50 double next_dx = path_msg.pathpoint_array[closest_idx + 1].x - x;
51 double next_dy = path_msg.pathpoint_array[closest_idx + 1].y - y;
52 double next_sq_distance = next_dx * next_dx + next_dy * next_dy;
53 double prev_sq_distance = prev_dx * prev_dx + prev_dy * prev_dy;
54 if (prev_sq_distance < next_sq_distance) closest_idx--;
55 }
56
57 return closest_idx;
58}

60void limit_velocity_according_to_current(custom_interfaces::msg::PathPointArray& path_msg, double max_acceleration = 10.0) {
61 if (path_msg.pathpoint_array.empty()) {
62 return;
63 }
64
65 // Limit velocity segment-by-segment using the previous point as the starting state
66 // Start from the vehicle's current pose (front of the path) and current_velocity
67 custom_interfaces::msg::PathPoint prev_point = path_msg.pathpoint_array.front();
68 double prev_velocity = prev_point.v;
69
70 for (size_t i = 0; i < path_msg.pathpoint_array.size(); ++i) {
71 double dx = path_msg.pathpoint_array[i].x - prev_point.x;
72 double dy = path_msg.pathpoint_array[i].y - prev_point.y;
73 double distance = std::sqrt(dx * dx + dy * dy);
74 double max_velocity = std::sqrt(std::max(0.0, prev_velocity * prev_velocity + 2 * max_acceleration * distance));
75 if (path_msg.pathpoint_array[i].v > max_velocity) {
76 path_msg.pathpoint_array[i].v = max_velocity;
77 }
78
79 // Use the velocity and point as the previous for the next iteration
80 prev_velocity = path_msg.pathpoint_array[i].v;
81 prev_point = path_msg.pathpoint_array[i];
82 }
83}

137 double next_yaw_rate = car_yaw_rate - effective_yaw_acceleration * dt;
