95void RosOutputAdapter::load_publish_frequencies(const std::string& config_file) {
96 std::string full_path =
97 common_lib::config_load::get_config_yaml_path("invictasim", "invictasim/output", config_file);
98 YAML::Node config = YAML::LoadFile(full_path);
99 if (!config["publish_frequencies"]) return;
100
101 load_group_from_yaml(config, "vehicle_model");
102 load_group_from_yaml(config, "visualization");
103 load_group_from_yaml(config, "sensors");
104 load_group_from_yaml(config, "map");
105 load_group_from_yaml(config, "vehicle_state");
106
107 if (config["publish_frequencies"]["execution_time"]) {
108 topic_frequencies_["execution_time"] =
109 config["publish_frequencies"]["execution_time"].as<int>();
110 }
111
112 // After loading the config, build the dispatch table
113 map_callbacks();
114}

191void RosOutputAdapter::load_group_from_yaml(const YAML::Node& config,
192 const std::string& group_name) {
193 const YAML::Node& group_node = config["publish_frequencies"][group_name];
194 if (group_node && group_node.IsMap()) {
195 for (const auto& node : group_node) {
196 std::string topic_key = node.first.as<std::string>();
197 int frequency = node.second.as<int>();
198
199 topic_frequencies_[topic_key] = frequency;
200 }
201 }
202}

204void RosOutputAdapter::register_pub_helper(const std::string& topic,
205 std::function<void(const rclcpp::Time&)> func) {
206 if (topic_frequencies_.count(topic) && topic_frequencies_[topic] > 0) {
207 frequency_callbacks_[topic_frequencies_[topic]].push_back(func);

211void RosOutputAdapter::setup_timers() {
212 for (const auto& [frequency_hz, callbacks] : frequency_callbacks_) {
213 auto period = std::chrono::duration_cast<std::chrono::nanoseconds>(
214 std::chrono::duration<double>(1.0 / static_cast<double>(frequency_hz)));
215
216 frequency_timers_[frequency_hz] = this->create_wall_timer(period, [this, frequency_hz]() {
217 if (!running_) return;
218 on_frequency_tick(frequency_hz);
219 });
220 }
221}

223void RosOutputAdapter::on_frequency_tick(int frequency_hz) {
224 const rclcpp::Time stamp = this->now();
225
226 // Refresh snapshots
227 refresh_vehicle_model_snapshot();
228 refresh_execution_times_snapshot();
229 refresh_map_snapshot();
230 refresh_sensors_snapshot();
231 refresh_vehicle_state_snapshot();
232
233 // Execute functions for this frequency
234 for (const auto& publish_func : frequency_callbacks_[frequency_hz]) {
235 publish_func(stamp);
236 }
237}

623 m.scale.y = 1.0;

626 m.color.r = 1.0f;
627 m.color.g = 1.0f;
628 m.color.b = 1.0f;
629 m.color.a = 1.0f;

633 switch (cone.color) {

647 case common_lib::competition_logic::Color::GREEN:

652 m.mesh_resource = path + "cone_orange.dae";

696void RosOutputAdapter::add_wheel_markers(visualization_msgs::msg::MarkerArray& marker_array,
697 const rclcpp::Time& stamp, double dt) {
698 const auto car_params = simulator_->get_params().car_parameters;
699 const double wheel_center_z = car_params->wheel_diameter * 0.5;
700 const double long_offset = 0.15;
701
702 if (dt > 0.0) {
703 const auto wheel_speed = vehicle_model_snapshot_cache_.wheel_speed;
704 wheel_spin_fl_ += (wheel_speed.front_left) * dt;
705 wheel_spin_fr_ += (wheel_speed.front_right) * dt;
706 wheel_spin_rl_ += (wheel_speed.rear_left) * dt;
707 wheel_spin_rr_ += (wheel_speed.rear_right) * dt;
708 }
709
710 const double steer = vehicle_model_snapshot_cache_.steering_angle;
711
712 const double front_axle_x = car_params->wheelbase - car_params->cg_2_rear_axis + long_offset;
713 const double rear_axle_x = -car_params->cg_2_rear_axis + long_offset;
714 const double half_track = car_params->track_width * 0.5;
715
716 const double local_x[4] = {front_axle_x, front_axle_x, rear_axle_x, rear_axle_x};
717 const double local_y[4] = {half_track, -half_track, half_track, -half_track};
718 const double steer_angles[4] = {steer, steer, 0.0, 0.0};
719 const double spins[4] = {wheel_spin_fl_, wheel_spin_fr_, wheel_spin_rl_, wheel_spin_rr_};
720
721 for (int i = 0; i < 4; ++i) {
722 // Rotation logic in the local frame:
723 tf2::Quaternion q_steer;
724 q_steer.setRPY(0.0, 0.0, steer_angles[i]);
725
726 tf2::Quaternion q_spin;
727 q_spin.setRPY(0.0, spins[i], 0.0);
728
729 tf2::Quaternion q_mesh_offset;
730 q_mesh_offset.setRPY(-M_PI_2, 0.0, 0.0);
731
732 tf2::Quaternion q_side_offset;
733 q_side_offset.setRPY(0.0, 0.0, (i == 1 || i == 3) ? M_PI : 0.0);
734
735 // Order: apply steering, then wheel spin, then mesh corrections
736 tf2::Quaternion q_wheel = q_steer * q_spin * q_side_offset * q_mesh_offset;
737 q_wheel.normalize();
738
739 visualization_msgs::msg::Marker wheel;
740 wheel.header.stamp = stamp;
741 wheel.frame_locked = true;
742 wheel.header.frame_id = "car"; // Locked to the moving car frame
743 wheel.ns = "invictasim_vehicle";
744 wheel.id = i + 1;
745 wheel.type = visualization_msgs::msg::Marker::MESH_RESOURCE;
746 wheel.action = visualization_msgs::msg::Marker::ADD;
747
748 // Use pure local offsets
749 wheel.pose.position.x = local_x[i];
750 wheel.pose.position.y = local_y[i];
751 wheel.pose.position.z = wheel_center_z;
752
753 wheel.pose.orientation.x = q_wheel.x();
754 wheel.pose.orientation.y = q_wheel.y();
755 wheel.pose.orientation.z = q_wheel.z();
756 wheel.pose.orientation.w = q_wheel.w();
757
758 wheel.scale.x = 0.01;
759 wheel.scale.y = 0.01;
760 wheel.scale.z = 0.01;
761 wheel.color.a = 1.0f;
762 wheel.color.r = 0.08f;
763 wheel.color.g = 0.08f;
764 wheel.color.b = 0.08f;
765 wheel.mesh_resource = "package://invictasim/resources/meshes/tire.stl";
766 wheel.mesh_use_embedded_materials = false;
767
768 marker_array.markers.push_back(wheel);
769 }
770}
