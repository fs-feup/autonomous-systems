242void RosOutputAdapter::register_pub_helper(const std::string& topic,
243 std::function<void()> refresh_snapshot,
244 std::function<void(const rclcpp::Time&)> func) {
245 if (topic_frequencies_.count(topic) && topic_frequencies_[topic] > 0) {
246 const int frequency = topic_frequencies_[topic];
247 frequency_callbacks_[frequency].push_back([refresh_snapshot, func](const rclcpp::Time& stamp) {
248 refresh_snapshot();
249 func(stamp);
250 });
251 }

254void RosOutputAdapter::setup_timers() {
255 for (const auto& [frequency_hz, callbacks] : frequency_callbacks_) {
256 auto period = std::chrono::duration_cast<std::chrono::nanoseconds>(
257 std::chrono::duration<double>(1.0 / static_cast<double>(frequency_hz)));
258
259 frequency_timers_[frequency_hz] = this->create_wall_timer(period, [this, frequency_hz]() {
260 if (!running_) return;
261 on_frequency_tick(frequency_hz);
262 });
263 }

833visualization_msgs::msg::MarkerArray RosOutputAdapter::convert_cone_array_to_markers(
834 const std::vector<common_lib::structures::Cone>& cone_array, const rclcpp::Time& stamp,
835 const std::string& frame_id) const {
836 visualization_msgs::msg::MarkerArray marker_array;
837 int cone_id = 0;
838 static const std::string cone_mesh_path = "package://invictasim/resources/meshes/cones/";
839 for (const auto& cone : cone_array) {
840 visualization_msgs::msg::Marker m;
841 m.header.stamp = stamp;
842 m.header.frame_id = frame_id;
843 if (frame_id == "car") {
844 m.frame_locked = true;
845 m.lifetime = rclcpp::Duration::from_seconds(0.1);
846 }
847 m.ns = "cones";
848 const int current_cone_id = cone_id++;
849 m.id = current_cone_id;
850 m.type = visualization_msgs::msg::Marker::MESH_RESOURCE;
851 m.action = visualization_msgs::msg::Marker::ADD;
852
853 m.pose.position.x = cone.position.x;
854 m.pose.position.y = cone.position.y;
855 m.pose.position.z = 0.1425;
856
857 m.pose.orientation.w = 1.0;
858 m.scale.x = 1.0;
859 m.scale.y = 1.0;
860 m.scale.z = 1.0;
861 m.mesh_use_embedded_materials = true;
862 m.color.r = 1.0f;
863 m.color.g = 1.0f;
864 m.color.b = 1.0f;
865 m.color.a = 1.0f;
866
867 switch (cone.color) {
868 case common_lib::competition_logic::Color::BLUE:
869 m.mesh_resource = cone_mesh_path + "cone_blue.dae";
870 break;
871 case common_lib::competition_logic::Color::YELLOW:
872 m.mesh_resource = cone_mesh_path + "cone_yellow.dae";
873 break;
874 case common_lib::competition_logic::Color::LARGE_ORANGE:
875 m.mesh_resource = cone_mesh_path + "cone_orange_big.dae";
876 m.pose.position.z = 0.03; // Adjusted height for large cone
877 break;
878 case common_lib::competition_logic::Color::RED:
879 m.mesh_resource = cone_mesh_path + "cone_red.dae";
880 break;
881 case common_lib::competition_logic::Color::GREEN:
882 m.mesh_resource = cone_mesh_path + "cone_green.dae";
883 break;
884 case common_lib::competition_logic::Color::ORANGE:
885 default:
886 m.mesh_resource = cone_mesh_path + "cone_orange.dae";
887 break;
888 }
889 marker_array.markers.push_back(m);
890 if (visualize_cone_hitboxes_) {
891 const bool is_large =
892 cone.is_large || cone.color == common_lib::competition_logic::Color::LARGE_ORANGE;
893 const double radius = is_large ? cone_large_radius_ : cone_standard_radius_;
894 visualization_msgs::msg::Marker hitbox = cone_hitbox_marker_template_;
895 hitbox.header = m.header;
896 hitbox.frame_locked = m.frame_locked;
897 hitbox.lifetime = m.lifetime;
898 hitbox.id = 10000 + current_cone_id;
899 hitbox.pose.position.x = cone.position.x;
900 hitbox.pose.position.y = cone.position.y;
901 hitbox.scale.x = radius * 2.0;
902 hitbox.scale.y = radius * 2.0;
903 marker_array.markers.push_back(hitbox);
904 }
905 }
906 return marker_array;
907}

1165std::array<double, 2> RosOutputAdapter::track_segment_outward_normal(
1166 const std::vector<common_lib::structures::Cone>& side_cones,
1167 const std::vector<common_lib::structures::Cone>& opposite_cones, std::size_t start_i,
1168 std::size_t end_i) const {
1169 const auto& start = side_cones[start_i];
1170 const auto& end = side_cones[end_i];
1171 const double dx = end.position.x - start.position.x;
1172 const double dy = end.position.y - start.position.y;
1173 const double length = std::hypot(dx, dy);
1174 if (length <= std::numeric_limits<double>::epsilon()) {
1175 return {0.0, 0.0};
1176 }
1177
1178 std::array<double, 2> normal{-dy / length, dx / length};
1179 const double mid_x = 0.5 * (start.position.x + end.position.x);
1180 const double mid_y = 0.5 * (start.position.y + end.position.y);
1181 double closest_distance_squared = std::numeric_limits<double>::max();
1182 std::size_t closest_opposite_i = 0;
1183 for (std::size_t i = 0; i < opposite_cones.size(); ++i) {
1184 const double x = opposite_cones[i].position.x - mid_x;
1185 const double y = opposite_cones[i].position.y - mid_y;
1186 const double distance_squared = x * x + y * y;
1187 if (distance_squared < closest_distance_squared) {
1188 closest_distance_squared = distance_squared;
1189 closest_opposite_i = i;
1190 }
1191 }
1192
1193 const double to_opposite_x = opposite_cones[closest_opposite_i].position.x - mid_x;
1194 const double to_opposite_y = opposite_cones[closest_opposite_i].position.y - mid_y;
1195 if (normal[0] * to_opposite_x + normal[1] * to_opposite_y > 0.0) {
1196 normal[0] = -normal[0];
1197 normal[1] = -normal[1];
1198 }
1199 return normal;
1200}

1240RosOutputAdapter::TrackRibbon RosOutputAdapter::build_track_ribbon(
1241 const std::vector<common_lib::structures::Cone>& side_cones,
1242 const std::vector<common_lib::structures::Cone>& opposite_cones) const {
1243 TrackRibbon ribbon;
1244 ribbon.inner.resize(side_cones.size());
1245 ribbon.outer.resize(side_cones.size());
1246 ribbon.closed = side_cones.back().position.euclidean_distance(side_cones.front().position) <=
1247 track_surface_max_loop_segment_length_;
1248
1249 for (std::size_t i = 0; i < side_cones.size(); ++i) {
1250 const bool has_previous = ribbon.closed || i > 0;
1251 const bool has_next = ribbon.closed || i + 1 < side_cones.size();
1252 std::array<double, 2> normal{0.0, 0.0};
1253 if (has_previous) {
1254 const std::size_t previous_i = i == 0 ? side_cones.size() - 1 : i - 1;
1255 const auto previous_normal =
1256 track_segment_outward_normal(side_cones, opposite_cones, previous_i, i);
1257 normal[0] += previous_normal[0];
1258 normal[1] += previous_normal[1];
1259 }
1260 if (has_next) {
1261 const std::size_t next_i = i + 1 == side_cones.size() ? 0 : i + 1;
1262 const auto next_normal = track_segment_outward_normal(side_cones, opposite_cones, i, next_i);
1263 normal[0] += next_normal[0];
1264 normal[1] += next_normal[1];
1265 }
1266
1267 const double normal_length = std::hypot(normal[0], normal[1]);
1268 if (normal_length <= std::numeric_limits<double>::epsilon()) {
1269 continue;
1270 }
1271 normal[0] /= normal_length;
1272 normal[1] /= normal_length;
1273
1274 const auto& cone = side_cones[i];
1275 ribbon.inner[i] = make_track_point(cone.position.x + normal[0] * track_surface_cone_clearance_,
1276 cone.position.y + normal[1] * track_surface_cone_clearance_);
1277 ribbon.outer[i] = make_track_point(
1278 cone.position.x + normal[0] * (track_surface_cone_clearance_ + track_surface_curb_width_),
1279 cone.position.y + normal[1] * (track_surface_cone_clearance_ + track_surface_curb_width_));
1280 }
1281 return ribbon;
1282}

1284double RosOutputAdapter::track_path_length(const std::vector<geometry_msgs::msg::Point>& points,
1285 bool closed) const {
1286 double length = 0.0;
1287 const std::size_t segment_count = closed ? points.size() : points.size() - 1;

1289 const std::size_t next_i = (i + 1) % points.size();

1291 }

1295geometry_msgs::msg::Point RosOutputAdapter::sample_track_path(
1296 const std::vector<geometry_msgs::msg::Point>& points, bool closed,
1297 double target_distance) const {
1298 const std::size_t segment_count = closed ? points.size() : points.size() - 1;
1299 double distance = std::max(0.0, target_distance);
1300 for (std::size_t i = 0; i < segment_count; ++i) {
1301 const std::size_t next_i = (i + 1) % points.size();
1302 const double segment_length =
1303 std::hypot(points[next_i].x - points[i].x, points[next_i].y - points[i].y);
1304 if (segment_length <= std::numeric_limits<double>::epsilon()) {
1305 continue;
1306 }
1307 if (distance <= segment_length || i + 1 == segment_count) {
1308 const double t = std::clamp(distance / segment_length, 0.0, 1.0);
1309 return make_track_point(points[i].x + (points[next_i].x - points[i].x) * t,
1310 points[i].y + (points[next_i].y - points[i].y) * t);
1311 }
1312 distance -= segment_length;
1313 }
1314 return points.back();
1315}

1317geometry_msgs::msg::Point RosOutputAdapter::closest_point_on_track_path(
1318 const std::vector<geometry_msgs::msg::Point>& points, bool closed,
1319 const geometry_msgs::msg::Point& query) const {
1320 geometry_msgs::msg::Point closest = points.front();
1321 double closest_distance_squared = std::numeric_limits<double>::max();
1322 const std::size_t segment_count = closed ? points.size() : points.size() - 1;
1323 for (std::size_t i = 0; i < segment_count; ++i) {
1324 const std::size_t next_i = (i + 1) % points.size();
1325 const double dx = points[next_i].x - points[i].x;
1326 const double dy = points[next_i].y - points[i].y;
1327 const double length_squared = dx * dx + dy * dy;
1328 if (length_squared <= std::numeric_limits<double>::epsilon()) {
1329 continue;
1330 }
1331
1332 const double query_dx = query.x - points[i].x;
1333 const double query_dy = query.y - points[i].y;
1334 const double t = std::clamp((query_dx * dx + query_dy * dy) / length_squared, 0.0, 1.0);
1335 const geometry_msgs::msg::Point candidate =
1336 make_track_point(points[i].x + dx * t, points[i].y + dy * t);
1337 const double distance_x = candidate.x - query.x;
1338 const double distance_y = candidate.y - query.y;
1339 const double distance_squared = distance_x * distance_x + distance_y * distance_y;
1340 if (distance_squared < closest_distance_squared) {
1341 closest = candidate;
1342 closest_distance_squared = distance_squared;
1343 }
1344 }
1345 return closest;
1346}

1388void RosOutputAdapter::add_track_asphalt_from_ribbon(
1389 visualization_msgs::msg::Marker& asphalt_marker, const TrackRibbon& source_ribbon,
1390 const TrackRibbon& target_ribbon, const std_msgs::msg::ColorRGBA& asphalt_color,
1391 bool reverse_winding) const {
1392 constexpr double asphalt_sample_length = 2.0;
1393 const double source_length = track_path_length(source_ribbon.inner, source_ribbon.closed);
1394 const int segment_count =
1395 std::max(1, static_cast<int>(std::ceil(source_length / asphalt_sample_length)));
1396 std::vector<AsphaltSample> samples;
1397 samples.reserve(static_cast<std::size_t>(segment_count) + 1);
1398 std::vector<double> widths;
1399 widths.reserve(static_cast<std::size_t>(segment_count) + 1);
1400 for (int sample = 0; sample <= segment_count; ++sample) {
1401 const double distance =
1402 source_length * static_cast<double>(sample) / static_cast<double>(segment_count);
1403 AsphaltSample asphalt_sample;
1404 asphalt_sample.source = sample_track_path(source_ribbon.inner, source_ribbon.closed, distance);
1405 asphalt_sample.target = closest_point_on_track_path(target_ribbon.inner, target_ribbon.closed,
1406 asphalt_sample.source);
1407 asphalt_sample.width = std::hypot(asphalt_sample.target.x - asphalt_sample.source.x,
1408 asphalt_sample.target.y - asphalt_sample.source.y);
1409 widths.push_back(asphalt_sample.width);
1410 samples.push_back(asphalt_sample);
1411 }
1412
1413 std::nth_element(widths.begin(), widths.begin() + static_cast<long>(widths.size() / 2),
1414 widths.end());
1415 const double max_width = widths[widths.size() / 2] * 1.5;
1416 for (int segment = 0; segment < segment_count; ++segment) {
1417 const auto& start = samples[static_cast<std::size_t>(segment)];
1418 const auto& end = samples[static_cast<std::size_t>(segment + 1)];
1419 const double target_step =
1420 std::hypot(end.target.x - start.target.x, end.target.y - start.target.y);
1421 if (start.width > max_width || end.width > max_width ||
1422 target_step > track_surface_max_boundary_segment_length_) {
1423 continue;
1424 }
1425
1426 if (reverse_winding) {
1427 add_track_quad(asphalt_marker, start.target, end.target, end.source, start.source,
1428 asphalt_color);
1429 } else {
1430 add_track_quad(asphalt_marker, start.source, end.source, end.target, start.target,
1431 asphalt_color);
1432 }
1433 }

1514 const rclcpp::Time& stamp, double dt) {

1517 wheel_spin_fl_ += (wheel_speed.front_left) * dt;
1518 wheel_spin_fr_ += (wheel_speed.front_right) * dt;
1519 wheel_spin_rl_ += (wheel_speed.rear_left) * dt;
1520 wheel_spin_rr_ += (wheel_speed.rear_right) * dt;

1524 const double steer_angles[4] = {steer, steer, 0.0, 0.0};

1527 for (int i = 0; i < 4; ++i) {
1528 // Rotation logic in the local frame:
