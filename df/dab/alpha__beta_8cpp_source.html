3double AlphaBeta::error_distance(custom_interfaces::msg::PathPointArray& msg, const custom_interfaces::msg::VehicleStateVector& vehicle_state) const {
4 if (msg.pathpoint_array.size() < 2) {
5 return 0.0;
6 }
7
8 const auto& p0 = msg.pathpoint_array[0];
9 const auto& p1 = msg.pathpoint_array[1];
10
11 const double sx = p1.x - p0.x;
12 const double sy = p1.y - p0.y;
13 const double vx = vehicle_state.x - p0.x;
14 const double vy = vehicle_state.y - p0.y;
15
16 const double seg_norm = std::sqrt(sx * sx + sy * sy);
17 if (seg_norm < 1e-9) {
18 return 0.0;
19 }
20
21 // Signed lateral distance to the segment's supporting line.
22 // Positive: vehicle is to the left of segment direction p0 -> p1.
23 return std::fabs((sx * vy - sy * vx) / seg_norm);
24}

30unsigned int AlphaBeta::number_of_points(custom_interfaces::msg::PathPointArray& msg, double convergence_distance) const {
31 double distance_along_path = 0.0;
32 unsigned int points = 1;
33
34 for (std::size_t i = 1; i < msg.pathpoint_array.size(); ++i) {
35 const auto& prev = msg.pathpoint_array[i - 1];
36 const auto& current = msg.pathpoint_array[i];
37
38 const double dx = current.x - prev.x;
39 const double dy = current.y - prev.y;
40 distance_along_path += std::sqrt(dx * dx + dy * dy);
41 ++points;
42
43 if (distance_along_path > convergence_distance) {
44 return points;
45 }

55void AlphaBeta::create_local_path(custom_interfaces::msg::PathPointArray& path_msg, const custom_interfaces::msg::VehicleStateVector& vehicle_state) {
56 if (path_msg.pathpoint_array.empty()) {
57 return;
58 }
59
60 auto& points = path_msg.pathpoint_array;
61
62 // Rejoin length scales with the lateral error, so large errors get room to stay feasible.
63 const double error_dist = error_distance(path_msg, vehicle_state);
64 const double conv_dist = convergence_distance(error_dist);
65 const unsigned int num_points = number_of_points(path_msg, conv_dist);
66 const unsigned int path_points_local = std::min((unsigned int)points.size(), num_points);
67
68 // Car boundary state (hard requirement: the local path must start here exactly).
69 const double x0 = vehicle_state.x;
70 const double y0 = vehicle_state.y;
71 const double yaw0 = vehicle_state.orientation;
72 const double v0 = vehicle_state.velocity_x;
73
74 // Not enough room to shape a spline: just pin the first point to the car.
75 if (path_points_local < 3) {
76 points[0].x = x0;
77 points[0].y = y0;
78 points[0].orientation = yaw0;
79 points[0].v = v0;
80 return;
81 }
82
83 // Rejoin point: the far boundary condition, so the local path arrives tangent to the global.
84 const unsigned int merge_idx = path_points_local - 1;
85 const double x1 = points[merge_idx].x;
86 const double y1 = points[merge_idx].y;
87 const double yaw1 = points[merge_idx].orientation;
88
89 // Cubic Bezier in Hermite form: matches position and heading at both ends, so the blend
90 // leaves along the car's yaw and arrives tangent to the global path. Handles ~1/3 of chord.
91 const double chord = std::hypot(x1 - x0, y1 - y0);
92 const double handle = chord / 3.0;
93 const double b0x = x0;
94 const double b0y = y0;
95 const double b1x = x0 + handle * std::cos(yaw0);
96 const double b1y = y0 + handle * std::sin(yaw0);
97 const double b2x = x1 - handle * std::cos(yaw1);
98 const double b2y = y1 - handle * std::sin(yaw1);
99 const double b3x = x1;
100 const double b3y = y1;
101
102 for (unsigned int i = 0; i <= merge_idx; ++i) {
103 const double t = static_cast<double>(i) / static_cast<double>(merge_idx);
104 const double u = 1.0 - t;
105
106 const double bx = u * u * u * b0x + 3 * u * u * t * b1x + 3 * u * t * t * b2x + t * t * t * b3x;
107 const double by = u * u * u * b0y + 3 * u * u * t * b1y + 3 * u * t * t * b2y + t * t * t * b3y;
108
109 // Bezier derivative -> tangent heading (starts at yaw0, ends at yaw1 by construction).
110 const double dx = 3 * u * u * (b1x - b0x) + 6 * u * t * (b2x - b1x) + 3 * t * t * (b3x - b2x);
111 const double dy = 3 * u * u * (b1y - b0y) + 6 * u * t * (b2y - b1y) + 3 * t * t * (b3y - b2y);
112
113 custom_interfaces::msg::PathPoint& p = points[i];
114 p.x = bx;
115 p.y = by;
116 if (dx * dx + dy * dy > 1e-12) {
117 p.orientation = std::atan2(dy, dx);
118 }
119
120 // Only the first point takes the car's velocity; blending the rest erases the speed error.
121 if (i == 0) {
122 p.v = v0;
123 }
124 }
125}
