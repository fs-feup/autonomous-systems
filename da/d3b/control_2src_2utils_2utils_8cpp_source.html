18std::tuple<Position, int, double> get_closest_point(
19 const std::vector<custom_interfaces::msg::PathPoint> &pathpoint_array, const Position& position) {
20 double min_distance = 1e9;
21 double closest_point_velocity = 0;
22 Position closest_point = Position();
23 Position aux_point = Position();
24 int closest_point_id = -1;
25 for (size_t i = 1; i < pathpoint_array.size(); i++) {
26 aux_point = Position(pathpoint_array[i].x, pathpoint_array[i].y);
27 double distance = position.euclidean_distance(aux_point);
28 if (distance < min_distance) {
29 min_distance = distance;
30 closest_point = aux_point;
31 closest_point_id = static_cast<int>(i);
32 closest_point_velocity = pathpoint_array[i].v;
33 }
34 }
35 return std::make_tuple(closest_point, closest_point_id, closest_point_velocity);
36}

39double get_interpolated_velocity(
40 const std::vector<custom_interfaces::msg::PathPoint> &pathpoint_array, int closest_point_id,
41 const Position &position) {
42 if (pathpoint_array.empty() || closest_point_id < 0) {
43 return 0.0;
44 }
45
46 const int last_id = static_cast<int>(pathpoint_array.size()) - 1;
47 const int closest_id = std::clamp(closest_point_id, 0, last_id);
48
49 if (last_id == 0) {
50 return pathpoint_array[0].v;
51 }
52
53 double best_distance = std::numeric_limits<double>::max();
54 double interpolated_velocity = pathpoint_array[closest_id].v;
55
56 // Both neighbours are tried: the closest point can sit either ahead of or behind the car.
57 const auto consider_segment = [&](int start_id, int end_id) {
58 const auto &start = pathpoint_array[start_id];
59 const auto &end = pathpoint_array[end_id];
60 const double segment_x = end.x - start.x;
61 const double segment_y = end.y - start.y;
62 const double segment_length_squared = segment_x * segment_x + segment_y * segment_y;
63
64 if (segment_length_squared <= 0.0) {
65 return;
66 }
67
68 double ratio = ((position.x - start.x) * segment_x + (position.y - start.y) * segment_y) /
69 segment_length_squared;
70 ratio = std::clamp(ratio, 0.0, 1.0);
71
72 const double distance = std::hypot(position.x - (start.x + ratio * segment_x),
73 position.y - (start.y + ratio * segment_y));
74 if (distance < best_distance) {
75 best_distance = distance;
76 interpolated_velocity = start.v + ratio * (end.v - start.v);
77 }
78 };
79
80 if (closest_id > 0) {
81 consider_segment(closest_id - 1, closest_id);
82 }
83 if (closest_id < last_id) {
84 consider_segment(closest_id, closest_id + 1);
85 }
86
87 return interpolated_velocity;
88}

90std::tuple<Position, double, bool> get_lookahead_point(
91 const std::vector<custom_interfaces::msg::PathPoint> &pathpoint_array,
92 int closest_point_id, double lookahead_distance, Position rear_axis_position, double last_to_first_max_dist) {
93
94 if (pathpoint_array.empty()) {
95 RCLCPP_DEBUG(rclcpp::get_logger("control"), "Empty path");
96 return std::make_tuple(Position(), 0.0, true);
97 }
98
99 // Clamp closest_point_id
100 closest_point_id = std::clamp(closest_point_id, 0, static_cast<int>(pathpoint_array.size()) - 1);
101
102 Position rear_axis_point = rear_axis_position;
103
104 for (size_t i = 0; i < pathpoint_array.size(); i++) {
105 size_t index_a = (static_cast<size_t>(closest_point_id) + i) % pathpoint_array.size();
106 size_t index_b = (static_cast<size_t>(closest_point_id) + i + 1) % pathpoint_array.size();
107
108 // We reached the end of the path
109 if (index_b < index_a) {
110 Position first_point(pathpoint_array.front().x, pathpoint_array.front().y);
111 Position last_point(pathpoint_array.back().x, pathpoint_array.back().y);
112 double start_end_distance = first_point.euclidean_distance(last_point);
113
114 // If the path is not a closed track, force last point as lookahead
115 // If the closest point is the first point, then the car is before the path start, so return first point, else return the last point
116 if (start_end_distance > last_to_first_max_dist) {
117 if (closest_point_id == 1) {
118 // Car might be before the first point, return it as lookahead
119 const auto &cp = pathpoint_array[static_cast<size_t>(closest_point_id)];
120 return std::make_tuple(Position(cp.x, cp.y), cp.v, false);
121 } else {
122 RCLCPP_DEBUG(rclcpp::get_logger("control"),
123 "Lookahead extends beyond path end and it is not a closed track, using last "
124 "point of the path as lookahead point");
125 return std::make_tuple(Position(pathpoint_array.back().x, pathpoint_array.back().y),
126 pathpoint_array.back().v, false);
127 }
128 }
129 }
130
131 auto point_a = Position(pathpoint_array[index_a].x, pathpoint_array[index_a].y);
132 auto point_b = Position(pathpoint_array[index_b].x, pathpoint_array[index_b].y);
133
134 if (!(rear_axis_point.euclidean_distance(point_a) < lookahead_distance &&
135 rear_axis_point.euclidean_distance(point_b) > lookahead_distance)) {
136 continue;
137 }
138
139 double result_x;
140 double result_y;
141
142 // Slope of the line is infinite, don't need line equation
143 if (point_a.x == point_b.x) {
144 RCLCPP_DEBUG(rclcpp::get_logger("control"), "Vertical line!!");
145 result_x = point_a.x;
146 double delta = lookahead_distance * lookahead_distance - std::pow(result_x - rear_axis_point.x, 2);
147 if (delta < 0) {
148 continue;
149 }
150 double y1 = rear_axis_point.y + std::sqrt(delta);
151 double y2 = rear_axis_point.y - std::sqrt(delta);
152
153 if (y1 >= std::min(point_a.y, point_b.y) && y1 <= std::max(point_a.y, point_b.y)) {
154 result_y = y1;
155 } else {
156 result_y = y2;
157 }
158
159 } else {
160 // y = mx + c
161 double m = (point_b.y - point_a.y) / (point_b.x - point_a.x);
162 double c = point_a.y - m * point_a.x;
163
164 // (x - x0)^2 + (y - y0)^2 = r^2
165 // with x0 = rear_axis_point.x, y0 = rear_axis_point.y, r = lookahead_distance
166
167 // with this information we can find the intersection point between the circle and the line
168 // and obtain the lookahead point.
169 // Expanded form of the circle equation with y substituted by mx + c gives a quadratic
170 // equation:
171 double A = 1 + m * m;
172 double B = 2 * (m * c - rear_axis_point.x - m * rear_axis_point.y);
173 double C = rear_axis_point.x * rear_axis_point.x + c * c +
174 rear_axis_point.y * rear_axis_point.y - 2 * c * rear_axis_point.y - lookahead_distance * lookahead_distance;
175
176 double delta = B * B - 4 * A * C;
177
178 if (delta < 0) {
179 continue;
180 }
181
182 double x1 = (-B + std::sqrt(delta)) / (2 * A);
183 double x2 = (-B - std::sqrt(delta)) / (2 * A);
184
185 double y1 = m * x1 + c;
186 double y2 = m * x2 + c;
187
188 // solution (x,y) which is within bounds of point_a and point_b
189 if (x1 >= std::min(point_a.x, point_b.x) && x1 <= std::max(point_a.x, point_b.x) &&
190 y1 >= std::min(point_a.y, point_b.y) && y1 <= std::max(point_a.y, point_b.y)) {
191 result_x = x1;
192 result_y = y1;
193 } else {
194 result_x = x2;
195 result_y = y2;
196 }
197 }
198
199 return std::make_tuple(Position(result_x, result_y),
200 (pathpoint_array[index_a].v + pathpoint_array[index_b].v) / 2.0, false);
201 }
202 RCLCPP_DEBUG(rclcpp::get_logger("control"), "No lookahead point found");
203 return std::make_tuple(Position(), 0.0, true);
204}
