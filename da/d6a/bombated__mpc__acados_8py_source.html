87def export_mpc_model() -> AcadosModel:
88 model = AcadosModel()
89 model.name = "bombated_mpc"
90 model.p = SX.sym("p", 4)
91
92 _, _, wheel_speed_scale = load_mpc_parameters()
93
94 x = SX.sym("x", 13)
95 xdot = SX.sym("xdot", 13)
96 u = SX.sym("u", 3)
97
98 # Steering angle at wheels (ackermann geometry). The sign of tan(state[8]) naturally adds or subtracts the track width!
99 sa_fl = atan(L * tan(x[8]) / (L - s * (1 + ackermann_deviation) * tan(x[8])))
100 sa_fr = atan(L * tan(x[8]) / (L + s * (1 + ackermann_deviation) * tan(x[8])))
101 sa_rl = 0
102 sa_rr = 0
103
104 # Longitudinal velocity of each wheel in the car's frame
105 v_fl_x = x[3] - x[5] * sf / 2
106 v_fr_x = x[3] + x[5] * sf / 2
107 v_rl_x = x[3] - x[5] * sr / 2
108 v_rr_x = x[3] + x[5] * sr / 2
109
110 # Lateral velocity of each wheel in the car's frame
111 v_fl_y = x[4] + x[5] * lf
112 v_fr_y = x[4] + x[5] * lf
113 v_rl_y = x[4] - x[5] * lr
114 v_rr_y = x[4] - x[5] * lr
115
116 # Smooth regularization factor (prevents division by zero)
117 eps_vx = 0.5
118
119 # Regularized longitudinal velocities (strictly positive)
120 # Breaks lateral dynamics if velocity is negative
121 v_fl_x_reg = sqrt(v_fl_x**2 + eps_vx**2)
122 v_fr_x_reg = sqrt(v_fr_x**2 + eps_vx**2)
123 v_rl_x_reg = sqrt(v_rl_x**2 + eps_vx**2)
124 v_rr_x_reg = sqrt(v_rr_x**2 + eps_vx**2)
125
126 # Slip angles at each wheel (No if_else needed)
127 slip_angle_fl = sa_fl - atan2(v_fl_y, v_fl_x_reg)
128 slip_angle_fr = sa_fr - atan2(v_fr_y, v_fr_x_reg)
129 slip_angle_rl = sa_rl - atan2(v_rl_y, v_rl_x_reg)
130 slip_angle_rr = sa_rr - atan2(v_rr_y, v_rr_x_reg)
131
132 # Longitudinal velocity at the frame of each wheel
133 vx_fl = cos(sa_fl) * (v_fl_x) + sin(sa_fl) * (v_fl_y)
134 vx_fr = cos(sa_fr) * (v_fr_x) + sin(sa_fr) * (v_fr_y)
135 vx_rl = cos(sa_rl) * (v_rl_x) + sin(sa_rl) * (v_rl_y)
136 vx_rr = cos(sa_rr) * (v_rr_x) + sin(sa_rr) * (v_rr_y)
137
138 # Regularized wheel-frame longitudinal velocities
139 vx_fl_reg = sqrt(vx_fl**2 + eps_vx**2)
140 vx_fr_reg = sqrt(vx_fr**2 + eps_vx**2)
141 vx_rl_reg = sqrt(vx_rl**2 + eps_vx**2)
142 vx_rr_reg = sqrt(vx_rr**2 + eps_vx**2)
143
144 # Slip ratios (No if_else needed)
145 slip_ratio_fl = (wheel_radius * wheel_speed_scale * x[9] - vx_fl) / vx_fl_reg
146 slip_ratio_fr = (wheel_radius * wheel_speed_scale * x[10] - vx_fr) / vx_fr_reg
147 slip_ratio_rl = (wheel_radius * wheel_speed_scale * x[11] - vx_rl) / vx_rl_reg
148 slip_ratio_rr = (wheel_radius * wheel_speed_scale * x[12] - vx_rr) / vx_rr_reg
149
150 # Aerodynamic forces
151 drag = 0.5 * air_density * drag_coefficient * frontal_area * x[3]**2
152 lift = 0.5 * air_density * lift_coefficient * frontal_area * x[3]**2
153 downforce = -lift # Lift is negative downforce
154
155 # Downforce distribution using center of pressure location
156 d_force_front = downforce * (lr + dx_cop) / L
157 d_force_rear = downforce * (lf - dx_cop) / L
158
159 # Load transfer calculations
160 fx = mass * x[6]
161 fy = mass * x[7]
162 longitudinal_weight_transfer = cog_height * fx / L
163 total_roll_stiffness = roll_stiffness_front + roll_stiffness_rear
164 lateral_weight_transfer_front = (cog_height * fy / sf) * (roll_stiffness_front / total_roll_stiffness)
165 lateral_weight_transfer_rear = (cog_height * fy / sr) * (roll_stiffness_rear / total_roll_stiffness)
166
167 # Load transfer distribution
168 total_load_transfer_fl = - longitudinal_weight_transfer / 2 - lateral_weight_transfer_front
169 total_load_transfer_fr = - longitudinal_weight_transfer / 2 + lateral_weight_transfer_front
170 total_load_transfer_rl = longitudinal_weight_transfer / 2 - lateral_weight_transfer_rear
171 total_load_transfer_rr = longitudinal_weight_transfer / 2 + lateral_weight_transfer_rear
172
173 # Normal load on each tire
174 vertical_load_fl = mass * gravity_acceleration * lr / (L * 2) + d_force_front / 2 + total_load_transfer_fl
175 vertical_load_fr = mass * gravity_acceleration * lr / (L * 2) + d_force_front / 2 + total_load_transfer_fr
176 vertical_load_rl = mass * gravity_acceleration * lf / (L * 2) + d_force_rear / 2 + total_load_transfer_rl
177 vertical_load_rr = mass * gravity_acceleration * lf / (L * 2) + d_force_rear / 2 + total_load_transfer_rr
178
179 # Lateral tire forces
180 fy_fl = (tire_lateral_D * vertical_load_fl * sin(tire_lateral_C * atan(tire_lateral_B * slip_angle_fl - tire_lateral_E * (tire_lateral_B * slip_angle_fl - atan(tire_lateral_B * slip_angle_fl)))))
181 fy_fr = (tire_lateral_D * vertical_load_fr * sin(tire_lateral_C * atan(tire_lateral_B * slip_angle_fr - tire_lateral_E * (tire_lateral_B * slip_angle_fr - atan(tire_lateral_B * slip_angle_fr)))))
182 fy_rl = (tire_lateral_D * vertical_load_rl * sin(tire_lateral_C * atan(tire_lateral_B * slip_angle_rl - tire_lateral_E * (tire_lateral_B * slip_angle_rl - atan(tire_lateral_B * slip_angle_rl)))))
183 fy_rr = (tire_lateral_D * vertical_load_rr * sin(tire_lateral_C * atan(tire_lateral_B * slip_angle_rr - tire_lateral_E * (tire_lateral_B * slip_angle_rr - atan(tire_lateral_B * slip_angle_rr)))))
184
185 # Longitudinal tire forces
186 speed_blend = 0.5 * (tanh((x[3] - 5.0)) + 1.0)
187
188 linear_stiffness_fl = tire_linear_coefficient * vertical_load_fl
189 linear_stiffness_fr = tire_linear_coefficient * vertical_load_fr
190 linear_stiffness_rl = tire_linear_coefficient * vertical_load_rl
191 linear_stiffness_rr = tire_linear_coefficient * vertical_load_rr
192
193 pacejka_fx_fl = (tire_D * vertical_load_fl * sin(tire_C * atan(tire_B * slip_ratio_fl - tire_E * (tire_B * slip_ratio_fl - atan(tire_B * slip_ratio_fl)))))
194 pacejka_fx_fr = (tire_D * vertical_load_fr * sin(tire_C * atan(tire_B * slip_ratio_fr - tire_E * (tire_B * slip_ratio_fr - atan(tire_B * slip_ratio_fr)))))
195 pacejka_fx_rl = (tire_D * vertical_load_rl * sin(tire_C * atan(tire_B * slip_ratio_rl - tire_E * (tire_B * slip_ratio_rl - atan(tire_B * slip_ratio_rl)))))
196 pacejka_fx_rr = (tire_D * vertical_load_rr * sin(tire_C * atan(tire_B * slip_ratio_rr - tire_E * (tire_B * slip_ratio_rr - atan(tire_B * slip_ratio_rr)))))
197
198 atan_normalization = 2.0 / np.pi
199 rolling_resistance_fl = rolling_resistance_coefficient * vertical_load_fl * atan(x[9]) * atan_normalization
200 rolling_resistance_fr = rolling_resistance_coefficient * vertical_load_fr * atan(x[10]) * atan_normalization
201 rolling_resistance_rl = rolling_resistance_coefficient * vertical_load_rl * atan(x[11]) * atan_normalization
202 rolling_resistance_rr = rolling_resistance_coefficient * vertical_load_rr * atan(x[12]) * atan_normalization
203
204 fx_fl = (1.0 - speed_blend) * (linear_stiffness_fl * slip_ratio_fl) + (speed_blend) * pacejka_fx_fl - rolling_resistance_fl
205 fx_fr = (1.0 - speed_blend) * (linear_stiffness_fr * slip_ratio_fr) + (speed_blend) * pacejka_fx_fr - rolling_resistance_fr
206 fx_rl = (1.0 - speed_blend) * (linear_stiffness_rl * slip_ratio_rl) + (speed_blend) * pacejka_fx_rl - rolling_resistance_rl
207 fx_rr = (1.0 - speed_blend) * (linear_stiffness_rr * slip_ratio_rr) + (speed_blend) * pacejka_fx_rr - rolling_resistance_rr
208
209 # FX in the car's frame
210 fx_car_fl = - fy_fl * sin(sa_fl) + fx_fl * cos(sa_fl)
211 fx_car_fr = - fy_fr * sin(sa_fr) + fx_fr * cos(sa_fr)
212 fx_car_rl = - fy_rl * sin(sa_rl) + fx_rl * cos(sa_rl)
213 fx_car_rr = - fy_rr * sin(sa_rr) + fx_rr * cos(sa_rr)
214 acceleration_x = (fx_car_fl + fx_car_fr + fx_car_rl + fx_car_rr - drag) / mass
215
216 # FY in the car's frame
217 fy_car_fl = fy_fl * cos(sa_fl) + fx_fl * sin(sa_fl)
218 fy_car_fr = fy_fr * cos(sa_fr) + fx_fr * sin(sa_fr)
219 fy_car_rl = fy_rl * cos(sa_rl) + fx_rl * sin(sa_rl)
220 fy_car_rr = fy_rr * cos(sa_rr) + fx_rr * sin(sa_rr)
221 acceleration_y = (fy_car_fl + fy_car_fr + fy_car_rl + fy_car_rr) / mass
222
223 yaw_moment_fl = lf * (fy_car_fl) - s * (fx_car_fl)
224 yaw_moment_fr = lf * (fy_car_fr) + s * (fx_car_fr)
225 yaw_moment_rl = - lr * (fy_car_rl) - s * (fx_car_rl)
226 yaw_moment_rr = - lr * (fy_car_rr) + s * (fx_car_rr)
227
228 yaw_rate_dot = (yaw_moment_fl + yaw_moment_fr + yaw_moment_rl + yaw_moment_rr) / Izz
229 steering_dot = (u[2] - x[8]) / steering_motor_tau
230
231 # Dynamic torque limits
232 max_torque_fl = max_motor_torque
233 max_torque_fr = max_motor_torque
234 max_torque_rl = max_motor_torque
235 max_torque_rr = max_motor_torque
236
237 # Demanded torque at wheels
238 demanded_torque_fl = max_torque_fl * 0
239 demanded_torque_fr = max_torque_fr * 0
240 demanded_torque_rl = max_torque_rl * u[0]
241 demanded_torque_rr = max_torque_rr * u[1]
242
243 # Angular acceleration at wheels
244 alpha_fl = (demanded_torque_fl - wheel_radius * fx_fl) / wheel_rotational_inertia
245 alpha_fr = (demanded_torque_fr - wheel_radius * fx_fr) / wheel_rotational_inertia
246 alpha_rl = (demanded_torque_rl - wheel_radius * fx_rl) / wheel_rotational_inertia
247 alpha_rr = (demanded_torque_rr - wheel_radius * fx_rr) / wheel_rotational_inertia
248
249 # dynamics
250 f_expl = vertcat(
251 x[3] * cos(x[2]) - x[4] * sin(x[2]),
252 x[3] * sin(x[2]) + x[4] * cos(x[2]),
253 x[5],
254 acceleration_x + x[5] * x[4],
255 acceleration_y - x[5] * x[3],
256 yaw_rate_dot,
257 (acceleration_x - x[6]) / 10, # simplified dynamics
258 (acceleration_y - x[7]) / 10, # simplified dynamics
259 steering_dot,
260 alpha_fl / wheel_speed_scale,
261 alpha_fr / wheel_speed_scale,
262 alpha_rl / wheel_speed_scale,
263 alpha_rr / wheel_speed_scale,
264 )
265
266 model.x = x
267 model.xdot = xdot
268 model.u = u
269 model.f_expl_expr = f_expl
270 model.f_impl_expr = xdot - f_expl
271
272 model.con_h_expr_0 = u[0] - u[1]
273 model.con_h_expr = vertcat(
274 u[0] - u[1], # Index 0: Torque difference limit
275 slip_ratio_fl, # Index 1: Front-Left slip
276 slip_ratio_fr, # Index 2: Front-Right slip
277 slip_ratio_rl, # Index 3: Rear-Left slip
278 slip_ratio_rr # Index 4: Rear-Right slip
279 )
280 return model

282def setup_cost_function(ocp: AcadosOcp):
283 """
284 Sets up the cost function to track a SPECIFIC reference point 'p' per stage.
285 """
286 # 1. Extract symbolic variables
287 x = ocp.model.x
288 u = ocp.model.u
289 p = ocp.model.p
290
291 # 2. Configure Cost Type
292 ocp.cost.cost_type = "NONLINEAR_LS"
293 ocp.cost.cost_type_e = "NONLINEAR_LS"
294
295 # 3. Define the Residuals
296 psi = x[2]
297 ref_theta = p[3]
298 theta_cost_term = sin(psi - ref_theta)
299
300 # --- FIX: Add 'u' to the residuals ---
301 # We want u[0] (throttle), u[1] (brake), u[2] (steer) to ideally be 0 (or low)
302 # unless necessary to minimize the state error.
303 cost_expression = vertcat(
304 x[0] - p[0], # X Position Error
305 x[1] - p[1], # Y Position Error
306 x[3] - p[2], # Velocity Error
307 theta_cost_term, # Orientation Error
308 u[0], # Penalty on Throttle
309 u[1], # Penalty on Brake
310 u[2] # Penalty on Steering
311 )
312
313 # Terminal Residual (No controls at the last step)
314 cost_expression_e = vertcat(
315 x[0] - p[0],
316 x[1] - p[1],
317 x[3] - p[2],
318 theta_cost_term
319 )
320
321 ocp.model.cost_y_expr = cost_expression
322 ocp.model.cost_y_expr_e = cost_expression_e
323
324 # 4. Define Weight Matrices (W)
325 # Weights: [X, Y, V, Theta, Throttle, Steer]
326 # Note: High weight on Steer (e.g. 5.0 or 10.0) prevents shaky steering
327 weights = np.array([2.0, 2.0, 4.0, 1.0, 0.2, 0.2, 10.0])
328
329 # Terminal weights (Only 4 states)
330 weights_e = np.array([2.0, 2.0, 1.0, 1.0])
331
332 ocp.cost.W = np.diag(weights)
333 ocp.cost.W_e = np.diag(weights_e)
334
335 # 5. Set Numeric References to Zero
336 # ny = 7 (4 states + 3 controls)
337 ocp.cost.yref = np.zeros(7)
338 ocp.cost.yref_e = np.zeros(4)
339
340 # IMPORTANT: Update Dimensions
341 ocp.dims.ny = 7
342 ocp.dims.ny_e = 4

345def create_ocp_solver(gen_base_dir: str = "./build/acados", acados_dir: str | None = None, acados_lib_dir: str | None = None):
346 # gen_base_dir should point to a folder that will contain the generated
347 # acados artifacts. We place the C sources under <gen_base_dir>/c_generated_code
348 # and the json model at <gen_base_dir>/acados_ocp_mpc.json
349 c_code_dir = os.path.abspath(os.path.join(gen_base_dir, "c_generated_code"))
350 json_path = os.path.abspath(os.path.join(gen_base_dir, "acados_ocp_mpc.json"))
351
352 if os.path.exists(c_code_dir):
353 shutil.rmtree(c_code_dir)
354 if os.path.exists(json_path):
355 os.remove(json_path)
356
357 os.makedirs(os.path.dirname(c_code_dir), exist_ok=True)
358 os.makedirs(os.path.dirname(json_path), exist_ok=True)
359
360 prediction_horizon_seconds, prediction_horizon_steps, _ = load_mpc_parameters()
361
362 if acados_dir is not None:
363 ocp = AcadosOcp(acados_path=acados_dir, acados_lib_path=acados_lib_dir)
364 else:
365 ocp = AcadosOcp()
366 ocp.model = export_mpc_model()
367 ocp.code_export_directory = c_code_dir
368
369 ocp.solver_options.N_horizon = prediction_horizon_steps
370 ocp.solver_options.tf = prediction_horizon_seconds
371 ocp.solver_options.nlp_solver_type = "SQP"
372 ocp.solver_options.qp_solver_cond_N = 10
373 ocp.solver_options.integrator_type = "IRK"
374 ocp.solver_options.levenberg_marquardt = 1e-2
375 ocp.solver_options.with_batch_functionality = False
376 ocp.solver_options.ext_fun_compile_flags = "-O3 -march=native -ffast-math"
377 ocp.solver_options.nlp_solver_max_iter = 10
378 ocp.solver_options.with_batch_functionality = True
379 ocp.solver_options.num_threads_in_batch_ext_fun = 4
380
381 ocp.solver_options.nlp_solver_tol_stat = 1e-2
382 ocp.solver_options.nlp_solver_tol_eq = 1e-2
383
384 # Initial state constraint (required for set_state logic)
385 ocp.constraints.x0 = np.zeros(13)
386 ocp.parameter_values = np.zeros(4)
387
388 setup_cost_function(ocp)
389
390 # Lower bounds for controls
391 u_min = np.array([-1.0, -1.0, -0.335]) # adjust these values as needed
392
393 # Upper bounds for controls
394 u_max = np.array([1.0, 1.0, 0.335]) # adjust these values as needed
395
396 ocp.constraints.lbu = u_min # lower bound on u
397 ocp.constraints.ubu = u_max # upper bound on u
398 ocp.constraints.idxbu = np.array([0, 1, 2]) # which control inputs have bounds
399
400 max_diff = 0.1 # |u[0] - u[1]| <= max_diff
401 max_slip = 0.1367
402 ocp.constraints.lh = np.array([-max_diff, -max_slip, -max_slip, -max_slip, -max_slip]) # lower bound
403 ocp.constraints.uh = np.array([max_diff, max_slip, max_slip, max_slip, max_slip]) # upper bound
404 ocp.dims.nh = 5
405
406 ocp.constraints.lh_0 = np.array([-max_diff]) # lower bound
407 ocp.constraints.uh_0 = np.array([max_diff]) # upper bound
408 ocp.dims.nh_0 = 1
409
410 # Make the constraints soft
411 ocp.constraints.idxsh = np.array([1, 2, 3, 4])
412 ocp.dims.nsh = 4
413
414 # Define high penalties for the slack variables
415 slack_L1_penalty = 1e5
416 slack_L2_penalty = 1e5
417
418 ocp.cost.Zl = slack_L2_penalty * np.ones(4)
419 ocp.cost.Zu = slack_L2_penalty * np.ones(4)
420 ocp.cost.zl = slack_L1_penalty * np.ones(4)
421 ocp.cost.zu = slack_L1_penalty * np.ones(4)
422
423 try:
424 solver = AcadosOcpSolver(ocp, json_file=json_path)
425 return solver
426 except Exception as e:
427 print(f"FAIL: {e}")
428 return None
