15 Eigen::VectorXi& nearest_landmarks, Eigen::VectorXd& nearest_distances) const {
16 const int num_observations = observations.size() / 2;
17 const int num_landmarks = landmarks.size() / 2;
18
19 nearest_landmarks = Eigen::VectorXi::Constant(num_observations, -1);
20 nearest_distances =
21 Eigen::VectorXd::Constant(num_observations, std::numeric_limits<double>::infinity());
22
23 for (int observation_idx = 0; observation_idx < num_observations; ++observation_idx) {
24 if (observation_confidences(observation_idx) < this->_params_.new_landmark_confidence_gate) {
25 continue;
26 }
27
28 if (const double observation_distance_to_pose =
29 std::hypot(observations(2 * observation_idx) - pose(0),
30 observations(2 * observation_idx + 1) - pose(1));
31 observation_distance_to_pose > this->_params_.seed_radius) {
32 continue;
33 }
34
35 for (int landmark_idx = 0; landmark_idx < num_landmarks; ++landmark_idx) {
36 const double distance =
37 std::hypot(observations(2 * observation_idx) - landmarks(2 * landmark_idx),
38 observations(2 * observation_idx + 1) - landmarks(2 * landmark_idx + 1));
39
40 if (distance < nearest_distances(observation_idx)) {
41 nearest_distances(observation_idx) = distance;
42 nearest_landmarks(observation_idx) = landmark_idx;

44 }

48Eigen::VectorXi AdaptiveNearestNeighbor::get_seed_observation_indices(
49 const Eigen::VectorXi& nearest_landmarks, const Eigen::VectorXd& nearest_distances) const {
50 const int max_seed_observations = std::max(2, this->_params_.seed_count);
51 const int num_seed_slots =
52 std::min(static_cast<int>(nearest_landmarks.size()), max_seed_observations);
53 Eigen::VectorXi closest_observation_indices = Eigen::VectorXi::Constant(num_seed_slots, -1);
54 Eigen::VectorXd closest_distances =
55 Eigen::VectorXd::Constant(num_seed_slots, std::numeric_limits<double>::infinity());
56
57 for (int observation_idx = 0; observation_idx < nearest_landmarks.size(); ++observation_idx) {
58 if (nearest_landmarks(observation_idx) == -1 ||
59 !std::isfinite(nearest_distances(observation_idx))) {
60 continue;
61 }
62
63 for (int insert_idx = 0; insert_idx < num_seed_slots; ++insert_idx) {
64 if (nearest_distances(observation_idx) >= closest_distances(insert_idx)) {
65 continue;
66 }
67
68 for (int shift_idx = num_seed_slots - 1; shift_idx > insert_idx; --shift_idx) {
69 closest_distances(shift_idx) = closest_distances(shift_idx - 1);
70 closest_observation_indices(shift_idx) = closest_observation_indices(shift_idx - 1);
71 }
72
73 closest_distances(insert_idx) = nearest_distances(observation_idx);
74 closest_observation_indices(insert_idx) = observation_idx;
75 break;
76 }
77 }
78
79 int num_seed_observations = 0;
80 while (num_seed_observations < closest_observation_indices.size() &&
81 closest_observation_indices(num_seed_observations) != -1) {
82 ++num_seed_observations;
83 }
84
85 Eigen::VectorXi seed_observation_indices(num_seed_observations);
86 for (int seed_idx = 0; seed_idx < num_seed_observations; ++seed_idx) {
87 seed_observation_indices(seed_idx) = closest_observation_indices(seed_idx);
88 }
89
90 return seed_observation_indices;
91}

93bool AdaptiveNearestNeighbor::estimate_transform_from_pair(
94 const Eigen::VectorXd& landmarks, const Eigen::VectorXd& observations,
95 int first_observation_idx, int first_landmark_idx, int second_observation_idx,
96 int second_landmark_idx, Eigen::Vector3d& transform) const {
97 if (first_observation_idx == second_observation_idx ||
98 first_landmark_idx == second_landmark_idx) {
99 return false;
100 }
101
102 const Eigen::Vector2d first_observation = observations.segment(2 * first_observation_idx, 2);
103 const Eigen::Vector2d second_observation = observations.segment(2 * second_observation_idx, 2);
104 const Eigen::Vector2d first_landmark = landmarks.segment(first_landmark_idx, 2);
105 const Eigen::Vector2d second_landmark = landmarks.segment(second_landmark_idx, 2);
106
107 const Eigen::Vector2d observation_delta = second_observation - first_observation;
108 const Eigen::Vector2d landmark_delta = second_landmark - first_landmark;
109
110 const double rotation =
111 common_lib::maths::normalize_angle(std::atan2(landmark_delta.y(), landmark_delta.x()) -
112 std::atan2(observation_delta.y(), observation_delta.x()));
113 const Eigen::Matrix2d rotation_matrix = common_lib::maths::get_rotation_matrix(rotation);
114 const Eigen::Vector2d translation = first_landmark - rotation_matrix * first_observation;
115
116 transform << translation.x(), translation.y(), rotation;
117 return true;
118}

120void AdaptiveNearestNeighbor::score_associations(const Eigen::VectorXd& landmarks,
121 const Eigen::VectorXd& observations,
122 const Eigen::VectorXi& associations,
123 int& num_associations,
124 double& total_distance) const {
125 num_associations = 0;
126 total_distance = 0.0;
127
128 for (int observation_idx = 0; observation_idx < associations.size(); ++observation_idx) {
129 if (associations(observation_idx) < 0) {
130 continue;
131 }
132
133 ++num_associations;
134 total_distance +=
135 std::hypot(observations(2 * observation_idx) - landmarks(associations(observation_idx)),
136 observations(2 * observation_idx + 1) -
137 landmarks(associations(observation_idx) + 1));
138 }

141Eigen::VectorXi AdaptiveNearestNeighbor::associate(const Eigen::VectorXd& landmarks,
142 const Eigen::VectorXd& observations,
143 const Eigen::MatrixXd& covariance,
144 const Eigen::VectorXd& observation_confidences,
145 const Eigen::Vector3d& pose) const {
146 NearestNeighbor nearest_neighbor(this->_params_);
147
148 Eigen::VectorXi best_associations =
149 nearest_neighbor.associate(landmarks, observations, covariance, observation_confidences, pose);
150 int best_num_associations = 0;
151 double best_total_distance = 0.0;
152 score_associations(landmarks, observations, best_associations, best_num_associations,
153 best_total_distance);
154
155 Eigen::VectorXi nearest_landmarks;
156 Eigen::VectorXd nearest_distances;
157 get_tentative_nearest_neighbors(landmarks, observations, observation_confidences, pose,
158 nearest_landmarks, nearest_distances);
159
160 const Eigen::VectorXi seed_observation_indices =
161 get_seed_observation_indices(nearest_landmarks, nearest_distances);
162
163 auto try_pair = [&](const int first_seed_idx, const int second_seed_idx) {
164 Eigen::Vector3d transform;
165 if (!estimate_transform_from_pair(
166 landmarks, observations, seed_observation_indices(first_seed_idx),
167 2 * nearest_landmarks(seed_observation_indices(first_seed_idx)),
168 seed_observation_indices(second_seed_idx),
169 2 * nearest_landmarks(seed_observation_indices(second_seed_idx)), transform)) {
170 return;
171 }
172
173 const Eigen::VectorXd adjusted_observations = adjust_observations(observations, transform);
174 const Eigen::VectorXi associations = nearest_neighbor.associate(
175 landmarks, adjusted_observations, covariance, observation_confidences, pose);
176
177 int num_associations = 0;
178 double total_distance = 0.0;
179 score_associations(landmarks, adjusted_observations, associations, num_associations,
180 total_distance);
181
182 if (num_associations > best_num_associations ||
183 (num_associations == best_num_associations && total_distance < best_total_distance)) {
184 best_associations = associations;
185 best_num_associations = num_associations;
186 best_total_distance = total_distance;
187 }
188 };
189
190 for (int first_seed_idx = 0; first_seed_idx < seed_observation_indices.size(); ++first_seed_idx) {
191 for (int second_seed_idx = first_seed_idx + 1;
192 second_seed_idx < seed_observation_indices.size(); ++second_seed_idx) {
193 try_pair(first_seed_idx, second_seed_idx);
194 }
195 }
196
197 return best_associations;
198}
