64bool MPC::stopping_the_car() {
65 double speed_threshold = 0.4; // m/s
66
67 // Check current speed
68 if (std::fabs(this->solver_state_.velocity_x) > speed_threshold) {
69 return false;
70 }
71
72 // Check average goal speed over the horizon
73 double goal_speed_average = 0;
74 for (size_t i = 0; i <= this->params_->mpc_prediction_horizon_steps_; ++i) {
75 goal_speed_average += this->path_data.pathpoint_array[i].v;
76 }
77 goal_speed_average /= (this->params_->mpc_prediction_horizon_steps_ + 1);
78 if (goal_speed_average > speed_threshold) {
79 return false;
80 }
81
82 // Check if speed is tending towards 0: calculate average speed goal variation
83 double speed_variation_average = 0;
84 for (size_t i = 0; i < this->params_->mpc_prediction_horizon_steps_; ++i) {
85 double current_goal_speed = this->path_data.pathpoint_array[i].v;
86 double next_goal_speed = this->path_data.pathpoint_array[i + 1].v;
87 speed_variation_average += (next_goal_speed - current_goal_speed);
88 }
89 speed_variation_average /= this->params_->mpc_prediction_horizon_steps_;
90 double current_velocity = this->solver_state_.velocity_x;
91 double allowed_negative_speed_due_to_noise = -0.1;
92 if (speed_variation_average > 0 && current_velocity > allowed_negative_speed_due_to_noise) {
93 return false;
94 }
95
96 return true;
97}
