64void AcadosSolver::initialize_solver_memory() {
65 int N = this->control_params_->mpc_prediction_horizon_steps_;
66 double u_zero[3] = {0.5, 0.5, 0.0}; // Baseline control guess
67 double time_step = this->control_params_->mpc_prediction_horizon_seconds_ / static_cast<double>(N);
68
69 double wheel_radius = 0.203;
70 double wheelbase = 1.50;
71
72 // Fill the state guess across the entire horizon (stages 0 to N)
73 double next_yaw = 0;
74 double next_v = 0;
75 double yaw_rate = 0;
76 double acceleration = 0;
77 for (int i = 0; i <= N; ++i) {
78 double x = this->parameters_per_stage[i*4];
79 double y = this->parameters_per_stage[i*4 + 1];
80 double v = this->parameters_per_stage[i*4 + 2];
81 double yaw = this->parameters_per_stage[i*4 + 3];
82
83 if (i != N) {
84 next_yaw = this->parameters_per_stage[(i+1)*4 + 3];
85 next_v = this->parameters_per_stage[(i+1)*4 + 2];
86 yaw_rate = (next_yaw - yaw) / time_step;
87 acceleration = (next_v - v) / time_step;
88 }
89
90 double steering = std::atan(wheelbase * yaw_rate / (v + 1e-6)); // Simple bicycle model for steering angle
91
92 double wheel_speed_radians_scaled = (v / wheel_radius) / this->control_params_->wheel_speeds_scale_mpc_; // Convert linear velocity to angular velocity
93 double state_guess[13] = {x, y, yaw, v, 0.0, yaw_rate, acceleration, 0.0, steering, wheel_speed_radians_scaled, wheel_speed_radians_scaled , wheel_speed_radians_scaled, wheel_speed_radians_scaled}; // [x, y, theta, v, ...] with some initial guess for velocities and other states
94 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, i, "x", (void*)state_guess);
95 }
96
97 // Fill the control guess across the entire horizon (stages 0 to N-1)
98 for (int i = 0; i < N; ++i) {
99 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, i, "u", (void*)u_zero);

174common_lib::structures::ControlCommand AcadosSolver::solve(int* solver_status) {
175 common_lib::structures::ControlCommand command;
176 if (!(this->has_state_ && this->has_path_)) {
177 return command;
178 }
179
180 this->set_path_point_per_stage();
181
182 double first_x = this->parameters_per_stage[0];
183 double first_y = this->parameters_per_stage[1];
184 double first_v = this->parameters_per_stage[2];
185 double first_orientation = this->parameters_per_stage[3];
186 first_x -= this->latest_state_.x;
187 first_y -= this->latest_state_.y;
188 first_orientation -= this->latest_state_.orientation;
189 first_v -= this->latest_state_.velocity_x;
190 //DEBUG PRINT
191 if (std::fabs(first_x) > 0.01 || std::fabs(first_y) > 0.01 || std::fabs(first_v) > 0.01 || std::fabs(first_orientation) > 0.01) {
192 RCLCPP_ERROR(rclcpp::get_logger("AcadosSolver"), "ERROR: first point doens't match state x:%.2f, y:%.2f, v:%.2f, orientation:%.2f", this->latest_state_.x, this->latest_state_.y, this->latest_state_.velocity_x, this->latest_state_.orientation);
193 }
194
195
196 if (!this->is_initialized_) {
197 this->initialize_solver_memory();
198 }
199
200 int status = bombated_mpc_acados_solve(this->capsule_);
201 if (status != ACADOS_SUCCESS) {
202 RCLCPP_ERROR(rclcpp::get_logger("AcadosSolver"), "Acados solver failed with status %d", status);
203 if (this->sanity_check_output()) {
204 *solver_status = 1; // Positive to indicate benign failure (e.g. infeasibility)
205 } else {
206 print_debug_info(); // If solver failed, print debug info
207 *solver_status = -1; // Negative to indicate malign failure
208 }
209 }
210
211 this->update_mpc_stats();
212 // Extracting 3 controls
213 std::vector<common_lib::structures::ControlCommand> full_solution = this->get_full_solution();
214
215 // Compensate for solver delay
216 double total_solver_time_ms = this->_execution_times_->at(0);
217 // Add an extra 5 ms as estimate for state estimation delay
218 double total_delay_ms = total_solver_time_ms + 5.0;
219 double total_delay_s = total_delay_ms / 1000.0;
220
221 double time_step = this->control_params_->mpc_prediction_horizon_seconds_ / static_cast<double>(this->control_params_->mpc_prediction_horizon_steps_);
222 unsigned int steps_ahead = static_cast<unsigned int>(std::floor(total_delay_s / time_step));
223 if (steps_ahead >= full_solution.size() - 1) {
224 RCLCPP_WARN(rclcpp::get_logger("AcadosSolver"), "Total delay of %.2f ms exceeds prediction horizon, using last available control", total_delay_ms);
225 return full_solution.back();
226 }
227
228 common_lib::structures::ControlCommand command_zero = full_solution[steps_ahead];
229 common_lib::structures::ControlCommand command_one = full_solution[steps_ahead + 1];
230
231 // Linear interpolation of control commands
232 double alpha = (total_delay_s - steps_ahead * time_step) / time_step;
233
234 // DEBUG STRING
235 this->total_delay_debug = "Total delay: " + std::to_string(total_delay_ms) + " ms, steps ahead: " + std::to_string(steps_ahead) + ", interpolation alpha: " + std::to_string(alpha);
236
237 command.throttle_rl = (1 - alpha) * command_zero.throttle_rl + alpha * command_one.throttle_rl;
238 command.throttle_rr = (1 - alpha) * command_zero.throttle_rr + alpha * command_one.throttle_rr;
239 command.steering_angle = (1 - alpha) * command_zero.steering_angle + alpha * command_one.steering_angle;
240 return command;
241}
