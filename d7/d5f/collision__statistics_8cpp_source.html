114bool CollisionStatistics::collides_with_cone(const VehicleModelSnapshot& vehicle_snapshot,
115 const common_lib::structures::Cone& cone) const {
116 const double dx = cone.position.x - vehicle_snapshot.x;
117 const double dy = cone.position.y - vehicle_snapshot.y;
118 const double cos_yaw = std::cos(vehicle_snapshot.yaw);
119 const double sin_yaw = std::sin(vehicle_snapshot.yaw);
120 const double local_x = cos_yaw * dx + sin_yaw * dy;
121 const double local_y = -sin_yaw * dx + cos_yaw * dy;
122
123 const double radius = cone_radius(cone);
124 for (const Hitbox& hitbox : car_hitboxes_) {
125 const double min_x = hitbox.center_x - hitbox.half_length;
126 const double max_x = hitbox.center_x + hitbox.half_length;
127 const double min_y = hitbox.center_y - hitbox.half_width;
128 const double max_y = hitbox.center_y + hitbox.half_width;
129 const double closest_x = std::clamp(local_x, min_x, max_x);
130 const double closest_y = std::clamp(local_y, min_y, max_y);
131 if (std::hypot(local_x - closest_x, local_y - closest_y) <= radius) {
132 return true;
133 }
134 }
135
136 return false;
137}
