147double KeyboardInputAdapter::approach(double current, double target, double max_delta) {
148 if (current < target) {
149 return std::min(current + max_delta, target);
150 }
151 if (current > target) {
152 return std::max(current - max_delta, target);
153 }
154 return current;
155}

157void KeyboardInputAdapter::input_loop() {
158 double throttle = 0.0;
159 double steering = 0.0;
160
161 while (running_) {
162 SDL_Event event;
163 while (SDL_PollEvent(&event) != 0) {
164 if (event.type == SDL_QUIT) {
165 running_ = false;
166 } else if (event.type == SDL_KEYDOWN && event.key.keysym.sym == SDLK_q) {
167 running_ = false;
168 }
169 }
170
171 SDL_PumpEvents();
172 const Uint8* keys = SDL_GetKeyboardState(nullptr);
173
174 const bool accelerate = keys[SDL_SCANCODE_W];
175 const bool brake = keys[SDL_SCANCODE_S];
176 const bool steer_left = keys[SDL_SCANCODE_A];
177 const bool steer_right = keys[SDL_SCANCODE_D];
178
179 double target_throttle = 0.0;
180 if (accelerate && !brake) {
181 target_throttle = max_throttle_;
182 } else if (brake && !accelerate) {
183 target_throttle = -max_throttle_;
184 }
185
186 double target_steering = 0.0;
187 if (steer_left && !steer_right) {
188 target_steering = max_steering_radians_;
189 } else if (steer_right && !steer_left) {
190 target_steering = -max_steering_radians_;
191 }
192
193 throttle = approach(throttle, target_throttle, throttle_step_);
194 steering = approach(steering, target_steering, steering_step_);
195
196 common_lib::structures::Wheels throttle_command{throttle, throttle, throttle, throttle};
197 simulator_->set_input(throttle_command, steering);
198
199 if (renderer_ != nullptr) {
200 render_bars(throttle, steering);
201 }
202
203 std::this_thread::sleep_for(std::chrono::milliseconds(loop_period_ms_));
204 }
205}
