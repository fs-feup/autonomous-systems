153 def build_cones(self):
154 cones = []
155 seen = set()
156 unknown_count = 0
157
158 for cone_msg in self.latest_map.cone_array:
159 x = float(cone_msg.position.x)
160 y = float(cone_msg.position.y)
161 key = (round(x, 3), round(y, 3))
162 if key in seen:
163 continue
164
165 seen.add(key)
166 color = self.find_planning_color(x, y)
167 if color is None:
168 color = COLOR_ALIASES.get(cone_msg.color, cone_msg.color or "unknown_cone")
169 if color == "unknown_cone":
170 unknown_count += 1
171 cones.append([x, y, color])
172
173 cones.sort(key=lambda cone: (cone[2], cone[0], cone[1]))
174 return cones, unknown_count

176 def find_planning_color(self, x, y):
177 best_color = None
178 best_distance_sq = MATCH_DISTANCE_METERS * MATCH_DISTANCE_METERS
179
180 for color, markers in self.colored_cones.items():
181 for marker_x, marker_y in markers:
182 distance_sq = (x - marker_x) ** 2 + (y - marker_y) ** 2
183 if distance_sq <= best_distance_sq:
184 best_distance_sq = distance_sq
185 best_color = color

209 def timing_line_from_pose(pose):

211
212 normal_theta = pose.theta + math.pi / 2.0
213 dx = TIMING_LINE_HALF_WIDTH_METERS * math.cos(normal_theta)
