13std::vector<custom_interfaces::msg::PathPoint> create_path_msg(std::string track_file) {
14 std::string track_file_path = "../../src/control/test/assets/" + track_file + ".csv";
15 std::ifstream trackFile(track_file_path);
16
17 std::vector<custom_interfaces::msg::PathPoint> pathpoint_array;
18
19 std::string line;
20
21 std::getline(trackFile, line); // Skip the first line
22
23 while (std::getline(trackFile, line)) {
24 std::istringstream iss(line);
25 std::string x, y, v;
26 if (std::getline(iss, x, ',') && std::getline(iss, y, ',') && std::getline(iss, v)) {
27 // Create a PathPoint and add it to the PathPointArray
28 custom_interfaces::msg::PathPoint point;
29 point.x = std::stod(x); // string to double
30 point.y = std::stod(y);
31 point.v = std::stod(v);
32 pathpoint_array.push_back(point);
33 }
34 }
35 trackFile.close();
36 return pathpoint_array;
37};

79static std::vector<custom_interfaces::msg::PathPoint> straight_path(
80 const std::vector<double>& velocities) {
81 std::vector<custom_interfaces::msg::PathPoint> path;
82 for (size_t i = 0; i < velocities.size(); i++) {
83 custom_interfaces::msg::PathPoint point;
84 point.x = static_cast<double>(i);
85 point.y = 0.0;
86 point.v = velocities[i];
87 path.push_back(point);
88 }
89 return path;
90}
