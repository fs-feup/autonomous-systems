8Inputs are the artefacts the optimizer writes to performance/invictasim_tuning/:

32def plot_convergence(log_path, out_path):
33 rows = read_csv(log_path)
34 if not rows:
35 print("empty convergence log")
36 return
37 t = [float(r["elapsed_s"]) for r in rows]
38 ev = [float(r["evals"]) for r in rows]
39 score = [float(r["score"]) for r in rows]
40 pos = [float(r["position_rmse"]) for r in rows]
41 vel = [float(r["velocity_rmse"]) for r in rows]
42 # score ~ survival_seconds * 1e6 (weighted); expose approximate survival.
43 surv = [s / 1e6 for s in score]
44
45 fig, ax = plt.subplots(2, 2, figsize=(13, 8))
46 ax[0, 0].plot(t, surv, "-o", ms=3, color="#2563eb")
47 ax[0, 0].set(xlabel="wall time [s]", ylabel="weighted survival [s]",
48 title="Survival vs wall time")
49 ax[0, 1].plot(ev, surv, "-o", ms=3, color="#2563eb")
50 ax[0, 1].set(xlabel="evaluations", ylabel="weighted survival [s]",
51 title="Survival vs evaluations")
52 ax[1, 0].plot(t, pos, "-o", ms=3, color="#dc2626", label="position RMSE [m]")
53 ax[1, 0].plot(t, vel, "-o", ms=3, color="#16a34a", label="velocity RMSE [m/s]")
54 ax[1, 0].set(xlabel="wall time [s]", ylabel="RMSE", title="Tracking error of incumbent")
55 ax[1, 0].legend()
56 ax[1, 1].axis("off")
57 ax[1, 1].text(0.02, 0.95,
58 f"improvements logged: {len(rows)}\n"
59 f"final weighted survival: {surv[-1]:.2f} s\n"
60 f"final position RMSE: {pos[-1]:.3f} m\n"
61 f"final velocity RMSE: {vel[-1]:.3f} m/s\n"
62 f"total evaluations: {int(ev[-1])}\n"
63 f"wall time: {t[-1]:.0f} s",
64 va="top", family="monospace", fontsize=11)
65 for a in ax.flat:
66 a.grid(alpha=0.3)
67 fig.tight_layout()
68 fig.savefig(out_path, dpi=110)
69 print("wrote", out_path)

72def plot_trajectory(diag_path, out_path):
73 rows = read_csv(diag_path)

87 # Trim to the survived portion (first limit crossing) for a readable path plot.
88 surv_n = len(rows)
89 for i, (p, v) in enumerate(zip(pe, ve)):

91 surv_n = i
92 break
93
94 fig, ax = plt.subplots(2, 2, figsize=(13, 9))
95 ax[0, 0].plot(rx[:surv_n], ry[:surv_n], color="#111", lw=2, label="real")

97 ax[0, 0].set(xlabel="x [m]", ylabel="y [m]",
98 title=f"Path (survived {t[surv_n-1] if surv_n else 0:.1f}s)")
