81void Track::fit_timing_line_to_track_width() {
82 if (timing_lines_.empty()) {
83 return;
84 }
85
86 auto& timing_line = timing_lines_.front();
87 const double line_x = timing_line.second.x - timing_line.first.x;
88 const double line_y = timing_line.second.y - timing_line.first.y;
89 const double line_length = std::hypot(line_x, line_y);
90 if (line_length <= std::numeric_limits<double>::epsilon() || cones_.size() < 2) {
91 return;
92 }
93
94 const common_lib::structures::Position center(0.5 * (timing_line.first.x + timing_line.second.x),
95 0.5 * (timing_line.first.y + timing_line.second.y));
96 const double axis_x = line_x / line_length;
97 const double axis_y = line_y / line_length;
98 const double normal_x = -axis_y;
99 const double normal_y = axis_x;

101 struct Candidate {
102 double along = 0.0;
103 double score = std::numeric_limits<double>::max();
104 };
105
106 std::optional<Candidate> blue_candidate;
107 std::optional<Candidate> yellow_candidate;
108 std::optional<Candidate> best_any_candidate;
109 std::optional<Candidate> second_any_candidate;
110
111 auto consider_boundary = [](std::optional<Candidate>& candidate, double along, double score) {
112 if (!candidate || score < candidate->score) {
113 candidate = Candidate{along, score};
114 }
115 };
116
117 for (const auto& cone : cones_) {
118 const double rel_x = cone.position.x - center.x;
119 const double rel_y = cone.position.y - center.y;
120 const double along = rel_x * axis_x + rel_y * axis_y;
121 const double plane_distance = std::abs(rel_x * normal_x + rel_y * normal_y);
122 const double score = plane_distance + 0.25 * std::abs(along);
123
124 if (cone.color == common_lib::competition_logic::Color::BLUE) {
125 consider_boundary(blue_candidate, along, score);
126 } else if (cone.color == common_lib::competition_logic::Color::YELLOW) {
127 consider_boundary(yellow_candidate, along, score);
128 }
129
130 Candidate current{along, score};
131 if (!best_any_candidate || score < best_any_candidate->score) {
132 second_any_candidate = best_any_candidate;
133 best_any_candidate = current;
134 } else if ((!second_any_candidate || score < second_any_candidate->score) &&
135 (!best_any_candidate || std::abs(along - best_any_candidate->along) >
136 common_lib::structures::Cone::equality_tolerance)) {
137 second_any_candidate = current;
138 }
139 }
140
141 std::optional<double> lower_along;
142 std::optional<double> upper_along;
143 if (blue_candidate && yellow_candidate) {
144 lower_along = std::min(blue_candidate->along, yellow_candidate->along);
145 upper_along = std::max(blue_candidate->along, yellow_candidate->along);
146 } else if (best_any_candidate && second_any_candidate) {
147 lower_along = std::min(best_any_candidate->along, second_any_candidate->along);
148 upper_along = std::max(best_any_candidate->along, second_any_candidate->along);
149 }
150
151 if (!lower_along || !upper_along ||
152 (*upper_along - *lower_along) <= std::numeric_limits<double>::epsilon()) {
153 return;
154 }
155
156 timing_line.first = common_lib::structures::Position(center.x + axis_x * *lower_along,
157 center.y + axis_y * *lower_along);
158 timing_line.second = common_lib::structures::Position(center.x + axis_x * *upper_along,
159 center.y + axis_y * *upper_along);
160}
