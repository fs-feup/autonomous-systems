75def filter_window(commands, steering, start_s, duration_s, min_abs_steering):
76 if start_s is not None:

90def simulate_first_order(commands, sample_times, tau, initial_value):
91 values = []
92 current = initial_value
93 last_t = sample_times[0]
94 for t in sample_times:
95 dt = max(0.0, t - last_t)
96 target = command_at(commands, t)
97 if tau <= 1e-9:
98 current = target
99 else:
100 alpha = 1.0 - math.exp(-dt / tau)
101 current += alpha * (target - current)
102 values.append(current)
103 last_t = t
104 return values

107def rmse(errors):
108 if not errors:
109 return float("inf")
110 return math.sqrt(sum(error * error for error in errors) / len(errors))

113def score_tau(commands, steering, tau):
114 sample_times = [t for t, _ in steering]
115 real_values = [value for _, value in steering]
116 simulated = simulate_first_order(commands, sample_times, tau, real_values[0])
117 return rmse([sim - real for sim, real in zip(simulated, real_values)])

120def linear_fit(xs, ys):
121 n = len(xs)
122 if n == 0:
123 return None
124 mean_x = sum(xs) / n
125 mean_y = sum(ys) / n
126 var_x = sum((x - mean_x) ** 2 for x in xs)
127 if var_x <= 1e-12:
128 return None
129 cov_xy = sum((x - mean_x) * (y - mean_y) for x, y in zip(xs, ys))
130 scale = cov_xy / var_x
131 offset = mean_y - scale * mean_x
132 errors = [scale * x + offset - y for x, y in zip(xs, ys)]
133 return {
134 "scale": scale,
135 "offset": offset,
136 "rmse": rmse(errors),
137 }

140def fit_time_constant(commands, steering, tau_min, tau_max):
141 best = {"time_constant": None, "rmse": float("inf")}
142 low = tau_min
143 high = tau_max
144 for _ in range(5):
145 steps = 80
146 candidates = [low + (high - low) * i / steps for i in range(steps + 1)]
147 for tau in candidates:
148 score = score_tau(commands, steering, tau)
149 if score < best["rmse"]:
150 best = {"time_constant": tau, "rmse": score}
151 span = (high - low) / 8.0
152 low = max(tau_min, best["time_constant"] - span)
153 high = min(tau_max, best["time_constant"] + span)
154 return best
