1#!/usr/bin/env python3
2"""One-at-a-time (finite-difference) sensitivity screening for the 02 tuning set.
3
4For every optimization parameter we perturb it by +/- a fraction of its search
5range around the current baseline and measure how the *survival time* of the
6trackdrive replay changes (survival = first time either the 1.5 m position or
72.0 m/s velocity limit is exceeded, as reported by `--diagnose`). Parameters
8whose perturbation moves survival the most are the ones the optimizer should
9spend its budget on; parameters with ~0 effect can be frozen.
10
11This reuses the compiled optimizer's `--diagnose --set name=value` path, so it
12measures sensitivity of the exact production physics, not a re-implementation.
13
14Usage:
15 python3 sensitivity_analysis.py [--frac 0.15] [--out sensitivity.csv]
16"""
17import argparse
18import re
19import subprocess

25WS = Path("/home/ws")
26BIN = WS / "install/invictasim/lib/invictasim/vehicle_model_optimizer"
27CFG = WS / "config/invictasim/tuning/02_feup_trackdrive_tuning.yaml"
28DATA = WS / "src/invictasim/tuning_csvs"
29PERF = WS / "performance/invictasim_tuning"

31VIOL_RE = re.compile(r"first (pos|vel)-limit .* violation: (never|[0-9.]+)s")
32
33
34def run_survival(overrides):
35 """Return trackdrive survival seconds for the given {name: value} overrides."""
36 cmd = ["bash", "-lc",
37 "source /opt/ros/humble/setup.bash && source /home/ws/install/setup.bash && " +
38 " ".join([str(BIN), "--config", str(CFG), "--data-dir", str(DATA), "--diagnose"] +
39 [f"--set {k}={v}" for k, v in overrides.items()])]
40 out = subprocess.run(cmd, capture_output=True, text=True).stdout
41 # First two violation lines belong to dataset 0 (trackdrive).
42 pos = vel = None
43 for m in VIOL_RE.finditer(out):
44 val = float("inf") if m.group(2) == "never" else float(m.group(2))
45 if m.group(1) == "pos" and pos is None:
46 pos = val
47 elif m.group(1) == "vel" and vel is None:
48 vel = val
49 if pos is not None and vel is not None:
50 break
51 if pos is None or vel is None:
52 return 0.0
53 return min(pos, vel)

56def main():
57 ap = argparse.ArgumentParser()
58 ap.add_argument("--frac", type=float, default=0.15,
59 help="perturbation as fraction of each parameter's search range")
60 ap.add_argument("--out", default=str(PERF / "sensitivity.csv"))
61 args = ap.parse_args()
62
63 cfg = yaml.safe_load(open(CFG))
64 params = cfg["tuning"]["parameters"]
65
66 base = run_survival({})
67 print(f"baseline trackdrive survival: {base:.3f} s\n")
68 print(f"{'parameter':45s} {'-frac':>8s} {'+frac':>8s} {'sensitivity':>12s}")
69
70 results = []
71 for p in params:
72 name = p["name"]
73 lo, hi = float(p["min"]), float(p["max"])
74 step = args.frac * (hi - lo)
75 # perturb around baseline value read back from a neutral --diagnose run is
76 # awkward, so perturb around mid-range-clamped baseline via +/- step of the
77 # live config value (the optimizer applies the baseline from config).
78 # We approximate the baseline value as the config-loaded value by using two
79 # symmetric sets offset by the step directly on top of the model default.
80 # To keep it simple and robust we set the parameter to (default +/- step)
81 # using the same --set mechanism; the default is whatever the config holds.
82 # Read default by probing is unnecessary because --set overrides absolutely,
83 # so we instead sweep to lo+step and hi-step ends of range for a coarse but
84 # meaningful gradient.
85 v_minus = lo + step
86 v_plus = hi - step
87 s_minus = run_survival({name: v_minus})
88 s_plus = run_survival({name: v_plus})
89 sens = abs(s_plus - s_minus)
90 results.append((name, v_minus, s_minus, v_plus, s_plus, sens))
91 print(f"{name:45s} {s_minus:8.3f} {s_plus:8.3f} {sens:12.3f}")
92
93 results.sort(key=lambda r: r[5], reverse=True)
94 with open(args.out, "w") as f:
95 f.write("parameter,val_lo,surv_lo,val_hi,surv_hi,sensitivity\n")
96 for r in results:
97 f.write(f"{r[0]},{r[1]},{r[2]},{r[3]},{r[4]},{r[5]}\n")
98 print(f"\nRanked sensitivity written to {args.out}")
99 print("\nTop influential parameters:")
100 for r in results[:12]:
101 print(f" {r[0]:45s} range-sweep survival delta = {r[5]:.3f} s")
102

104if __name__ == "__main__":
105 main()
